module tb;
localparam DATA_WIDTH=8,KEEP_WIDTH=1,ID_WIDTH=8,DEST_WIDTH=8,USER_WIDTH=1,DEPTH=4096,EMPTY_WIDTH=0,ERROR_WIDTH=0,CHANNEL_WIDTH=0,USE_STORE_FORWARD=0;

reg reset;
reg [DATA_WIDTH-1: 0]     in_data;
reg in_valid;
reg in_startofpacket;
reg in_endofpacket;
reg [((EMPTY_WIDTH>0) ? (EMPTY_WIDTH-1):0) : 0]     in_empty;
reg [((ERROR_WIDTH>0) ? (ERROR_WIDTH-1):0) : 0]     in_error;
reg [((CHANNEL_WIDTH>0) ? (CHANNEL_WIDTH-1):0): 0]  in_channel;
wire in_ready;

wire out_valid;
wire out_startofpacket;
wire out_endofpacket;
wire [((EMPTY_WIDTH>0) ? (EMPTY_WIDTH-1):0) : 0]    out_empty;
wire [((ERROR_WIDTH>0) ? (ERROR_WIDTH-1):0) : 0]    out_error;
wire [((CHANNEL_WIDTH>0) ? (CHANNEL_WIDTH-1):0): 0] out_channel;
reg out_ready;

altera_avalon_sc_fifo dut(.clk(clk),.reset(reset),.in_data(in_data),.in_valid(in_valid),.in_startofpacket(in_startofpacket),.in_endofpacket(in_endofpacket),.in_empty(in_empty),.in_error(in_error),.in_channel(in_channel),.in_ready(in_ready),.out_data(out_data),.out_valid(out_valid),.out_startofpacket(out_startofpacket),.out_endofpacket(out_endofpacket),.out_empty(out_empty),.out_error(out_error),.out_channel(out_channel),.out_ready(out_ready),.csr_address(csr_address),.csr_write(csr_write),.csr_read(csr_read),.csr_writedata(csr_writedata),.csr_readdata(csr_readdata),.almost_full_data(almost_full_data),.almost_empty_data(almost_empty_data));

clk='0;

in_data='0;
in_valid='0;
in_startofpacket='0;
in_endofpacket='0;
in_empty='0;
in_error='0;
in_channel='0;
out_ready='0;
csr_address='0;
csr_write='0;
csr_read='0;
csr_writedata='0;
reset=1;clk=0;#5;clk=1;#5;clk=0;reset=0;
for(n=0;n<20000;n=n+1)begin
 clk=0;in_valid=(n<37)?1:((n<14000)?($random&1):0);out_ready=(n<37)?0:((n<14000)?($random&1):1);in_data=$random;
 in_error=$random;in_channel=$random;in_empty=$random;csr_read=$random;csr_write=$random;csr_address=1;csr_writedata=$random;
 #5;
 if(out_valid && out_ready)begin
 if(rd==wr || out_data!==q[rd])$fatal(1,"queue mismatch cycle=%0d rd=%0d wr=%0d got=%h expect=%h",n,rd,wr,out_data,q[rd]);
 if(out_startofpacket!==0 || out_endofpacket!==0 || out_error!==in_error || out_empty!==in_error || out_channel!==in_channel || almost_full_data!==0 || almost_empty_data!==0 || csr_readdata!==0)$fatal(1,"sideband mismatch");
 rd=rd+1;end
 if(in_valid && in_ready)begin q[wr]=in_data; wr=wr+1;end
 if(n==36 && (wr!=17 || in_ready!==0))$fatal(1,"capacity mismatch wr=%0d",wr);
 clk=1;#5;
end
if(rd!=wr)$fatal(1,"did not drain");
$display("PASS altera_avalon_sc_fifo 20000 cycles; capacity 17; randomized stalls and payload queue; accepted=%0d read=%0d",wr,rd);$finish;
