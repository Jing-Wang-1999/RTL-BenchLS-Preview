module origen_049589 (
    input [15:0] PORT1,
    input [15:0] PORT2,
    input [2:0] ALUCON,
    output reg [15:0] ALUOUT,
    output reg carry
);

    reg [16:0] temp_result; // 17-bit temporary for arithmetic

    always @(*) begin
        ALUOUT = 16'b0;
        carry = 1'b0;
        temp_result = 17'b0;

        case (ALUCON)
            3'b000: begin // Addition
                temp_result = {1'b0, PORT1} + {1'b0, PORT2};
                ALUOUT = temp_result[15:0];
                // Signed overflow detection: (A>0 && B>0 && R<0) || (A<0 && B<0 && R>=0)
                if ((~PORT1[15] & ~PORT2[15] & ALUOUT[15]) || (PORT1[15] & PORT2[15] & ~ALUOUT[15])) begin
                    carry = 1'b1;
                end else begin
                    carry = 1'b0;
                end
            end
            3'b001: begin // Subtraction
                temp_result = {1'b0, PORT1} - {1'b0, PORT2};
                ALUOUT = temp_result[15:0];
                // Signed overflow detection: (A>0 && B<0 && R<0) || (A<0 && B>0 && R>=0)
                if ((~PORT1[15] & PORT2[15] & ALUOUT[15]) || (PORT1[15] & ~PORT2[15] & ~ALUOUT[15])) begin
                    carry = 1'b1;
                end else begin
                    carry = 1'b0;
                end
            end
            3'b010: begin // Bitwise AND
                ALUOUT = PORT1 & PORT2;
                carry = 1'b0;
            end
            3'b011: begin // Bitwise OR
                ALUOUT = PORT1 | PORT2;
                carry = 1'b0;
            end
            3'b100: begin // Bitwise XOR
                ALUOUT = PORT1 ^ PORT2;
                carry = 1'b0;
            end
            3'b101: begin // Bitwise NOT (PORT1 only)
                ALUOUT = ~PORT1;
                carry = 1'b0;
            end
            3'b110: begin // Left shift by one
                ALUOUT = PORT1 << 1;
                carry = 1'b0;
            end
            3'b111: begin // Right shift by one
                ALUOUT = PORT1 >> 1;
                carry = 1'b0;
            end
            default: begin // Undefined ALUCON
                ALUOUT = 16'b0;
                carry = 1'b0;
            end
        endcase
    end

endmodule
