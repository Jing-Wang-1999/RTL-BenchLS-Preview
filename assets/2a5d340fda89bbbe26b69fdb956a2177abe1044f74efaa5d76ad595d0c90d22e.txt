#!/usr/bin/env python3
"""Replay pinned whole upstream benches with semantic completion checks."""
import argparse,json,shutil,subprocess,os,re,time
from pathlib import Path

def judge(meta,text,rc,source):
 if rc!=0:return False
 if meta['protocol']=='aes':return re.findall(r'Test Done\. Found (\d+) Errors\.',text)==['0'] and 'ERROR:' not in text
 if meta['protocol']=='cordic':
  table={int(a):(int(x),int(y)) for a,x,y in re.findall(r"x\[(\d+)\]\s*<=\s*17'd(\d+);\s*y\[\d+\]\s*<=\s*17'd(\d+)",(source/'tb_cordic.v').read_text())}
  rows=re.findall(r'Angle:\s+(\d+)\s+sin = ([\d.-]+)\s+cos = ([\d.-]+)',text)
  return len(rows)==91 and [int(r[0]) for r in rows]==list(range(91)) and all(abs(float(sn)*32768-table[int(a)][1])<=10.02 and abs(float(cs)*32768-table[int(a)][0])<=10.02 for a,sn,cs in rows)
 if meta['protocol']=='i2c':
  reads=re.findall(r'I2C read 0x([0-9a-f]{2})',text);acks=re.findall(r'I2C (N?ACK) at',text)
  return 'Testbench done.' in text and reads==['00','00','00','11','11','11','22','ff','ff','be','ef','ff','ff'] and acks==['ACK','NACK','NACK','NACK','ACK','ACK','ACK','NACK','ACK','NACK']
 raise ValueError(meta['protocol'])
def main():
 a=argparse.ArgumentParser();a.add_argument('--payload',type=Path,required=True);a.add_argument('--candidate-dir',type=Path,required=True);a.add_argument('--output',type=Path,required=True);z=a.parse_args();m=json.loads((z.payload/'meta.json').read_text());z.output.mkdir(parents=True,exist_ok=False);s=z.output/'source';shutil.copytree(z.payload/'source',s)
 for f in m['rtl_files']:shutil.copyfile(z.candidate_dir/f,s/f)
 cmd=[m['iverilog'],'-o',str((z.output/'sim.vvp').resolve())]+m['compile_args'];t=time.monotonic();c=subprocess.run(cmd,cwd=s,capture_output=True,text=True,timeout=90);(z.output/'compile.log').write_text(c.stdout+c.stderr);r=dict(compile_pass=c.returncode==0,compile_returncode=c.returncode,compile_command=cmd,passed_testpoints=0,total_testpoints=1,score=0,timeout=False)
 if c.returncode==0:
  try:
   v=subprocess.run([m['vvp'],'-n',str((z.output/'sim.vvp').resolve())],cwd=s,capture_output=True,text=True,timeout=120,env={**os.environ,'IVERILOG_DUMPER':'none'});txt=v.stdout+v.stderr;(z.output/'run.log').write_text(txt);ok=judge(m,txt,v.returncode,s);r.update(passed_testpoints=int(ok),score=int(ok),returncode=v.returncode)
  except subprocess.TimeoutExpired as e:(z.output/'run.log').write_bytes((e.stdout or b'')+(e.stderr or b''));r['timeout']=True
 r['elapsed_seconds']=round(time.monotonic()-t,3);(z.output/'result.json').write_text(json.dumps(r,indent=2)+'\n');print(json.dumps(r))
if __name__=='__main__':main()
