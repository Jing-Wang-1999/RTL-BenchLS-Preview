module rtlpp_03901 (
    input  wire areset,
    input  wire inclk0,
    output wire c0,
    output reg  locked
);

    // Output clock is directly derived from input clock
    assign c0 = inclk0;

    // Lock detection logic
    always @(posedge inclk0 or posedge areset) begin
        if (areset) begin
            locked <= 1'b0;
        end else begin
            locked <= 1'b1;
        end
    end

endmodule
