module full_adder_32bit(
    input [31:0] a,
    input [31:0] b,
    input cin,
    output [31:0] sum,
    output cout
);

    assign {cout, sum} = a + b + cin;

endmodule

module full_subtractor_32bit(
    input [31:0] a,
    input [31:0] b,
    input bin,
    output [31:0] diff,
    output bout
);

    assign {bout, diff} = a - b - bin;

endmodule
