module mg_10431 (
    input [7:0] INP,
    input [7:0] INN,
    input CLK,
    input PWRDN,
    output reg GREATER,
    output reg EQUAL
);

    always @(posedge CLK) begin
        if (INP > INN) begin
            GREATER <= 1'b1;
            EQUAL <= 1'b0;
        end
        else if (INP == INN) begin
            GREATER <= 1'b0;
            EQUAL <= 1'b1;
        end
        else begin
            GREATER <= 1'b0;
            EQUAL <= 1'b0;
        end
    end

endmodule
