module origen_001099(
    input wire clk_m,
    input wire rst,
    input wire en,
    input wire CPOL,
    input wire CPH,
    input wire [2:0] clk_div,
    input wire [7:0] RX_data,
    input wire MISO,
    output reg [7:0] TX_data,
    output reg busy,
    output reg SS,
    output wire SCK,
    output reg MOSI
);

    localparam RESET = 2'b00;
    localparam IDLE = 2'b01;
    localparam RUNNING = 2'b10;

    reg [1:0] state, next_state;
    reg [2:0] bit_count;
    reg [7:0] data_in;
    wire clk_SPI;

    CLOCK clock_gen (
        .clk_m(clk_m),
        .clk_div(clk_div),
        .CPOL(CPOL),
        .clk_SPI(clk_SPI)
    );

    assign SCK = clk_SPI;

    always @(posedge clk_m or posedge rst) begin
        if (rst) begin
            state <= RESET;
        end else begin
            state <= next_state;
        end
    end

    always @(*) begin
        case (state)
            RESET: next_state = rst ? RESET : IDLE;
            IDLE: next_state = en ? RUNNING : IDLE;
            RUNNING: next_state = (bit_count == 3'b111 && clk_SPI == CPOL) ? IDLE : RUNNING;
            default: next_state = RESET;
        endcase
    end

    always @(posedge clk_m) begin
        case (state)
            RESET: begin
                TX_data <= 8'b0;
                busy <= 1'b0;
                SS <= 1'b1;
                MOSI <= 1'b0;
                bit_count <= 3'b0;
                data_in <= 8'b0;
            end
            IDLE: begin
                if (en) begin
                    busy <= 1'b1;
                    SS <= 1'b0;
                    data_in <= RX_data;
                    bit_count <= 3'b0;
                end
            end
            RUNNING: begin
                if (clk_SPI == ~CPOL) begin
                    MOSI <= data_in[7];
                    data_in <= {data_in[6:0], MISO};
                    bit_count <= bit_count + 1'b1;
                end
                if (bit_count == 3'b111 && clk_SPI == CPOL) begin
                    TX_data <= data_in;
                    busy <= 1'b0;
                    SS <= 1'b1;
                end
            end
        endcase
    end
endmodule

module CLOCK(
    input wire clk_m,
    input wire [2:0] clk_div,
    input wire CPOL,
    output reg clk_SPI
);

    reg [6:0] counter;
    wire [6:0] compare_value;

    assign compare_value = (7'b1 << clk_div) - 1'b1;

    always @(posedge clk_m) begin
        if (counter == compare_value) begin
            counter <= 7'b0;
            clk_SPI <= ~clk_SPI;
        end else begin
            counter <= counter + 1'b1;
        end
    end

    initial begin
        clk_SPI = CPOL;
        counter = 7'b0;
    end
endmodule
