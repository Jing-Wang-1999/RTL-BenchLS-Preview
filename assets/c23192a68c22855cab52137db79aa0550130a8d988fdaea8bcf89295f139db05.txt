always @(posedge clk or posedge rst) begin
        if (rst) begin
            tx_start <= 1'b0;
            tx_data <= 8'b0;
            rx_data_reg <= 8'b0;
            rx_valid_reg <= 1'b0;
        end else begin
            tx_start <= 1'b0;

            if (rx_valid_wire) begin
                rx_data_reg <= rx_data_wire;
                rx_valid_reg <= 1'b1;
            end

            if (we && addr[3:2] == 2'b00 && ~tx_busy) begin
                tx_data <= wdata[7:0];
                tx_start <= 1'b1;
            end

            if (re && addr[3:2] == 2'b01) begin
                rx_valid_reg <= 1'b0;
            end
        end
    end
