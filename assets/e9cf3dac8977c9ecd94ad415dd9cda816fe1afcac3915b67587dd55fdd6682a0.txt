def test(mode):
 s=HEAD;rng=random.Random(9060800+mode)
 if mode==0:
  s+='readcheck(32\'h60000008,255);readcheck(32\'h60000010,37);readcheck(32\'h60000018,32\'h800025);\n'
  for v in [0,0x12345678,0x3fffffff,0x222204]:
   s+=f"access(1,32'h60000014,32'h{v:x},15,0);readcheck(32'h60000014,32'h{v:x});\n"
  return s+END
 for i in range(48):
  region=i%2 if mode!=1 else 0;a=(0x10000000 if region else 0)+4*rng.randrange(4096);v=rng.getrandbits(32)
  # Independent expected memory starts with data unavailable to the agent.
  expected=(0xa5000000 if region else 0x5a000000)^(a//4&4095)
  if mode==1:s+=f"readcheck(32'h{a:x},32'h{expected:x});\n";continue
  s+=f"access(1,32'h{a:x},32'h{v:x},15,0);readcheck(32'h{a:x},32'h{v:x});\n"
  if mode==3:
   mask=rng.randrange(1,16);d=rng.getrandbits(32);bits=sum(255<<(8*k) for k in range(4) if mask>>k&1);v=(v&~bits)|(d&bits)
   s+=f"access(1,32'h{a:x},32'h{d:x},{mask},0);readcheck(32'h{a:x},32'h{v:x});\n"
 if mode==4:
  s+="access(1,32'h60000010,32'h125,15,0);access(1,0,32'hdeadbeef,15,1);readcheck(0,32'h5a000000);\n"
 s+='if(reads<1'+(' || writes<1' if mode!=1 else '')+')$fatal(1,"no physical traffic");\n'
 return s+END
