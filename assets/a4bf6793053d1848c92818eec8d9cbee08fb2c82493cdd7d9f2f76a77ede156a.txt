module origen_202046 (
    input [5:0] instr_op_i,
    output reg RegWrite_o,
    output reg [2:0] ALUOp_o,
    output reg ALUSrc_o,
    output reg RegDst_o,
    output reg Branch_o,
    output reg BranchType_o,
    output reg Jump_o,
    output reg MemRead_o,
    output reg MemWrite_o,
    output reg MemtoReg_o
);

    always @(*) begin
        // Default values (unrecognized opcode)
        RegWrite_o = 1'b0;
        ALUOp_o = 3'b000;
        ALUSrc_o = 1'b0;
        RegDst_o = 1'b0;
        Branch_o = 1'b0;
        BranchType_o = 1'b0;
        Jump_o = 1'b0;
        MemRead_o = 1'b0;
        MemWrite_o = 1'b0;
        MemtoReg_o = 1'b0;

        case (instr_op_i)
            // R-type instructions (opcode 000000)
            6'b000000: begin
                RegWrite_o = 1'b1;
                ALUOp_o = 3'b010;  // R-type mode
                ALUSrc_o = 1'b0;   // Register source
                RegDst_o = 1'b1;   // rd destination
                // All others default to 0
            end

            // Load word (LW) (opcode 100011)
            6'b100011: begin
                RegWrite_o = 1'b1;
                ALUOp_o = 3'b000;  // Add
                ALUSrc_o = 1'b1;   // Immediate source
                RegDst_o = 1'b0;   // rt destination
                MemRead_o = 1'b1;
                MemtoReg_o = 1'b1; // Memory write-back
            end

            // Store word (SW) (opcode 101011)
            6'b101011: begin
                ALUOp_o = 3'b000;  // Add
                ALUSrc_o = 1'b1;   // Immediate source
                MemWrite_o = 1'b1;
                // RegWrite_o remains 0 (default)
            end

            // Branch equal (BEQ) (opcode 000100)
            6'b000100: begin
                ALUOp_o = 3'b001;  // Subtract for comparison
                ALUSrc_o = 1'b0;   // Register source
                Branch_o = 1'b1;
                BranchType_o = 1'b0; // BEQ type
            end

            // Branch not equal (BNE) (opcode 000101)
            6'b000101: begin
                ALUOp_o = 3'b001;  // Subtract for comparison
                ALUSrc_o = 1'b0;   // Register source
                Branch_o = 1'b1;
                BranchType_o = 1'b1; // BNE type
            end

            // Jump (J) (opcode 000010)
            6'b000010: begin
                Jump_o = 1'b1;
                // ALUOp_o can be anything (unused)
            end

            // Jump and link (JAL) (opcode 000011)
            6'b000011: begin
                RegWrite_o = 1'b1;
                Jump_o = 1'b1;
                // ALUOp_o can be anything (unused)
                // Note: Typically RegDst would be set to write to $ra (register 31)
                // but specification doesn't specify this detail
            end

            // Default case handles unrecognized opcodes
            default: begin
                // All signals already set to default values
            end
        endcase
    end

endmodule
