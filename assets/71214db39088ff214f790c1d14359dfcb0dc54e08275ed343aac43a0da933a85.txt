module origen_162721(
    input clk,
    output reg hsync,
    output reg vsync,
    output reg [3:0] r,
    output reg [3:0] g,
    output reg [3:0] b
);

parameter hva = 640;
parameter hfp = 16;
parameter hsp = 96;
parameter hbp = 48;

parameter vva = 480;
parameter vfp = 10;
parameter vsp = 2;
parameter vbp = 33;

reg [10:0] hpos;
reg [9:0] vpos;

wire clkr;
assign clkr = ~clk;

always @(posedge clkr) begin
    if (hpos == (hva + hfp + hsp + hbp - 1)) begin
        hpos <= 0;
        if (vpos == (vva + vfp + vsp + vbp - 1)) begin
            vpos <= 0;
        end else begin
            vpos <= vpos + 1;
        end
    end else begin
        hpos <= hpos + 1;
    end
end

always @(*) begin
    if (hpos < hva && vpos < vva) begin
        r = 0;
        g = 0;
        b = 15;
    end else if (475 <= hpos && hpos < 485 && 280 <= vpos && vpos < 290) begin
        r = 15;
        g = 15;
        b = 15;
    end else begin
        r = 0;
        g = 0;
        b = 0;
    end

    hsync = (hpos >= hva + hfp) && (hpos < hva + hfp + hsp);
    vsync = (vpos >= vva + vfp) && (vpos < vva + vfp + vsp);
end

endmodule
