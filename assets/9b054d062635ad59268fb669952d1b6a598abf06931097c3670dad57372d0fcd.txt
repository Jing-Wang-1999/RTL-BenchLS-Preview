module origen_034978(
    input wire clock,
    output reg aux
);

    // Parameters for note frequencies
    parameter DIV_FA = 95602;
    parameter DIV_LA = 85131;
    parameter DIV_DO = 71633;
    parameter DIV_RE = 63775;
    parameter DIV_MI = 56818;
    parameter DIV_SOL = 47801;
    parameter DIV_SI = 42565;
    parameter DIV_CHG = 25000000;

    // Internal registers
    reg [27:0] C_CHG = 0;
    reg [18:0] C_NOTA = 0;
    reg [8:0] PASA_NOTA = 0;
    wire clock_CHG;

    // Generate clock_CHG signal
    assign clock_CHG = (C_CHG == DIV_CHG - 1);

    // Counter for changing notes
    always @(posedge clock) begin
        if (C_CHG == DIV_CHG - 1)
            C_CHG <= 0;
        else
            C_CHG <= C_CHG + 1;
    end

    // Counter for note progression
    always @(posedge clock) begin
        if (clock_CHG)
            PASA_NOTA <= (PASA_NOTA == 3'd7) ? 0 : PASA_NOTA + 1;
    end

    // Note generation
    always @(posedge clock) begin
        case (PASA_NOTA)
            3'd0: begin
                if (C_NOTA == DIV_FA - 1) begin
                    C_NOTA <= 0;
                    aux <= ~aux;
                end else
                    C_NOTA <= C_NOTA + 1;
            end
            3'd1: begin
                if (C_NOTA == DIV_LA - 1) begin
                    C_NOTA <= 0;
                    aux <= ~aux;
                end else
                    C_NOTA <= C_NOTA + 1;
            end
            3'd2: begin
                if (C_NOTA == DIV_DO - 1) begin
                    C_NOTA <= 0;
                    aux <= ~aux;
                end else
                    C_NOTA <= C_NOTA + 1;
            end
            3'd3: begin
                if (C_NOTA == DIV_RE - 1) begin
                    C_NOTA <= 0;
                    aux <= ~aux;
                end else
                    C_NOTA <= C_NOTA + 1;
            end
            3'd4: begin
                if (C_NOTA == DIV_MI - 1) begin
                    C_NOTA <= 0;
                    aux <= ~aux;
                end else
                    C_NOTA <= C_NOTA + 1;
            end
            3'd5: begin
                if (C_NOTA == DIV_SOL - 1) begin
                    C_NOTA <= 0;
                    aux <= ~aux;
                end else
                    C_NOTA <= C_NOTA + 1;
            end
            3'd6: begin
                if (C_NOTA == DIV_SI - 1) begin
                    C_NOTA <= 0;
                    aux <= ~aux;
                end else
                    C_NOTA <= C_NOTA + 1;
            end
            3'd7: begin
                if (C_NOTA == DIV_DO - 1) begin
                    C_NOTA <= 0;
                    aux <= ~aux;
                end else
                    C_NOTA <= C_NOTA + 1;
            end
        endcase
    end

endmodule
