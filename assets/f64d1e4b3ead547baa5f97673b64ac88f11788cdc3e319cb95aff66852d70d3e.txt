module rtlpp_00081
  (
   input           clk,
   output reg      DALtx, 
   inout [21:0]    DAL,
   input           RBS7,
   input           RSYNC,
   input           RDIN,
   input           RDOUT,
   output reg      TRPLY,

   output reg [12:0] iADDR,   
   output reg      iBS7,   
   input           iREAD_MATCH, 
   input           iWRITE_MATCH, 

   output reg [15:0] iWDATA,
   output reg      iWRITE,    
   input [15:0]    iRDATA   
   );

   reg [15:0]      read_reg;    
   reg [21:0]      TDAL;   
   assign DAL = TDAL;

   always @(posedge RSYNC) begin
      iADDR <= DAL[12:0];
      iBS7 <= RBS7;
   end

   reg [1:0] SYNCra;
   always @(posedge clk) SYNCra <= { SYNCra[0], RSYNC };
   wire sSYNC = SYNCra[1]; 

   reg start_state = 0;
   always @(posedge clk) begin
      if (sSYNC) begin
         if ((start_state == 0) && iREAD_MATCH) begin
            read_reg <= iRDATA;   
         end
         start_state <= 1;
      end else
         start_state <= 0;
   end 
      
   always @(*) begin
      DALtx <= 0;
      TDAL <= 22'bZ;
      TRPLY <= 0;

      if (RDIN && iREAD_MATCH) begin
         DALtx <= 1;
         TDAL <= { 6'b0, read_reg };
         TRPLY <= 1;
      end else if (RDOUT && iWRITE_MATCH) begin
         TRPLY <= 1;
      end 
   end

   always @(posedge RDOUT) begin
      iWDATA <= DAL[15:0];
   end

   reg [1:0] DOUTra;
   always @(posedge clk) DOUTra <= { DOUTra[0], RDOUT };
   wire sDOUT = DOUTra[1];   

   reg write_state = 0;
   always @(posedge clk) begin
      iWRITE <= 0;
      if (sDOUT) begin
         if (write_state == 0) begin
            iWRITE <= 1;
         end
         write_state <= 1;
      end else
         write_state <= 0;
   end

endmodule
