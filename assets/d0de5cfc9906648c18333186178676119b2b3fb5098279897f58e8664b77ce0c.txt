    A,
    B,
    Cin,
    C,
    Cout

      .Sum(C[15:4]),

      .A(B[3]),
      .B(A[3]),
      .Y(rca_cin)
  );
  OR2X1 U7 (
      .A(B[3]),
      .B(A[3]),
      .Y(C[3])
  );
  OR2X1 U8 (
      .A(A[2]),
      .B(B[2]),
      .Y(C[2])
  );
  OR2X1 U9 (
      .A(A[1]),
      .B(B[1]),
      .Y(C[1])
  );
  OR2X1 U10 (
      .A(A[0]),
      .B(B[0]),
      .Y(C[0])
  );
endmodule
