module rtlpp_11479 (
    input wire [1:0] address,
    input wire chipselect,
    input wire clk,
    input wire reset_n,
    input wire write_n,
    input wire [31:0] writedata,
    output reg [31:0] out_port,
    output reg [31:0] readdata
);

    reg [31:0] data_out;

    always @(posedge clk or negedge reset_n) begin
        if (!reset_n) begin
            data_out <= 32'b0;
        end else if (chipselect && !write_n && (address == 2'b00)) begin
            data_out <= writedata;
        end
    end

    always @(*) begin
        if (address == 2'b00) begin
            readdata = data_out;
        end else begin
            readdata = 32'b0;
        end
    end

    always @(*) begin
        out_port = data_out;
    end

endmodule
