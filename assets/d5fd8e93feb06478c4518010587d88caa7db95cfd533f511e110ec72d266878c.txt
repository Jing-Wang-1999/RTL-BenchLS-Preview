`timescale 1ns/1ps
module tb;
reg clk=0,mc_clk=0,rst=0,cyc=0,stb=0,we=0;reg[31:0]addr=0,di=0;reg[3:0]sel=15;
wire[31:0]dout,mdout,poc;wire ack,err,doe,oe_n,we_n;wire[23:0]ma;wire[3:0]dqm;wire[7:0]cs;
wire[31:0]mdin=(!cs[0] && !oe_n)?mem0[ma[11:0]]:(!cs[1] && !oe_n)?mem1[ma[11:0]]:32'h0;
reg[31:0]mem0[0:4095],mem1[0:4095];integer i,j,reads=0,writes=0;reg[31:0]last;
always #5 clk=~clk;always #10 mc_clk=~mc_clk;
mc_top dut(.clk_i(clk),.rst_i(rst),.wb_data_i(di),.wb_data_o(dout),.wb_addr_i(addr),.wb_sel_i(sel),.wb_we_i(we),.wb_cyc_i(cyc),.wb_stb_i(stb),.wb_ack_o(ack),.wb_err_o(err),
.susp_req_i(1'b0),.resume_req_i(1'b0),.poc_o(poc),.mc_clk_i(mc_clk),.mc_br_pad_i(1'b0),.mc_ack_pad_i(1'b1),.mc_addr_pad_o(ma),.mc_data_pad_i(mdin),.mc_data_pad_o(mdout),.mc_dp_pad_i(4'b0),.mc_doe_pad_doe_o(doe),.mc_dqm_pad_o(dqm),.mc_oe_pad_o_(oe_n),.mc_we_pad_o_(we_n),.mc_cs_pad_o_(cs),.mc_sts_pad_i(1'b0));
always @(negedge mc_clk)if(!rst)begin
if(!cs[0] && !cs[1])$fatal(1,"overlapping chip selects");
if((!cs[0] || !cs[1]) && !we_n && doe)begin
writes=writes+1;
for(j=0;j<4;j=j+1)if(!dqm[j])begin
if(!cs[0])mem0[ma[11:0]][8*j+:8]=mdout[8*j+:8];
if(!cs[1])mem1[ma[11:0]][8*j+:8]=mdout[8*j+:8];end end
if((!cs[0] || !cs[1]) && !oe_n && we_n)reads=reads+1;
end
task access;input wr;input[31:0]a,d;input[3:0]mask;input expected_error;integer count;begin
@(negedge clk);addr=a;di=d;sel=mask;we=wr;cyc=1;stb=1;
count=0;@(posedge clk);while(ack!==1 && err!==1 && count<2000)begin @(posedge clk);count=count+1;end
if(count>=2000)$fatal(1,"bus timeout %h",a);
if(err!==expected_error)$fatal(1,"bus error %h %b expected%0d",a,err,expected_error);
last=dout;
@(negedge clk);cyc=0;stb=0;we=0;repeat(8)@(posedge clk);end endtask
task readcheck;input[31:0]a,expected;begin access(0,a,0,15,0);if(last!==expected)$fatal(1,"read %h %h != %h",a,last,expected);end endtask
initial begin #10000000;$fatal(1,"timeout");end
initial begin
for(i=0;i<4096;i=i+1)begin mem0[i]=32'h5a000000^i;mem1[i]=32'ha5000000^i;end
#1;rst=1;repeat(12)@(posedge clk);@(negedge clk);rst=0;repeat(320)@(posedge mc_clk);
access(1,32'h60000008,32'hff,15,0);
access(1,32'h60000014,32'h222204,15,0);
access(1,32'h6000001c,32'h222204,15,0);
access(1,32'h60000010,32'h25,15,0);
access(1,32'h60000018,32'h800025,15,0);
readcheck(32'h60000008,255);readcheck(32'h60000010,37);readcheck(32'h60000018,32'h800025);
access(1,32'h60000014,32'h0,15,0);readcheck(32'h60000014,32'h0);
access(1,32'h60000014,32'h12345678,15,0);readcheck(32'h60000014,32'h12345678);
access(1,32'h60000014,32'h3fffffff,15,0);readcheck(32'h60000014,32'h3fffffff);
access(1,32'h60000014,32'h222204,15,0);readcheck(32'h60000014,32'h222204);
$display("RTL_BENCHLS_PASS");$finish(0);end endmodule
