module mg_07574 (
    input [3:0] a,
    input [3:0] b,
    input cin,
    output [3:0] s,
    output cout
);
    wire [3:0] G; // Generate
    wire [3:0] P; // Propagate
    wire [3:0] c; // Intermediate carry

    // Generate and Propagate calculation
    assign G = a & b;       // Generate: G[i] = a[i] & b[i]
    assign P = a ^ b;       // Propagate: P[i] = a[i] ^ b[i]

    // Carry calculation
    assign c[0] = cin;
    assign c[1] = G[0] | (P[0] & c[0]);
    assign c[2] = G[1] | (P[1] & c[1]);
    assign c[3] = G[2] | (P[2] & c[2]);
    assign cout = G[3] | (P[3] & c[3]);

    // Sum calculation
    assign s = P ^ c;

endmodule
