module origen_217971(
  input clk,
  output reg hsync,
  output reg vsync,
  output reg [9:0] x,
  output reg [9:0] y
);

  parameter H_PULSE = 96;
  parameter H_BACK  = 48;
  parameter H_ACTIVE = 640;
  parameter H_FRONT = 16;
  parameter H_TOTAL = H_PULSE + H_BACK + H_ACTIVE + H_FRONT;

  parameter V_PULSE = 2;
  parameter V_BACK  = 33;
  parameter V_ACTIVE = 480;
  parameter V_FRONT = 10;
  parameter V_TOTAL = V_PULSE + V_BACK + V_ACTIVE + V_FRONT;

  reg [9:0] cnt_x;
  reg [9:0] cnt_y;

  always @(posedge clk) begin
    if (cnt_x == H_TOTAL - 1) begin
      cnt_x <= 0;
      if (cnt_y == V_TOTAL - 1)
        cnt_y <= 0;
      else
        cnt_y <= cnt_y + 1;
    end
    else
      cnt_x <= cnt_x + 1;
  end

  reg in_hs;
  reg in_vs;

  always @(*) begin
    if (cnt_x < H_PULSE)
      in_hs = 1'b0;
    else
      in_hs = 1'b1;

    if (cnt_y < V_PULSE)
      in_vs = 1'b0;
    else
      in_vs = 1'b1;
  end

  always @(*) begin
    hsync = ~in_hs;
    vsync = ~in_vs;
    x = cnt_x - H_PULSE - H_BACK;
    y = cnt_y - V_PULSE - V_BACK;
  end

endmodule
