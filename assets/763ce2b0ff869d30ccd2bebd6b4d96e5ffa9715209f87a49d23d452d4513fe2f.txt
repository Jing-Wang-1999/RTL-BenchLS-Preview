    if (req_ready) begin
      last_eot <= req_last_burst_length == 'h0;
      last_non_eot <= 1'b0;
      beat_counter <= 'h1;
    end else if (m_axi_valid == 1'b1) begin
      last_eot <= beat_counter == last_burst_length;
      last_non_eot <= beat_counter == BEAT_COUNTER_MAX;
      beat_counter <= beat_counter + 1'b1;
    end
  end
