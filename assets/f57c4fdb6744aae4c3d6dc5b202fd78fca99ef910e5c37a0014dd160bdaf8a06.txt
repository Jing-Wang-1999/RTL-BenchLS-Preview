#!/usr/bin/env python3
"""One upstream testbench = one score point. Candidate RTL keeps source-relative paths."""
import argparse,json,re,subprocess,shutil
from pathlib import Path
p=argparse.ArgumentParser();p.add_argument('--candidate-dir',type=Path);p.add_argument('--output',type=Path,required=True);a=p.parse_args();base=Path(__file__).resolve().parent;cfg=json.loads((base/'case.json').read_text());a.output.mkdir(parents=True,exist_ok=True);source=a.output/'src';shutil.copytree(base/'source',source,dirs_exist_ok=True)
if a.candidate_dir:
 for f in cfg['rtl_files']:
  candidate=a.candidate_dir/f
  if not candidate.is_file():raise SystemExit('Missing candidate RTL: '+str(candidate))
  shutil.copy2(candidate,source/f)
results=[]
for tp in cfg['testpoints']:
 out=a.output/tp['id'];out.mkdir(exist_ok=True);incs=sorted({str((source/f).parent) for f in tp['source_files']});cmd=[cfg['simulator'],'--binary','--timing','--assert','-Wno-fatal','--top-module',tp['testbench_top'],'--Mdir',str((out/'obj').resolve())]+['-I'+x for x in incs]+[str((source/f).resolve()) for f in tp['compiled_files']]
 def run(command,log,timeout):
  try:
   with log.open('w') as fh:q=subprocess.run(command,cwd=source,stdout=fh,stderr=subprocess.STDOUT,timeout=timeout)
   return q.returncode
  except subprocess.TimeoutExpired:return None
 compile_rc=run(cmd,out/'compile.log',180);rc=run([str((out/'obj'/('V'+tp['testbench_top'])).resolve())],out/'simulation.log',60) if compile_rc==0 else None;log=(out/'simulation.log').read_text() if (out/'simulation.log').exists() else '';passed=compile_rc==0 and rc==0 and bool(re.search(r'^SUCCESS$',log,re.M)) and not re.search(r'^FAILED$',log,re.M);results.append({'id':tp['id'],'passed':bool(passed),'compile_exit':compile_rc,'simulation_exit':rc,'command':cmd})
summary={'parent_case_id':cfg['case_id'],'passed_testpoints':sum(x['passed'] for x in results),'total_testpoints':len(results),'testpoints':results};summary['score']=summary['passed_testpoints']/summary['total_testpoints'];(a.output/'result.json').write_text(json.dumps(summary,indent=2)+'\n');print(json.dumps(summary,indent=2))
