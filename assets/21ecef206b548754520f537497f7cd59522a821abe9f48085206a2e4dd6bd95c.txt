module uart_txfsm(
             reset_n        ,
             baud_clk_16x   ,

             cfg_tx_enable  ,
             cfg_stop_bit   ,
             cfg_pri_mod    ,

       
             fifo_empty     ,
             fifo_rd        ,
             fifo_data      ,

          
             so  
          );
input             reset_n        ;
input             baud_clk_16x   ;
input             cfg_tx_enable  ;
input             cfg_stop_bit   ;
input   [1:0]     cfg_pri_mod    ;
input             fifo_empty     ;
output            fifo_rd        ;
input  [7:0]      fifo_data      ;
output            so             ;
endmodule
