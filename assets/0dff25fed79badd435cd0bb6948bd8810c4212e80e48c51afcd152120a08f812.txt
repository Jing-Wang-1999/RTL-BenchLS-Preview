module origen_101100 (
    input wire clk,
    input wire rst_n,
    input wire clr,
    input wire set,
    input wire en,
    input wire d,
    output reg q
);
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            q <= 1'b1;
        end else if (clr) begin
            q <= 1'b0;
        end else if (set) begin
            q <= 1'b1;
        end else if (en) begin
            q <= d;
        end
    end
endmodule
