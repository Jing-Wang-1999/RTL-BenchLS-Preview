// Required path: rtl/verilog/fifo4.v
module fifo4(clk, rst, clr,  din, we, dout, re, full, empty);
parameter dw = 8;
input		clk, rst;
input		clr;
input   [dw:1]	din;
input		we;
output  [dw:1]	dout;
input		re;
output		full, empty;
endmodule
