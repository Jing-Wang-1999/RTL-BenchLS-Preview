module origen_221118(
    input clk,
    input rst_n,
    input [1:0] br_cfg,
    input rda,
    input tbr,
    inout [7:0] databus,
    output reg iocs,
    output reg iorw,
    output reg [1:0] ioaddr,
    output reg [7:0] databus_out
);

    reg [7:0] data_buf;
    reg capture_data, send_data;
    reg load_db_high, load_db_low;
    reg drive_tx_rx;
    reg [2:0] st, nxt_st;

    localparam IDLE = 3'b000, INIT_HIGH = 3'b001, INIT_LOW = 3'b010, RECEIVE = 3'b011, TRANSMIT = 3'b100;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            st <= IDLE;
        end else begin
            st <= nxt_st;
        end
    end

    always @(*) begin
        nxt_st = st;
        iocs = 1'b0;
        iorw = 1'b0;
        ioaddr = 2'b00;
        capture_data = 1'b0;
        send_data = 1'b0;
        load_db_high = 1'b0;
        load_db_low = 1'b0;
        drive_tx_rx = 1'b0;

        case (st)
            IDLE: begin
                nxt_st = INIT_HIGH;
            end
            INIT_HIGH: begin
                iocs = 1'b1;
                iorw = 1'b1;
                ioaddr = 2'b00;
                load_db_high = 1'b1;
                drive_tx_rx = 1'b1;
                nxt_st = INIT_LOW;
            end
            INIT_LOW: begin
                iocs = 1'b1;
                iorw = 1'b1;
                ioaddr = 2'b00;
                load_db_low = 1'b1;
                drive_tx_rx = 1'b1;
                nxt_st = RECEIVE;
            end
            RECEIVE: begin
                iocs = 1'b1;
                iorw = 1'b0;
                ioaddr = 2'b00;
                capture_data = rda;
                drive_tx_rx = 1'b1;
                if (rda) begin
                    nxt_st = TRANSMIT;
                end else begin
                    nxt_st = RECEIVE;
                end
            end
            TRANSMIT: begin
                iocs = 1'b1;
                iorw = 1'b1;
                ioaddr = 2'b00;
                send_data = tbr;
                drive_tx_rx = 1'b1;
                if (tbr) begin
                    nxt_st = RECEIVE;
                end else begin
                    nxt_st = TRANSMIT;
                end
            end
            default: begin
                nxt_st = IDLE;
            end
        endcase
    end

    always @(posedge clk) begin
        if (load_db_high) begin
            databus_out[7:0] <= {br_cfg, 6'b000000};
        end else if (load_db_low) begin
            databus_out[7:0] <= {2'b00, br_cfg};
        end else if (capture_data) begin
            data_buf <= databus;
        end
    end

    assign databus = (drive_tx_rx && send_data) ? data_buf : 8'bz;

endmodule
