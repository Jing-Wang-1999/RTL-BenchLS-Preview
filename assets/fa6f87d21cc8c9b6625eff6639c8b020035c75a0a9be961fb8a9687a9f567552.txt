    reg [31:0] a, b, c, d, e, f, g, h;

    // Helper functions
    function [31:0] ch;
        input [31:0] x, y, z;
        begin
            ch = (x & y) ^ (~x & z);
        end
    endfunction

    function [31:0] maj;
        input [31:0] x, y, z;
        begin
            maj = (x & y) ^ (x & z) ^ (y & z);
        end
    endfunction

    function [31:0] ep0;
        input [31:0] x;
        begin
            ep0 = {x[1:0], x[31:2]} ^ {x[12:0], x[31:13]} ^ {x[21:0], x[31:22]};
        end
    endfunction

    function [31:0] ep1;
        input [31:0] x;
        begin
            ep1 = {x[5:0], x[31:6]} ^ {x[10:0], x[31:11]} ^ {x[24:0], x[31:25]};
        end
    endfunction

    function [31:0] sig0;
        input [31:0] x;
        begin
            sig0 = {x[6:0], x[31:7]} ^ {x[17:0], x[31:18]} ^ (x >> 3);
        end
    endfunction

    function [31:0] sig1;
        input [31:0] x;
        begin
            sig1 = {x[16:0], x[31:17]} ^ {x[18:0], x[31:19]} ^ (x >> 10);
        end
    endfunction

            {a, b, c, d, e, f, g, h} <= rx_state;

        end else begin
            // Perform hash computation
            if (cnt < 64) begin

                    h + ep1(e) + ch(e,f,g) + K[cnt] + w[cnt] + ep0(a) + maj(a,b,c),
                    a,
                    b,
                    c,
                    d + h + ep1(e) + ch(e,f,g) + K[cnt] + w[cnt],
                    e,
                    f,
                    g
                };
            end else begin
                tx_hash <= {a, b, c, d, e, f, g, h} + rx_state;
            end
        end
    end
