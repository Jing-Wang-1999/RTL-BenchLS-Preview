    input wire [3:0] HEXnumber,

    always @* begin
        case (HEXnumber)

            4'h7:    Segments = 8'b11111000;  // 7
            4'h8:    Segments = 8'b10000000;  // 8
            4'h9:    Segments = 8'b10010000;  // 9
            4'hA:    Segments = 8'b10001000;  // A
            4'hB:    Segments = 8'b10000011;  // b
            4'hC:    Segments = 8'b11000110;  // C

        endcase
    end

endmodule
