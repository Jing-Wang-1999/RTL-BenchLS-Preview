module origen_216083 (
    input [5:0] op,
    input [5:0] func,
    output addu,
    output subu,
    output ori,
    output lui,
    output lw,
    output sw,
    output beq,
    output j,
    output jal,
    output jr,
    output blezals,
    output rev
);

    assign addu = (op == 6'b000000) && (func == 6'b100001);
    assign subu = (op == 6'b000000) && (func == 6'b100011);
    assign ori  = (op == 6'b001101);
    assign lui  = (op == 6'b001111);
    assign lw   = (op == 6'b100011);
    assign sw   = (op == 6'b101011);
    assign beq  = (op == 6'b000100);
    assign j    = (op == 6'b000010);
    assign jal  = (op == 6'b000011);
    assign jr   = (op == 6'b000000) && (func == 6'b001000);
    assign blezals = (op == 6'b011000);
    assign rev  = (op == 6'b000000) && (func == 6'b010100);

endmodule
