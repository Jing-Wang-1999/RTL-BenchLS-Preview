module origen_179422 (
    input [31:0] opA,
    input [31:0] opB,
    input [2:0] sel,
    output reg [31:0] res,
    output reg z,
    output reg c,
    output reg v
);

always @(*) begin
    case (sel)
        3'b000: begin // Addition
            {c, res} = opA + opB;
            v = (~c & (opA[31] ^ opB[31]) & (res[31] ^ opA[31])) | (c & (opA[31] ^ opB[31]) & (res[31] ^ opB[31]));
            z = (res == 32'b0);
        end
        3'b001: begin // Subtraction
            {c, res} = opA - opB;
            v = (c & (opA[31] ^ opB[31]) & (res[31] ^ opA[31])) | (~c & (opA[31] ^ opB[31]) & (res[31] ^ opB[31]));
            z = (res == 32'b0);
        end
        3'b010: begin // Bitwise AND
            res = opA & opB;
            z = (res == 32'b0);
            c = 1'b0;
            v = 1'b0;
        end
        3'b011: begin // Bitwise OR
            res = opA | opB;
            z = (res == 32'b0);
            c = 1'b0;
            v = 1'b0;
        end
        3'b100: begin // Bitwise NOT
            res = ~opA;
            z = (res == 32'b0);
            c = 1'b0;
            v = 1'b0;
        end
        default: begin
            $display("Error: Invalid ALU operation selected.");
            res = 32'b0;
            z = 1'b0;
            c = 1'b0;
            v = 1'b0;
        end
    endcase

    $display("z = %b, c = %b, v = %b", z, c, v);
end

endmodule
