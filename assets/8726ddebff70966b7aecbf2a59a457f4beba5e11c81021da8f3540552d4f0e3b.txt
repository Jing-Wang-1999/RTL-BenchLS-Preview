module autovcoder_063619
#(
    parameter DW = 32,
    parameter DD = 1024,
    parameter AW = 10
)
(
    // Clock Signals
    input  wire              clk0,
    input  wire              clk1,
    
    // Control Signals (Active-Low)
    input  wire              n_cs0,
    input  wire              n_cs1,
    input  wire              n_we0,
    input  wire              n_we1,
    input  wire              n_oe0,
    input  wire              n_oe1,
    
    // Address Inputs
    input  wire [AW-1:0]     ad0,
    input  wire [AW-1:0]     ad1,
    
    // Data Inputs
    input  wire [DW-1:0]     di0,
    input  wire [DW-1:0]     di1,
    
    // Data Outputs
    output reg  [DW-1:0]     do0,
    output reg  [DW-1:0]     do1
);

    // Memory Array
    reg [DW-1:0] mem [0:DD-1];

    // Internal Data Registers
    reg [DW-1:0] data_reg0;
    reg [DW-1:0] data_reg1;

    // Port 0 Logic
    always @(posedge clk0) begin
        if (!n_cs0) begin
            if (!n_we0) begin
                // Write Operation
                mem[ad0] <= di0;
                data_reg0 <= di0;
            end else begin
                // Read Operation
                data_reg0 <= mem[ad0];
            end
        end
        // If n_cs0 is high, data_reg0 retains its value (implicitly handled)
    end

    // Port 1 Logic
    always @(posedge clk1) begin
        if (!n_cs1) begin
            if (!n_we1) begin
                // Write Operation
                mem[ad1] <= di1;
                data_reg1 <= di1;
            end else begin
                // Read Operation
                data_reg1 <= mem[ad1];
            end
        end
        // If n_cs1 is high, data_reg1 retains its value (implicitly handled)
    end

    // Continuous Output Assignment (n_oe ignored)
    always @* begin
        do0 = data_reg0;
        do1 = data_reg1;
    end

endmodule
