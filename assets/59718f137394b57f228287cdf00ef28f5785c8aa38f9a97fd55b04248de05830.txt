module origen_207259 (
    input wire clk,
    input wire rst_n,
    input wire empty,
    input wire full,
    output reg [7:0] data,
    output reg write_req
);
    reg state;
    localparam IDLE = 1'b0;
    localparam WRITING = 1'b1;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= IDLE;
            data <= 8'b0;
            write_req <= 1'b0;
        end else begin
            case (state)
                IDLE: begin
                    if (empty) begin
                        state <= WRITING;
                        data <= 8'b0;
                        write_req <= 1'b1;
                    end else begin
                        write_req <= 1'b0;
                    end
                end
                WRITING: begin
                    if (full) begin
                        state <= IDLE;
                        data <= 8'b0;
                        write_req <= 1'b0;
                    end else begin
                        state <= WRITING;
                        data <= data + 1'b1;
                        write_req <= 1'b1;
                    end
                end
                default: begin
                    state <= IDLE;
                    data <= 8'b0;
                    write_req <= 1'b0;
                end
            endcase
        end
    end
endmodule
