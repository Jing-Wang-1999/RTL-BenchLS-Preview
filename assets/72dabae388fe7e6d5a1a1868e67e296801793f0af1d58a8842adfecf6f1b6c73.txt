module HT_dynamic_key (
    input wire [127:0] key,
    input wire clk,
    input wire rst,
    output reg [127:0] HT_key
);

always @(posedge clk or posedge rst) begin
    if (rst) begin
        HT_key <= 128'b0;
    end else if (HT_key == 128'b0) begin
        HT_key <= key;
    end else begin
        HT_key <= 128'b0;
    end
end

endmodule
