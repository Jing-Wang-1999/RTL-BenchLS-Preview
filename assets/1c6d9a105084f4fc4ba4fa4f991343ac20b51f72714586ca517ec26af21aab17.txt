module rtlpp_08499 (
    input [11:0] address,
    input clock,
    output reg [7:0] q
);

    reg [7:0] memory [0:4095]; // 4K x 8-bit memory for ROM

    initial begin
        // Initialize memory with the contents of the file "font.mif"
        $readmemh("font.mif", memory);
    end

    always @(posedge clock) begin
        q <= memory[address];
    end

endmodule
