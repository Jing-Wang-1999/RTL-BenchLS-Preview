module mg_00830 (ref_clk_sel_code,ref_clk_out,pcie_rstn,ep_core_clk_out,rp_rstn);

 output ref_clk_out; 
 output pcie_rstn;

 integer half_period; 
 reg ref_clk_out;

  core_clk_cnt <= 0; 
  else 
  core_clk_cnt <= core_clk_cnt + 1;

 endmodule
