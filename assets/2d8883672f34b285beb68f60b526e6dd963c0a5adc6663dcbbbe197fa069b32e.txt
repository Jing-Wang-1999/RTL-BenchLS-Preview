	   req_ack  = 1'b0;
	   req_ld = b2r_ack;
	   r2b_req = 1'b1;                       // req_gen to bank_req
	   next_req_st = (b2r_ack) ? 2'b00 : 2'b10;
	end // case: `REQ_ACTIVE

      endcase // case(req_st)
