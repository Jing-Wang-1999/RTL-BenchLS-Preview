module origen_179000(
    input  wire [7:0] din,
    input  wire       wr_en,
    input  wire       clk_50m,
    input  wire       clken,
    output reg        tx,
    output reg        tx_busy
);

localparam STATE_IDLE = 2'd0,
           STATE_START = 2'd1,
           STATE_DATA = 2'd2,
           STATE_STOP = 2'd3;

reg [1:0] state;
reg [2:0] bit_count;
reg [7:0] shift_reg;

always @(posedge clk_50m) begin
    if (clken) begin
        case (state)
            STATE_IDLE: begin
                tx <= 1'b1;
                tx_busy <= 1'b0;
                if (wr_en) begin
                    state <= STATE_START;
                    shift_reg <= din;
                end
            end
            STATE_START: begin
                tx <= 1'b0;
                tx_busy <= 1'b1;
                state <= STATE_DATA;
                bit_count <= 3'd0;
            end
            STATE_DATA: begin
                tx <= shift_reg[0];
                shift_reg <= {1'b0, shift_reg[7:1]};
                bit_count <= bit_count + 3'd1;
                if (bit_count == 3'd7)
                    state <= STATE_STOP;
            end
            STATE_STOP: begin
                tx <= 1'b1;
                bit_count <= 3'd0;
                state <= STATE_IDLE;
            end
        endcase
    end
end

endmodule
