module origen_131128 (
    input wire clk,
    input wire rst,
    input wire [7:0] shift_in,
    output wire [6:0] seg_out
);

    wire [3:0] counter_out;
    wire [7:0] shift_out;
    wire [7:0] sum;

    // 4-bit binary counter
    reg [3:0] counter;
    always @(posedge clk or posedge rst) begin
        if (rst)
            counter <= 4'b0000;
        else
            counter <= (counter == 4'b1111) ? 4'b0000 : counter + 1'b1;
    end
    assign counter_out = counter;

    // 8-bit shift register
    reg [7:0] shift_reg;
    always @(negedge clk or posedge rst) begin
        if (rst)
            shift_reg <= 8'b0;
        else
            shift_reg <= {shift_reg[6:0], shift_in[7]};
    end
    assign shift_out = shift_reg;

    // Functional module
    assign sum = {4'b0000, counter_out} + shift_out;

    // 7-segment display decoder
    reg [6:0] seg;
    always @(*) begin
        case (sum[3:0])
            4'h0: seg = 7'b1000000;
            4'h1: seg = 7'b1111001;
            4'h2: seg = 7'b0100100;
            4'h3: seg = 7'b0110000;
            4'h4: seg = 7'b0011001;
            4'h5: seg = 7'b0010010;
            4'h6: seg = 7'b0000010;
            4'h7: seg = 7'b1111000;
            4'h8: seg = 7'b0000000;
            4'h9: seg = 7'b0010000;
            default: seg = 7'b1111111;
        endcase
    end
    assign seg_out = seg;

endmodule
