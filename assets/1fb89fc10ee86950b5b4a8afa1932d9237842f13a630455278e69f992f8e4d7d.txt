module addsub #(parameter WIDTH = 16, parameter METHOD = "GENERIC") (
    input wire sub,
    input wire [WIDTH-1:0] a,
    input wire [WIDTH-1:0] b,
    output wire [WIDTH:0] o
);
    generate
        if (METHOD == "GENERIC") begin
            assign o = sub ? (a - b) : (a + b);
        end else if (METHOD == "OPTIMIZED") begin
            wire [WIDTH-1:0] b_xor;
            wire [WIDTH:0] carry;
            assign b_xor = b ^ {WIDTH{sub}};
            assign carry[0] = sub;
            assign o[WIDTH:1] = a + b_xor + carry[0];
            assign o[0] = carry[WIDTH];
        end else begin
            initial begin
                $error("Unsupported METHOD parameter value: %s", METHOD);
            end
        end
    endgenerate
endmodule
