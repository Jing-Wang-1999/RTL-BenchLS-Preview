module rtlpp_02749 (
   output reg DALtx,

   inout [21:0] DAL, 
   input RDOUT,
   output reg TDOUT,
   input RRPLY,
   output reg TRPLY,
   input RDIN,
   output reg TDIN,
   input RSYNC,
   output reg TSYNC,
   input RIRQ4,
   output reg TIRQ4,
   input RIRQ5,
   output reg TIRQ5,
   input RIRQ6,
   output reg TIRQ6,
   input RIRQ7,
   output reg TIRQ7,
   input RWTBT,
   output reg TWTBT, 
   input RREF, 
   output reg TREF, 

   input RINIT,
   input RDCOK,
   input RPOK,
   input RBS7,
   input RIAKI,
   input RDMGI,

   output reg TDMR,
   output reg TSACK,
   output reg TIAKO,
   output reg TDMGO
   );

   parameter addr = 22'o777777;

   reg [15:0] reg_data = 16'o123456;	
   reg [12:0] io_addr = 0;
   reg io_page = 0;
   wire addr_match = RSYNC && (io_addr == addr[12:0]) && io_page;
   reg drive_DAL;		

   assign DAL = drive_DAL ? { 6'b0, reg_data } : 22'bz;

   always @(posedge RSYNC) begin
      io_page <= RBS7;
      io_addr <= DAL[12:0];
   end

   always @(posedge RDOUT) begin
     if (RSYNC && addr_match)
       reg_data <= DAL[15:0];
   end

   always @(*) begin
      drive_DAL <= 0;
      DALtx <= 0;
      TDOUT <= 0;
      TRPLY <= 0;
      TDIN <= 0;
      TSYNC <= 0;
      TIRQ4 <= 0;
      TIRQ5 <= 0;
      TIRQ6 <= 0;
      TIRQ7 <= 0;
      TWTBT <= 0;
      TREF <= 0;
      TDMR <= 0;
      TSACK <= 0;
      TIAKO <= 0;
      TDMGO <= 0;
      
      if (addr_match) begin
         if (RDIN) begin
            DALtx <= 1;
            drive_DAL <= 1;
            TRPLY <= 1;
         end else if (RDOUT) begin
            TRPLY <= 1;
         end
      end
   end

endmodule
