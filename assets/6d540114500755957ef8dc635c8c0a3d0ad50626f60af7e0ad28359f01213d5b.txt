module bit_register (
		 
		 we,		 
		 clk,
		 reset_n,
		 data_in,
		 
		 
		 data_out
		 );
parameter  RESET_DEFAULT = 1'h0;
input	 we;
input	 clk;
input	 reset_n;
input	 data_in;
output data_out;
endmodule

module  generic_register	(
	      
	      we,		 
	      data_in,
	      reset_n,
	      clk,
	      
	      
	      data_out
	      );
parameter   WD               = 1;
parameter   RESET_DEFAULT    = 0;
input [WD-1:0]     we;
input [WD-1:0]     data_in;

output [WD-1:0]    data_out;
endmodule
