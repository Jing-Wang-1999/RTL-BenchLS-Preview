module origen_070964(
    input [63:0] base, exponent, modulus,
    input clk, rst, done,
    output reg [63:0] C,
    output reg done_encrypt
);

    reg [63:0] baseint, expint, modint;
    reg [127:0] temp;
    reg [6:0] count;
    reg beginsignal, start;

    always @(posedge clk or negedge rst) begin
        if (!rst) begin
            C <= 64'd1;
            baseint <= 64'd0;
            expint <= 64'd0;
            modint <= 64'd0;
            temp <= 128'd0;
            count <= 7'd0;
            done_encrypt <= 1'b0;
            beginsignal <= 1'b0;
            start <= 1'b0;
        end else begin
            if (done || start) begin
                baseint <= base;
                expint <= exponent;
                modint <= modulus;
                C <= 64'd1;
                count <= 7'd63;
                done_encrypt <= 1'b0;
                beginsignal <= 1'b1;
                start <= 1'b0;
            end else if (beginsignal) begin
                if (expint != 64'd0) begin
                    if (expint[0] == 1'b1) begin
                        temp <= C * baseint;
                        C <= (temp % modint);
                    end
                    temp <= baseint * baseint;
                    baseint <= (temp % modint);
                    expint <= expint >> 1;
                    count <= count - 1;
                end else begin
                    done_encrypt <= 1'b1;
                    beginsignal <= 1'b0;
                end
            end
        end
    end

endmodule
