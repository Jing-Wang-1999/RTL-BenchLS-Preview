module origen_054985 (
    input wire clk,
    input wire rst,
    input wire stall,
    input wire flush,
    
    input wire MEM_inst_en,
    input wire [31:0] MEM_ALU_Result,
    input wire [31:0] MEM_MEM_Result,
    input wire [4:0] MEM_Rs,
    input wire [4:0] MEM_Rt,
    input wire [4:0] MEM_Rdst,
    input wire MEM_RegW,
    input wire MEM_MemR,
    input wire MEM_MemW,
    
    output reg WB_inst_en,
    output reg [31:0] WB_ALU_Result,
    output reg [31:0] WB_MEM_Result,
    output reg [4:0] WB_Rs,
    output reg [4:0] WB_Rt,
    output reg [4:0] WB_Rdst,
    output reg WB_RegW,
    output reg WB_MemR,
    output reg WB_MemW
);

    always @(posedge clk or posedge rst) begin
        if (rst || flush) begin
            WB_inst_en <= 1'b0;
            WB_ALU_Result <= 32'b0;
            WB_MEM_Result <= 32'b0;
            WB_Rs <= 5'b0;
            WB_Rt <= 5'b0;
            WB_Rdst <= 5'b0;
            WB_RegW <= 1'b0;
            WB_MemR <= 1'b0;
            WB_MemW <= 1'b0;
        end
        else if (!stall) begin
            WB_inst_en <= MEM_inst_en;
            WB_ALU_Result <= MEM_ALU_Result;
            WB_MEM_Result <= MEM_MEM_Result;
            WB_Rs <= MEM_Rs;
            WB_Rt <= MEM_Rt;
            WB_Rdst <= MEM_Rdst;
            WB_RegW <= MEM_RegW;
            WB_MemR <= MEM_MemR;
            WB_MemW <= MEM_MemW;
        end
    end

endmodule
