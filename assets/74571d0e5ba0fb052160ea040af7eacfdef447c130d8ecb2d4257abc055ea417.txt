module inputRom (
    input          clka,
    input  [4:0]   addra,
    output [15:0]  douta
);
    reg [15:0] rom [0:31];
    initial begin
        // Initialize ROM with some values, should be replaced by actual data.
        $readmemh("inputRom.hex", rom);
    end
    assign douta = rom[addra];
endmodule
