module comparator2bit(
    input [1:0] x,
    input [1:0] y,
    output xeqy,
    output xlty,
    output xgty
);

assign xeqy = (x[1] == y[1]) && (x[0] == y[0]);
assign xgty = (x[1] > y[1]) || ((x[1] == y[1]) && (x[0] > y[0]));
assign xlty = (x[1] < y[1]) || ((x[1] == y[1]) && (x[0] < y[0]));

endmodule
