module origen_042275(
    input clk,
    output reg lcd_e, lcd_rs, lcd_rw,
    output reg cs1, cs2, reset,
    output reg [7:0] lcd_data
);

    reg [21:0] divider;
    wire sclk;
    reg [7:0] count;
    reg [7:0] address;
    reg pattern;

    assign sclk = divider[15];

    always @(posedge clk) begin
        divider <= divider + 1;
    end

    always @(posedge sclk) begin
        if (count < 16) begin
            count <= count + 1;
            case (count)
                0: begin lcd_data <= 8'h38; lcd_rs <= 0; end
                1: begin lcd_data <= 8'h38; lcd_rs <= 0; end
                2: begin lcd_data <= 8'h38; lcd_rs <= 0; end
                3: begin lcd_data <= 8'h38; lcd_rs <= 0; end
                4: begin lcd_data <= 8'h06; lcd_rs <= 0; end
                5: begin lcd_data <= 8'h0C; lcd_rs <= 0; end
                6: begin lcd_data <= 8'h01; lcd_rs <= 0; end
                7: begin lcd_data <= 8'h80; lcd_rs <= 0; end
                default: begin lcd_data <= 8'h00; lcd_rs <= 1; end
            endcase
        end
        else begin
            count <= count + 1;
            if (address == 64) begin
                address <= 0;
                pattern <= ~pattern;
            end
            else begin
                address <= address + 1;
            end
            lcd_data <= pattern ? 8'hFF : 8'h00;
            lcd_rs <= 1;
        end

        lcd_e <= ~lcd_e;
        lcd_rw <= 0;
        cs1 <= 1;
        cs2 <= 0;
        reset <= 1;
    end

endmodule
