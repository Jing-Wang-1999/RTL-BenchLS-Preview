#!/usr/bin/env python3
from build_misc import *

SPEC='''# First-quadrant CORDIC project
Implement the complete three-module cordic project in cordic.v using the original manual.pdf in specification/. This is one integrated project even though the three modules share a file. The original manual supplies the micro-rotation algorithm, gain compensation, modes, numeric formats and implementation choices.

Curator configuration and interface clarification:
- Freeze the upstream default PIPELINE/ROTATE/RADIAN_16 configuration. Inputs/outputs x/y/theta are signed 17-bit values with 15 fractional bits; theta is in radians. The useful sine/cosine envelope is first-quadrant angles 0..pi/2 with x_i=19896 and y_i=0, compensating the CORDIC gain. Output x is cosine and y is sine, both scaled by 32768. No full-circle coarse rotation, VECTOR, iterative or combinational alternatives are required.
- The original ITERATIONS define is 16, but the pinned pipeline generates fifteen rotator stages with indices 0..14. Preserve that fifteeen-stage registered configuration, receiving a new input each rising clk edge. rst is synchronous active-high and clears each rotator state; no valid or enable signals exist. After reset, discard pipeline fill outputs. Angles are quantized before application; at the original 91 integer-degree samples, each output must be within 10 integer LSB of the original fixed-point reference table.
- Each rotator uses arithmetic right shifts by its iteration parameter, signed addition/subtraction and the corresponding quantized atan(2^-iteration), choosing direction from the residual angle sign. signed_shifter implements sign-extending right shift. The two body parameters of rotator (iteration and tangle) are included explicitly in interfaces/modules.v. Top exports the final residual theta without gain correction. Internal wires and state encodings are unrestricted.
- The declaration and golden implementation are preprocessed from pinned default macros, with originals retained. The manual's high-level stage count is clarified above; no executable algorithm has been changed. The private point retains all 91 upstream angle vectors, adding assertions for unknown results, exact vector count and the original 10-LSB tolerance. One entire testbench is one testpoint.
'''

def main():
 src=source('arithmetic_core_configurable_cordic_core_in_verilog')
 raw=(src/'cordic.v').read_text();s=strip_comments((src/'tb_cordic.v').read_text())
 s=s.replace('`include "cordic.v"',raw[:raw.index('module signed_shifter')])
 s=s.replace('module tb ();','module tb ();\ninteger checked=0;')
 old='''if (ex > 10 || ey > 10)
        $display("Angle: %d  sin = %f  cos = %f        errors %d %d",j,ry,rx,ey,ex);    
      else
        $display("Angle: %d  sin = %f  cos = %f",j,ry,rx);'''
 s,n=re.subn(r'if \(ex > 10 \|\| ey > 10\)\s*\$display\([^;]+;\s*else\s*\$display\([^;]+;', 'if ((^lx === 1\'bx) || (^ly === 1\'bx) || ex > 10 || ey > 10)$fatal(1,"CORDIC vector mismatch");\nchecked=checked+1;',s);assert n==1
 s=re.sub(r'\$display\([^;]*;',';',s)
 s=s.replace('endmodule','initial begin #1000;if(checked!=91)$fatal(1,"incomplete CORDIC vectors");$display("RTL_BENCHLS_PASS");$finish(0);end\nendmodule')
 spec=dict(case_id='opencores_configurable_cordic_direct_v1',project='arithmetic_core_configurable_cordic_core_in_verilog',top='cordic',instruction=SPEC,configuration_record=dict(configuration_id='pipeline_rotate_radian16',top_module='cordic',source_closure=['cordic.v'],specification_paths=['manual.pdf']),preprocess=True,tests=[('upstream91','cordic',s)],original_tests=[str(src/'tb_cordic.v')],builder_source=str(Path(__file__).resolve()))
 d=build(spec)
 # ANSI module headers may also declare parameters in the body.
 p=d/'public/interfaces/modules.v';s=p.read_text();s=s.replace('module rotator (','module rotator #(parameter integer iteration=0, parameter signed [16:0] tangle=0) (');p.write_text(s)
 review=dict(review_date='2026-09-11',decision='pass_with_explicit_curator_clarifications',original_documents=[dict(path='manual.pdf',sha256=digest(src/'manual.pdf'))],original_document_coverage='CORDIC mathematics, first-quadrant function, formats, pipeline and rotation modes',curator_clarifications=['Pinned 15 generated stages despite ITERATIONS=16','Exact fixed configuration and 10-LSB finite verification envelope','Retain body parameters in declaration-only interface'],unresolved_material_findings=[],instruction_sha256=digest(d/'public/instruction.md'))
 write_json(d/'private/specification_review.json',review)
 g=evaluate(d,d/'private/reference',d/'private/qualification/golden');print(g,flush=True);assert g['full_project_pass']
 mutant=d/'private/qualification/output_corrupt/candidate';shutil.copytree(d/'private/reference',mutant);p=mutant/'cordic.v';s=p.read_text();s,n=re.subn(r'assign x_o = x\[16-1\];','assign x_o = x[16-1] ^ 17\'h4000;',s);assert n==1;p.write_text(s)
 c=evaluate(d,mutant,d/'private/qualification/output_corrupt/evaluation');print(c,flush=True);assert c['compile_pass'] and not c['full_project_pass']
 write_json(d/'private/qualification/review.json',dict(golden_score=1,total_points=1,controls=[dict(name='output_corrupt',compile_pass=True,detected=True)],specification_review='private/specification_review.json'))
 shutil.copyfile(TASK/'pipeline/evaluate_case_v2.py',d/'private/verification/evaluate_case_v2.py')
 m=json.loads((d/'case.json').read_text());m.update(status='finalized',clarification_provenance='Unchanged original CORDIC manual plus pinned-configuration and port clarification.',readiness_review=dict(approved=True,reviewer='Codex direct_misc 2026-09-11',evidence_paths=['private/qualification/review.json']));m['artifacts']={str(p.relative_to(d)):digest(p) for p in sorted(d.rglob('*')) if p.is_file() and p.name!='case.json'};write_json(d/'case.json',m)
 dst=BASE/'sealed_cases'/m['case_id'];assert not dst.exists();d.rename(dst)
 g=evaluate(dst,dst/'private/reference',BASE/'sealed_replay'/m['case_id']);assert g['full_project_pass']
 row={k:m[k] for k in ['case_id','project_id','configuration_id']};row.update(path=m['case_id'],manifest_sha256=digest(dst/'case.json'))
 with (BASE/'sealed_cases/index.jsonl').open('a') as f:f.write(json.dumps(row,sort_keys=True)+'\n')
 print(validate(BASE/'sealed_cases'))

if __name__=='__main__':main()
