module origen_155519(
    input PinRX,
    input clk,
    input flagIN_DataRedy,
    input [7:0] dataOUT,
    output reg PinTX,
    output reg [7:0] dataIN,
    output reg flagOUT_DataResive
);

reg [10:0] clok2048MbodCounterRX;
reg [3:0] dataCounter;
reg [7:0] shiftRegRX;

reg [10:0] clok2048MbodCounterTX;
reg [3:0] dataCounterTX;
reg [7:0] shiftRegTX;

always @(posedge clk) begin
    // Reception
    if (PinRX == 1'b0) begin
        clok2048MbodCounterRX <= 11'd35;
        dataCounter <= 4'd12;
    end else if (clok2048MbodCounterRX != 0) begin
        clok2048MbodCounterRX <= clok2048MbodCounterRX - 1;
        if (clok2048MbodCounterRX == 1) begin
            shiftRegRX <= {shiftRegRX[6:0], PinRX};
            dataCounter <= dataCounter - 1;
            if (dataCounter == 3) begin
                dataIN <= shiftRegRX;
                flagOUT_DataResive <= 1'b1;
            end
        end
    end else begin
        flagOUT_DataResive <= 1'b0;
    end

    // Transmission
    if (flagIN_DataRedy == 1'b1) begin
        clok2048MbodCounterTX <= 11'd35;
        dataCounterTX <= 4'd10;
        shiftRegTX <= dataOUT;
    end else if (clok2048MbodCounterTX != 0) begin
        clok2048MbodCounterTX <= clok2048MbodCounterTX - 1;
        if (clok2048MbodCounterTX == 1) begin
            PinTX <= shiftRegTX[0];
            shiftRegTX <= {1'b1, shiftRegTX[7:1]};
            dataCounterTX <= dataCounterTX - 1;
        end
    end else begin
        PinTX <= 1'b1;
    end
end

endmodule
