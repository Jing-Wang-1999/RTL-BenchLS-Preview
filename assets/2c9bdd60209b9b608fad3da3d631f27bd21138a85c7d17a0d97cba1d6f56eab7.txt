module CLA_MUSER_ACLA4 (
    input Ci,
    input G0, G1, G2, G3,
    input P0, P1, P2, P3,
    output C1, C2, C3,
    output GG, GP
);
    assign C1 = G0 | (P0 & Ci);
    assign C2 = G1 | (P1 & G0) | (P1 & P0 & Ci);
    assign C3 = G2 | (P2 & G1) | (P2 & P1 & G0) | (P2 & P1 & P0 & Ci);
    assign GG = G3 | (P3 & G2) | (P3 & P2 & G1) | (P3 & P2 & P1 & G0);
    assign GP = P3 & P2 & P1 & P0;
endmodule
