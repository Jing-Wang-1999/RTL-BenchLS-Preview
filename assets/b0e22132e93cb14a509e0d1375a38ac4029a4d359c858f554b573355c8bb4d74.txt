`timescale 1ns/1ps
module tb;
reg clk=0; always #5 clk=~clk;
reg rst=1,hvalid=0,valid=0,last=0,user=0;
reg [7:0] data=0;
reg ohready=0,oready=0;
wire hready,ready,ohvalid,ov,ol,ou,he,pe,busy;
wire[7:0] od;wire[15:0] src,dst,ulen,csum;
udp_ip_rx dut(.clk(clk),.rst(rst),.s_ip_hdr_valid(hvalid),.s_ip_hdr_ready(hready),
.s_eth_dest_mac(48'h123456789abc),.s_eth_src_mac(48'hfedcba987654),.s_eth_type(16'h0800),.s_ip_version(4'd4),.s_ip_ihl(4'd5),.s_ip_dscp(6'd7),.s_ip_ecn(2'd1),.s_ip_length(16'd999),.s_ip_identification(16'h2468),.s_ip_flags(3'd2),.s_ip_fragment_offset(13'd0),.s_ip_ttl(8'd64),.s_ip_protocol(8'd17),.s_ip_header_checksum(16'h8888),.s_ip_source_ip(32'h12345678),.s_ip_dest_ip(32'h87654321),
.s_ip_payload_axis_tdata(data),.s_ip_payload_axis_tvalid(valid),.s_ip_payload_axis_tready(ready),.s_ip_payload_axis_tlast(last),.s_ip_payload_axis_tuser(user),
.m_udp_hdr_valid(ohvalid),.m_udp_hdr_ready(ohready),.m_udp_source_port(src),.m_udp_dest_port(dst),.m_udp_length(ulen),.m_udp_checksum(csum),
.m_udp_payload_axis_tdata(od),.m_udp_payload_axis_tvalid(ov),.m_udp_payload_axis_tready(oready),.m_udp_payload_axis_tlast(ol),.m_udp_payload_axis_tuser(ou),.busy(busy),.error_header_early_termination(he),.error_payload_early_termination(pe));
reg[9:0] expected[0:299999];integer ehlen[0:99];
integer pushed=0,popped=0,hpushed=0,hpopped=0,hecount=0,pecount=0,cycles=0,seed=73;
reg held=0;reg[9:0] heldvalue;
always @(negedge clk) begin
seed=(seed*1103515245+12345)&32'h7fffffff;
oready=(cycles%31>9) && (seed%4!=0);
ohready=(cycles%29>11);
end
always @(posedge clk) begin
cycles=cycles+1;
if(!rst) begin
if(held && (!ov || {ou,ol,od}!==heldvalue)) $fatal(1,"output changed under stall");
held=ov && !oready;heldvalue={ou,ol,od};
if(ov&&oready) begin
if(popped>=pushed || {ou,ol,od}!==expected[popped]) $fatal(1,"payload mismatch index=%0d got=%h expected=%h",popped,{ou,ol,od},expected[popped]);
popped=popped+1;
end
if(ohvalid&&ohready) begin
if(hpopped>=hpushed || ulen!==ehlen[hpopped] || src!==16'h1234 || dst!==16'habcd || csum!==16'hbeef) $fatal(1,"header mismatch index=%0d",hpopped);
hpopped=hpopped+1;
end
if(he) hecount=hecount+1;
if(pe) pecount=pecount+1;
end
end
function[7:0] value(input integer n);begin value=(n*37+11)&255;end endfunction
function flag(input integer n);begin flag=(n%7==2);end endfunction
task beat(input[7:0] d,input l,input u);
begin
@(negedge clk);valid=1;data=d;last=l;user=u;
@(posedge clk);while(!ready) @(posedge clk);
end
endtask
task frame(input integer declared,input integer actual,input integer tag);
integer n,target,outcount;
begin
// Independent transaction oracle: declared count is modulo16; zero wraps to65536.
target=(declared-8)&65535;if(target==0)target=65536;
outcount=(actual<target)?actual:target;
for(n=0;n<outcount;n=n+1) begin
expected[pushed]={flag(n),(n==outcount-1),value(n)};
if(n==outcount-1 && actual<target)expected[pushed][9]=1;
if(n==outcount-1 && actual>target)expected[pushed][9]=flag(actual-1);
pushed=pushed+1;
end
ehlen[hpushed]=declared;hpushed=hpushed+1;
@(negedge clk);hvalid=1;
@(posedge clk);while(!hready) @(posedge clk);
@(negedge clk);hvalid=0;
beat(8'h12,0,0);beat(8'h34,0,1);beat(8'hab,0,0);beat(8'hcd,0,0);
beat((declared>>8)&255,0,0);beat(declared&255,0,0);beat(8'hbe,0,0);beat(8'hef,0,0);
for(n=0;n<actual;n=n+1) beat(value(n),n==actual-1,flag(n));
@(negedge clk);valid=0;last=0;user=0;
$display("FRAME tag=%0d length=%0d actual=%0d expected_output=%0d",tag,declared,actual,outcount);
end
endtask
initial begin
repeat(3) @(negedge clk);rst=0;
frame(9,1,1);frame(10,2,2);frame(15,7,3);frame(264,256,4);
frame(15,3,5);frame(9,5,6);frame(15,12,7);
frame(8,1,8);frame(7,2,9);
frame(65535,65527,10);frame(8,65536,11);
while(popped<pushed || hpopped<hpushed) @(negedge clk);
repeat(10) @(negedge clk);
if(pecount!=3 || hecount!=0) $fatal(1,"error pulses pe=%0d he=%0d",pecount,hecount);
$display("PASS frames=%0d payload_beats=%0d payload_errors=%0d header_errors=%0d randomized_backpressure=1",hpopped,popped,pecount,hecount);
$finish;
end
initial begin #10000000;$fatal(1,"timeout");end
endmodule
