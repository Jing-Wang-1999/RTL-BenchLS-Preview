













































































































`timescale 1ns/1ps





































































































































































































































































































 







 


























module gpio_top(
	
	wb_clk_i, wb_rst_i, wb_cyc_i, wb_adr_i, wb_dat_i, wb_sel_i, wb_we_i, wb_stb_i,
	wb_dat_o, wb_ack_o, wb_err_o, wb_inta_o,


	
	aux_i,
 

	
	ext_pad_i, ext_pad_o, ext_padoe_o

  , clk_pad_i 

);

parameter dw = 32;
parameter aw = 7+1;
parameter gw = 31;



input             wb_clk_i;	
input             wb_rst_i;	
input             wb_cyc_i;	
input   [aw-1:0]	wb_adr_i;	
input   [dw-1:0]	wb_dat_i;	
input	  [3:0]     wb_sel_i;	
input             wb_we_i;	
input             wb_stb_i;	
output  [dw-1:0]  wb_dat_o;	
output            wb_ack_o;	
output            wb_err_o;	
output            wb_inta_o;	



input	  [gw-1:0]  aux_i;		
 




input   [gw-1:0]  ext_pad_i;	

input             clk_pad_i;	
 
output  [gw-1:0]  ext_pad_o;	
output  [gw-1:0]  ext_padoe_o;	







reg	[gw-1:0]	rgpio_in;	








reg	[gw-1:0]	rgpio_out;	








reg	[gw-1:0]	rgpio_oe;	








reg	[gw-1:0]	rgpio_inte;	








reg	[gw-1:0]	rgpio_ptrig;	








reg	[gw-1:0]	rgpio_aux;	








reg	[1:0]		rgpio_ctrl;	








reg	[gw-1:0]	rgpio_ints;	








reg	[gw-1:0]	rgpio_eclk;	








reg	[gw-1:0]	rgpio_nec;	








 
reg  [gw-1:0]  sync      ,
               ext_pad_s ;









wire            rgpio_out_sel;  
wire            rgpio_oe_sel; 
wire            rgpio_inte_sel; 
wire            rgpio_ptrig_sel;
wire            rgpio_aux_sel;  
wire            rgpio_ctrl_sel; 
wire            rgpio_ints_sel; 
wire            rgpio_eclk_sel ;
wire            rgpio_nec_sel ;
wire            full_decoding;  
wire  [gw-1:0]  in_muxed; 
wire            wb_ack;   
wire            wb_err;   
wire            wb_inta;  
reg   [dw-1:0]  wb_dat;   

reg             wb_ack_o; 
reg             wb_err_o; 
reg             wb_inta_o;  
reg   [dw-1:0]  wb_dat_o; 

wire  [gw-1:0]  out_pad;  

reg   [gw-1:0]  ext_pad_o;  


wire  [gw-1:0]  extc_in;  
wire  [gw-1:0]  pext_clk; 
reg   [gw-1:0]  pextc_sampled;  






reg   [gw-1:0]  nextc_sampled;  

 












assign wb_ack = wb_cyc_i & wb_stb_i & !wb_err_o;





always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		wb_ack_o <=  1'b0;
	else
		wb_ack_o <=  wb_ack & ~wb_ack_o & (!wb_err) ;











assign wb_err = wb_cyc_i & wb_stb_i & !full_decoding;













always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		wb_err_o <=  1'b0;
	else
		wb_err_o <=  wb_err & ~wb_err_o;








assign full_decoding = (wb_adr_i[7:6] == {7-6+1{1'b0}}) &
			(wb_adr_i[1:0] == {1-0+1{1'b0}});








assign rgpio_out_sel = wb_cyc_i & wb_stb_i & (wb_adr_i[6-1:1+1] == 4'h1) & full_decoding;


assign rgpio_oe_sel = wb_cyc_i & wb_stb_i & (wb_adr_i[6-1:1+1] == 4'h2) & full_decoding;


assign rgpio_inte_sel = wb_cyc_i & wb_stb_i & (wb_adr_i[6-1:1+1] == 4'h3) & full_decoding;


assign rgpio_ptrig_sel = wb_cyc_i & wb_stb_i & (wb_adr_i[6-1:1+1] == 4'h4) & full_decoding;


assign rgpio_aux_sel = wb_cyc_i & wb_stb_i & (wb_adr_i[6-1:1+1] == 4'h5) & full_decoding;


assign rgpio_ctrl_sel = wb_cyc_i & wb_stb_i & (wb_adr_i[6-1:1+1] == 4'h6) & full_decoding;


assign rgpio_ints_sel = wb_cyc_i & wb_stb_i & (wb_adr_i[6-1:1+1] == 4'h7) & full_decoding;


assign rgpio_eclk_sel = wb_cyc_i & wb_stb_i & (wb_adr_i[6-1:1+1] == 4'h8) & full_decoding;


assign rgpio_nec_sel = wb_cyc_i & wb_stb_i & (wb_adr_i[6-1:1+1] == 4'h9) & full_decoding;







always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		rgpio_ctrl <=  2'b0;
	else if (rgpio_ctrl_sel && wb_we_i)
		rgpio_ctrl <=  wb_dat_i[1:0];
	else if (rgpio_ctrl[0])
		rgpio_ctrl[1] <=  rgpio_ctrl[1] | wb_inta_o;








always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		rgpio_out <=  {gw{1'b0}};
	else if (rgpio_out_sel && wb_we_i)
    begin





     if ( wb_sel_i [3] == 1'b1 )
       rgpio_out [gw-1:24] <=  wb_dat_i [gw-1:24] ;
     if ( wb_sel_i [2] == 1'b1 )
       rgpio_out [23:16] <=  wb_dat_i [23:16] ;
     if ( wb_sel_i [1] == 1'b1 )
       rgpio_out [15:8] <=  wb_dat_i [15:8] ;
     if ( wb_sel_i [0] == 1'b1 )
       rgpio_out [7:0] <=  wb_dat_i [7:0] ;



















   end









always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		rgpio_oe <=  {gw{1'b0}};
	else if (rgpio_oe_sel && wb_we_i)
  begin





     if ( wb_sel_i [3] == 1'b1 )
       rgpio_oe [gw-1:24] <=  wb_dat_i [gw-1:24] ;
     if ( wb_sel_i [2] == 1'b1 )
       rgpio_oe [23:16] <=  wb_dat_i [23:16] ;
     if ( wb_sel_i [1] == 1'b1 )
       rgpio_oe [15:8] <=  wb_dat_i [15:8] ;
     if ( wb_sel_i [0] == 1'b1 )
       rgpio_oe [7:0] <=  wb_dat_i [7:0] ;



















   end









always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		rgpio_inte <=  {gw{1'b0}};
	else if (rgpio_inte_sel && wb_we_i)
  begin





     if ( wb_sel_i [3] == 1'b1 )
       rgpio_inte [gw-1:24] <=  wb_dat_i [gw-1:24] ;
     if ( wb_sel_i [2] == 1'b1 )
       rgpio_inte [23:16] <=  wb_dat_i [23:16] ;
     if ( wb_sel_i [1] == 1'b1 )
       rgpio_inte [15:8] <=  wb_dat_i [15:8] ;
     if ( wb_sel_i [0] == 1'b1 )
       rgpio_inte [7:0] <=  wb_dat_i [7:0] ;



















   end










always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		rgpio_ptrig <=  {gw{1'b0}};
	else if (rgpio_ptrig_sel && wb_we_i)
  begin





     if ( wb_sel_i [3] == 1'b1 )
       rgpio_ptrig [gw-1:24] <=  wb_dat_i [gw-1:24] ;
     if ( wb_sel_i [2] == 1'b1 )
       rgpio_ptrig [23:16] <=  wb_dat_i [23:16] ;
     if ( wb_sel_i [1] == 1'b1 )
       rgpio_ptrig [15:8] <=  wb_dat_i [15:8] ;
     if ( wb_sel_i [0] == 1'b1 )
       rgpio_ptrig [7:0] <=  wb_dat_i [7:0] ;



















   end
    








always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		rgpio_aux <=  {gw{1'b0}};
	else if (rgpio_aux_sel && wb_we_i)
  begin





     if ( wb_sel_i [3] == 1'b1 )
       rgpio_aux [gw-1:24] <=  wb_dat_i [gw-1:24] ;
     if ( wb_sel_i [2] == 1'b1 )
       rgpio_aux [23:16] <=  wb_dat_i [23:16] ;
     if ( wb_sel_i [1] == 1'b1 )
       rgpio_aux [15:8] <=  wb_dat_i [15:8] ;
     if ( wb_sel_i [0] == 1'b1 )
       rgpio_aux [7:0] <=  wb_dat_i [7:0] ;



















   end










always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		rgpio_eclk <=  {gw{1'b0}};
	else if (rgpio_eclk_sel && wb_we_i)
  begin





     if ( wb_sel_i [3] == 1'b1 )
       rgpio_eclk [gw-1:24] <=  wb_dat_i [gw-1:24] ;
     if ( wb_sel_i [2] == 1'b1 )
       rgpio_eclk [23:16] <=  wb_dat_i [23:16] ;
     if ( wb_sel_i [1] == 1'b1 )
       rgpio_eclk [15:8] <=  wb_dat_i [15:8] ;
     if ( wb_sel_i [0] == 1'b1 )
       rgpio_eclk [7:0] <=  wb_dat_i [7:0] ;



















   end












always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		rgpio_nec <=  {gw{1'b0}};
	else if (rgpio_nec_sel && wb_we_i)
  begin





     if ( wb_sel_i [3] == 1'b1 )
       rgpio_nec [gw-1:24] <=  wb_dat_i [gw-1:24] ;
     if ( wb_sel_i [2] == 1'b1 )
       rgpio_nec [23:16] <=  wb_dat_i [23:16] ;
     if ( wb_sel_i [1] == 1'b1 )
       rgpio_nec [15:8] <=  wb_dat_i [15:8] ;
     if ( wb_sel_i [0] == 1'b1 )
       rgpio_nec [7:0] <=  wb_dat_i [7:0] ;



















   end










always @(posedge wb_clk_i or posedge wb_rst_i)
  if (wb_rst_i) begin
    sync      <=  {gw{1'b0}} ; 
    ext_pad_s <=  {gw{1'b0}} ; 
  end else begin
    sync      <=  ext_pad_i  ; 
    ext_pad_s <=  sync       ; 
  end


 
  




always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		rgpio_in <=  {gw{1'b0}};
	else
		rgpio_in <=  in_muxed;













reg  sync_clk,
     clk_s   ,
     clk_r   ;
wire pedge   ,
     nedge   ;
wire [gw-1:0] pedge_vec   ,
              nedge_vec   ;
wire [gw-1:0] in_lach     ;

assign pedge =  clk_s & !clk_r ; 
assign nedge = !clk_s &  clk_r ; 
assign pedge_vec = {gw{pedge}} ;   
assign nedge_vec = {gw{nedge}} ;   

assign in_lach = (~rgpio_nec & pedge_vec) | (rgpio_nec & nedge_vec) ;
assign extc_in = (in_lach & ext_pad_s) | (~in_lach & pextc_sampled) ;

always @(posedge wb_clk_i or posedge wb_rst_i)
  if (wb_rst_i) begin
    sync_clk <=  1'b0 ;
    clk_s    <=  1'b0 ;
    clk_r    <=  1'b0 ;
  end else begin
    sync_clk <=  clk_pad_i ;
    clk_s    <=  sync_clk  ;
    clk_r    <=  clk_s     ;
  end

always @(posedge wb_clk_i or posedge wb_rst_i)
  if (wb_rst_i) begin
    pextc_sampled <=  {gw{1'b0}};
  end else begin
    pextc_sampled <=  extc_in ;
  end

assign in_muxed = (rgpio_eclk & pextc_sampled) | (~rgpio_eclk & ext_pad_s) ;








































































































































 






 






always @(wb_adr_i or rgpio_in or rgpio_out or rgpio_oe or rgpio_inte or
		rgpio_ptrig or rgpio_aux or rgpio_ctrl or rgpio_ints or rgpio_eclk or rgpio_nec)
	case (wb_adr_i[6-1:1+1])	

  
  	4'h1: begin
			wb_dat[dw-1:0] = rgpio_out;
		end
  
  
		4'h2: begin
			wb_dat[dw-1:0] = rgpio_oe;
		end
  
  
		4'h3: begin
			wb_dat[dw-1:0] = rgpio_inte;
		end
  
  
		4'h4: begin
			wb_dat[dw-1:0] = rgpio_ptrig;
		end
  
  
		4'h9: begin
			wb_dat[dw-1:0] = rgpio_nec;
		end
  
  
		4'h8: begin
			wb_dat[dw-1:0] = rgpio_eclk;
		end
  
  
		4'h5: begin
			wb_dat[dw-1:0] = rgpio_aux;
		end
  
  
		4'h6: begin
			wb_dat[1:0] = rgpio_ctrl;
			wb_dat[dw-1:2] = {dw-2{1'b0}};
		end
  

  
		4'h7: begin
			wb_dat[dw-1:0] = rgpio_ints;
		end
  
		default: begin
			wb_dat[dw-1:0] = rgpio_in;
		end
	endcase





always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		wb_dat_o <=  {dw{1'b0}};
	else
		wb_dat_o <=  wb_dat;








always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		rgpio_ints <=  {gw{1'b0}};
	else if (rgpio_ints_sel && wb_we_i)
		rgpio_ints <=  wb_dat_i[gw-1:0];
	else if (rgpio_ctrl[0])
		rgpio_ints <=  (rgpio_ints | ((in_muxed ^ rgpio_in) & ~(in_muxed ^ rgpio_ptrig)) & rgpio_inte);







assign wb_inta = |rgpio_ints ? rgpio_ctrl[0] : 1'b0;





always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		wb_inta_o <=  1'b0;
	else
		wb_inta_o <=  wb_inta;


 




assign ext_padoe_o = rgpio_oe;





assign out_pad = rgpio_out & ~rgpio_aux | aux_i & rgpio_aux;


 





always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		ext_pad_o <=  {gw{1'b0}};
	else
		ext_pad_o <=  out_pad;


 



















 

endmodule
