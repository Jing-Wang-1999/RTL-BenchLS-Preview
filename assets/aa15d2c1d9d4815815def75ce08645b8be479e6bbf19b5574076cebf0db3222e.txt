    k_e <= 1'b0;
    k_last <= 1'b0;
    k_data <= 16'd0;
    if(j_sof) begin
        if(k_header_i < 4'd12) begin
            k_e <= 1'b1;
            k_data <= jls_header[k_header_i];

        end

    end
