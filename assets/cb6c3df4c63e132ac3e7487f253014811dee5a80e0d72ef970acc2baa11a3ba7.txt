module FP32_TO_FP17_mgc_out_stdreg_wait_v1
#(
    parameter integer rscid = 0,
    parameter integer width = 32
)
(
    input wire ld,
    input wire [width-1:0] d,
    output wire lz,
    output wire [width-1:0] z,
    input wire vz,
    output wire vd
);

    // Mirror the input load signal to the output load signal
    assign lz = ld;

    // Mirror the input data to the output data
    assign z = d;

    // Mirror the input valid signal to the output valid signal
    assign vd = vz;

endmodule
