module rtlpp_10333 (
    input [31:0] A_mul_src1,
    input [31:0] A_mul_src2,
    input clk,
    input reset_n,
    output reg [31:0] A_mul_cell_result
);

    reg [31:0] part_1_result;
    reg [31:0] part_2_result;

    always @(posedge clk or negedge reset_n) begin
        if (!reset_n) begin
            part_1_result <= 32'b0;
            part_2_result <= 32'b0;
            A_mul_cell_result <= 32'b0;
        end else begin
            // Part 1: Multiply lower 16 bits of both operands
            part_1_result <= A_mul_src1[15:0] * A_mul_src2[15:0];

            // Part 2: Multiply upper 16 bits of A_mul_src1 with lower 16 bits of A_mul_src2
            part_2_result <= {16'b0, A_mul_src1[31:16] * A_mul_src2[15:0]};

            // Combine results
            A_mul_cell_result <= {part_1_result[31:16] + part_2_result[15:0], part_1_result[15:0]};
        end
    end

endmodule
