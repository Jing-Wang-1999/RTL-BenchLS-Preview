module OBUF #(parameter CAPACITANCE = "DONT_CARE",
              parameter DRIVE = 12,
              parameter IOSTANDARD = "DEFAULT",
              parameter SLEW = "SLOW")
             (input wire I,
              output wire O);

    // Global tristate signal
    wire GTS = 1'b0; // Set to inactive (0) to allow signal transfer

    // Tristate buffer logic
    assign O = GTS ? 1'bz : I;

endmodule
