def normalize(t):
 # Remove only initial blocks that contain diagnostic $error and no assignments.
 removed=[]
 for m in reversed(list(re.finditer(r'\binitial\s+begin\b',t))):
  depth=0;end=None
  for tok in re.finditer(r'\bbegin\b|\bend\b',t[m.start():]):
   depth+=1 if tok[0]=='begin' else -1
   if depth==0:end=m.start()+tok.end();break
  block=t[m.start():end]
  if '$error' in block:
   assert not re.search(r'(?<![=!<>])=(?!=)',re.sub(r'"[^"\n]*"','',block)),block
   removed.append(block);t=t[:m.start()]+'// Curator: compile-time parameter diagnostics omitted.\n'+t[end:]
 return t,removed
