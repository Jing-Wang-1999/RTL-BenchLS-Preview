module origen_128895(
    input [99:0] a,
    input [99:0] b,
    input cin,
    output [99:0] sum,
    output cout,
    output [6:0] ones_count
);

    wire [24:0] group_carry;
    wire [4:0] super_carry;

    // Level 1: 25 4-bit CLA blocks
    genvar i;
    generate
        for (i = 0; i < 25; i = i + 1) begin : cla_4bit
            cla_4bit cla_4bit_inst (
                .a(a[i*4+3:i*4]),
                .b(b[i*4+3:i*4]),
                .cin(i == 0 ? cin : group_carry[i-1]),
                .sum(sum[i*4+3:i*4]),
                .cout(group_carry[i])
            );
        end
    endgenerate

    // Level 2: 5 5-bit CLA blocks for group carries
    genvar j;
    generate
        for (j = 0; j < 5; j = j + 1) begin : cla_5bit
            cla_5bit cla_5bit_inst (
                .a(group_carry[j*5+4:j*5]),
                .cin(j == 0 ? cin : super_carry[j-1]),
                .cout(super_carry[j])
            );
        end
    endgenerate

    // Level 3: Final carry-out
    assign cout = super_carry[4];

    // Ones counter
    ones_counter ones_counter_inst (
        .sum(sum),
        .count(ones_count)
    );

endmodule

module cla_4bit(
    input [3:0] a,
    input [3:0] b,
    input cin,
    output [3:0] sum,
    output cout
);
    wire [3:0] g, p;
    wire [4:0] c;

    assign g = a & b;
    assign p = a ^ b;
    assign c[0] = cin;

    genvar i;
    generate
        for (i = 0; i < 4; i = i + 1) begin : cla_logic
            assign c[i+1] = g[i] | (p[i] & c[i]);
            assign sum[i] = p[i] ^ c[i];
        end
    endgenerate

    assign cout = c[4];
endmodule

module cla_5bit(
    input [4:0] a,
    input cin,
    output cout
);
    wire [4:0] g, p;
    wire [5:0] c;

    assign g = a;
    assign p = 1'b1;
    assign c[0] = cin;

    genvar i;
    generate
        for (i = 0; i < 5; i = i + 1) begin : cla_logic
            assign c[i+1] = g[i] | (p[i] & c[i]);
        end
    endgenerate

    assign cout = c[5];
endmodule

module ones_counter(
    input [99:0] sum,
    output reg [6:0] count
);
    integer i;
    always @(*) begin
        count = 0;
        for (i = 0; i < 100; i = i + 1) begin
            if (sum[i]) count = count + 1;
        end
    end
endmodule
