module mg_06566 (vpp,pi_efa_prog_en,sbc_efa_read_en,sbc_efa_word_addr,sbc_efa_bit_addr,sbc_efa_margin0_rd,sbc_efa_margin1_rd,efa_sbc_data,pwr_ok,por_n,sbc_efa_sup_det_rd,sbc_efa_power_down,so,si,se,vddo,clk);
input vpp; 
 output [31:0] efa_sbc_data; 
 input pi_efa_prog_en; 
 input sbc_efa_read_en; 
 input [5:0] sbc_efa_word_addr; 
 input [4:0] sbc_efa_bit_addr; 
 input sbc_efa_margin0_rd; 
 input sbc_efa_margin1_rd; 
 input pwr_ok; 
 input por_n; 
 input sbc_efa_sup_det_rd; 
 input sbc_efa_power_down; 
 output so; 
 input si; 
 input se; 
 input vddo; 
 input clk; 
 parameter MAXFILENAME=200; 
 parameter EFA_READ_LAT = 45000 ; 
 reg [MAXFILENAME*8-1:0] efuse_data_filename; 
 reg [31:0] efuse_array[0:63],efuse_row,efa_read_data; 
 integer file_get_status,i; 
 reg [31:0] fpInVec; 
 wire [31:0] efa_sbc_data; 
 wire l1clk; 
 wire lvl_det_l; 
 wire vddc_ok_l; 
 wire vddo_ok_l; 
 wire vpp_ok_l; 
 reg efuse_rd_progress; 
 reg efuse_enable_write_check; 
 initial 
 begin 
  efuse_enable_write_check = 1; 
  if ($value$plusargs("efuse_data_file=%s", efuse_data_filename)) 
  begin 
 $display("INFO: efuse data file is being read--filename=%0s",efuse_data_filename); 
  $readmemh(efuse_data_filename, efuse_array); 
  $display("INFO: completed reading efuse data file"); 
  end 
  else 
  begin 
  $display("INFO: Using default efuse data for the efuse array"); 
  for (i=0;i<=63;i=i+1) begin 
  efuse_array[i] = 32'b0; 
  end 
  end 
 end 
 assign l1clk = clk & ~sbc_efa_power_down; 
 assign so = se ? si : 1'bx; 
 assign vddc_ok_l = 1'b0; 
 assign vddo_ok_l = 1'b0; 
 assign vpp_ok_l = 1'b0; 
 assign lvl_det_l = 1'b0; 
 always @(posedge l1clk) begin 
  if ((pi_efa_prog_en === 1'b1) && (pwr_ok === 1'b1) && (por_n === 1'b1)) begin 
  efuse_row = efuse_array[sbc_efa_word_addr]; 
  efuse_row[sbc_efa_bit_addr] = 1'b1; 
  efuse_array[sbc_efa_word_addr] <= efuse_row; 
  end 
 end 
 always @(posedge l1clk) begin 
  if ((sbc_efa_read_en) & ~efuse_rd_progress) begin 
  efa_read_data[31:0] <= #EFA_READ_LAT efuse_array[sbc_efa_word_addr]; 
  efuse_rd_progress = 1'b1; 
  end 
  if (~(sbc_efa_read_en)) begin 
  efuse_rd_progress = 1'b0; 
  end 
  if (~efuse_rd_progress) begin 
  efa_read_data[31:0] <= 32'bx; 
  end 
 end 
 assign efa_sbc_data[31:0] = por_n ? ((pwr_ok & sbc_efa_read_en) ? (sbc_efa_sup_det_rd ?{28'bx,~lvl_det_l,~vddc_ok_l,~vddo_ok_l,~vpp_ok_l}: efa_read_data[31:0] ) : efa_sbc_data[31:0]) : 32'b0; 
 endmodule
