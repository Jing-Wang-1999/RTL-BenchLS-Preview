module mg_01336 (
    input wire [1:0] in,
    input wire enable,
    output reg out
);
    always @(*) begin
        if (!enable) begin
            out = 1'b0;
        end else begin
            case (in)
                2'b00: out = 1'b0;
                2'b01: out = 1'b1;
                2'b10: out = 1'b0;
                2'b11: out = 1'b1;
                default: out = 1'b0;
            endcase
        end
    end
endmodule
