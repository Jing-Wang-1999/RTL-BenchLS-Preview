module origen_059421 (
    input wire CLK,
    input wire ENABLE,
    input wire SW,
    input wire [3:0] L1, L2, L3, L4, L5, L6,
    output reg [3:0] SA,
    output reg [3:0] L
);

    parameter MAX_COUNT = 3'd7;

    reg [2:0] sa_count_tmp;
    reg [3:0] sa_count;
    reg [3:0] L_tmp;

    always @(posedge CLK) begin
        if (ENABLE) begin
            if (sa_count_tmp == MAX_COUNT)
                sa_count_tmp <= 3'd0;
            else
                sa_count_tmp <= sa_count_tmp + 1'b1;
        end
    end

    always @(posedge CLK) begin
        if (sa_count_tmp[0] == 1'b0) begin
            sa_count <= 4'b1111;
        end else begin
            case (sa_count_tmp[2:1])
                2'b00: begin
                    sa_count <= 4'b1110;
                    L_tmp <= SW ? L3 : L1;
                end
                2'b01: begin
                    sa_count <= 4'b1101;
                    L_tmp <= SW ? L4 : L2;
                end
                2'b10: begin
                    sa_count <= 4'b1011;
                    L_tmp <= SW ? L5 : L3;
                end
                2'b11: begin
                    sa_count <= 4'b0111;
                    L_tmp <= SW ? L6 : L4;
                end
            endcase
        end
    end

    always @(posedge CLK) begin
        SA <= ~sa_count;
        L <= L_tmp;
    end

endmodule
