module rtlpp_00226(
    input clk,
    input reset,
    input [7:0] in_port,
    output reg [7:0] out_port
);

    wire [4:0] op5;
    wire [5:0] op6;
    wire [7:0] imm8, dmd, x, y;
    wire [11:0] imm12;
    wire [7:0] x0, y0;
    reg [7:0] s [0:15];
    reg [11:0] lreg [0:30];
    reg [11:0] pc;
    reg [4:0] lcount;
    wire [11:0] pc1;
    wire [11:0] dma;
    wire [17:0] pmd, ir;
    wire mem_ena, jc, not_clk, kflag;
    reg z_new, c_new;
    reg z, c;
    reg [8:0] res;
    wire [3:0] rd, rs;

    parameter
        add     = 5'b01000,
        addcy   = 5'b01001,
        sub     = 5'b01100,
        subcy   = 5'b01101,
        opand   = 5'b00001,
        opxor   = 5'b00010,
        opor    = 5'b00011,
        opinput = 5'b00100,
        opoutput= 5'b00101,
        store   = 5'b10110,
        fetch   = 5'b10111,
        load    = 5'b11000,
        jump    = 6'b100010,
        jumpz   = 6'b110010,
        jumpnz  = 6'b110110,
        call    = 6'b100000,
        opreturn= 6'b100101;

    always @(negedge clk or negedge reset) begin : P1
        if (~reset) begin 
            pc <= 0;
            lcount <= 0;
        end else begin
            if (op6 == call) begin
                lreg[lcount] <= pc1; 
                lcount <= lcount + 1;
            end
            if (op6 == opreturn) begin
                pc <= lreg[lcount-1]; 
                lcount <= lcount -1;
            end else if (jc)
                pc <= imm12;
            else
                pc <= pc1;
        end
    end

    assign pc1 = pc + 1;
    assign jc = (op6 == jumpz && z) || (op6 == jumpnz && ~z) || (op6 == jump) || (op6 == call);
    assign op6 = ir[17:12]; 
    assign op5 = ir[17:13]; 
    assign kflag = ir[12];
    assign imm8 = ir[7:0];
    assign imm12 = ir[11:0]; 
    assign rd = ir[11:8];
    assign rs = ir[7:4];
    assign x = s[rd];
    assign x0 = {1'b0 , x}; 
    assign y = (kflag) ? imm8 : s[rs]; 
    assign y0 = {1'b0, y};

    // ROM 
    reg [17:0] rom [0:4095];
    initial $readmemb("rom4096x18.hex", rom);
    assign pmd = rom[pc];

    // RAM 
    reg [7:0] ram [0:255];
    always @(posedge not_clk) begin
        if (mem_ena) 
            ram[y] <= x;
    end
    assign dmd = ram[y];

    assign ir = pmd;
    assign not_clk = ~clk;
    assign mem_ena = (op5 == store) ? 1 : 0;

    always @(*) begin : P2
        case (op5)
            add      : res = x0 + y0;
            addcy    : res = x0 + y0 + c;
            sub      : res = x0 - y0;
            subcy    : res = x0 - y0 - c;
            opand    : res = x0 & y0;
            opor     : res = x0 | y0;
            opxor    : res = x0 ^ y0;
            load     : res = y0;
            opinput  : res = {1'b0 , in_port};
            fetch    : res = {1'b0 , dmd};
            default  : res = x0;
        endcase
        z_new = (res == 0) ? 1 : 0;
        c_new = res[8];
    end

    always @(posedge clk or negedge reset) begin : P3
        if (~reset) begin
            z <= 0; 
            c <= 0; 
            out_port <= 0;
        end else begin
            case (op5) 
                addcy, subcy : begin 
                    z <= z & z_new;
                    c <= c_new; 
                end
                add , sub : begin 
                    z <= z_new; 
                    c <= c_new; 
                end
                opor , opand, opxor : begin 
                    z <= z_new; 
                    c <= 0; 
                end
                default : begin 
                    z <= z; 
                    c <= c; 
                end
            endcase
            s[rd] <= res[7:0];
            out_port = (op5 == opoutput) ? x : out_port;
        end
    end

endmodule
