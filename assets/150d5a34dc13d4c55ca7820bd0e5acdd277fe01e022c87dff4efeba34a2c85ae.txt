module origen_006185 (
    input clk,
    input rst_n,
    input [2:0] opcode,
    input [7:0] dina,
    input [7:0] dinb,
    output reg cout,
    output reg [7:0] dout
);

    wire [8:0] sum_result;
    wire [7:0] abs_diff_result;
    wire [7:0] shift_result;
    
    // Addition operation
    assign sum_result = {1'b0, dina} + {1'b0, dinb};
    
    // Absolute difference operation
    assign abs_diff_result = (dina >= dinb) ? (dina - dinb) : (dinb - dina);
    
    // Logical left shift operation
    assign shift_result = dina << dinb[2:0];
    
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            dout <= 8'b0;
            cout <= 1'b0;
        end else begin
            case (opcode)
                3'b000: begin  // Addition
                    dout <= sum_result[7:0];
                    cout <= sum_result[8];
                end
                3'b001: begin  // Absolute difference
                    dout <= abs_diff_result;
                    cout <= 1'b0;
                end
                3'b010: begin  // Bitwise AND
                    dout <= dina & dinb;
                    cout <= 1'b0;
                end
                3'b011: begin  // Bitwise OR
                    dout <= dina | dinb;
                    cout <= 1'b0;
                end
                3'b100: begin  // Bitwise XOR
                    dout <= dina ^ dinb;
                    cout <= 1'b0;
                end
                3'b101: begin  // Logical left shift
                    dout <= shift_result;
                    cout <= 1'b0;
                end
                3'b110: begin  // Logical right shift
                    dout <= dina >> dinb[2:0];
                    cout <= 1'b0;
                end
                default: begin  // Default case (opcode 111)
                    dout <= 8'b0;
                    cout <= 1'b0;
                end
            endcase
        end
    end

endmodule
