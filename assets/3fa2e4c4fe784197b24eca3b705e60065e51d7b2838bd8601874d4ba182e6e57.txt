module autovcoder_001201 #(parameter lpm_width = 1, parameter lpm_size = 1) (
    input  wire [lpm_size*lpm_width-1:0] data,
    output reg  [lpm_width-1:0] result
);

    // Check for valid parameters
    initial begin
        if (lpm_width <= 0 || lpm_size <= 0) begin
            $display("Error: lpm_width and lpm_size must be greater than zero.");
            $finish;
        end
    end

    integer i, j;
    always @(*) begin
        for (i = 0; i < lpm_width; i = i + 1) begin
            result[i] = 1'b1; // Initialize result bit to '1'
            for (j = 0; j < lpm_size; j = j + 1) begin
                result[i] = result[i] & data[j*lpm_width + i];
            end
        end
    end

endmodule
