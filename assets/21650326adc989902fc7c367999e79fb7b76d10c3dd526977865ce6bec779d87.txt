module ct_rtu_expand_8 (
    input [2:0] x_num,
    output [7:0] x_num_expand
);

assign x_num_expand = (x_num == 3'd0) ? 8'b00000001 :
                      (x_num == 3'd1) ? 8'b00000010 :
                      (x_num == 3'd2) ? 8'b00000100 :
                      (x_num == 3'd3) ? 8'b00001000 :
                      (x_num == 3'd4) ? 8'b00010000 :
                      (x_num == 3'd5) ? 8'b00100000 :
                      (x_num == 3'd6) ? 8'b01000000 :
                      (x_num == 3'd7) ? 8'b10000000 :
                      8'b00000000;

endmodule
