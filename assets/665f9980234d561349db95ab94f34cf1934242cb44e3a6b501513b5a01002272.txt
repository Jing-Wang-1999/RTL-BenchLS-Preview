`timescale 1ns/1ps
module tb;
reg clk=0; always #5 clk=~clk;
reg rst=1, sa=0,sb=0; reg[31:0] a=0,b=0;
wire aa,ab,v;wire[31:0] z;
adder dut(a,b,sa,sb,1'b1,clk,rst,z,v,aa,ab);
task run(input[31:0] av,bv,expected,input integer latency);
integer n;
begin
@(negedge clk); a=av;sa=1;
wait(aa); @(posedge clk); @(negedge clk);sa=0;b=bv;sb=1;
wait(ab); @(posedge clk); #1; n=0;
@(negedge clk); sb=0;
while(!v) begin @(posedge clk);#1;n=n+1;end
$display("a=%h b=%h z=%h latency=%0d",av,bv,z,n);
if(z!==expected || n!=latency) $fatal(1,"mismatch");
wait(!v);
end
endtask
initial begin
repeat(2) @(negedge clk);rst=0;
run(32'h3f800000,32'h3f800000,32'h40000000,10);
run(32'h3f800000,32'h3f000000,32'h3fc00000,11);
run(32'h3f800000,32'hbf800000,32'h00000000,136);
run(32'h00000000,32'hbf800000,32'hbf800000,3);
run(32'h7f800000,32'hff800000,32'hffc00000,3);
run(32'hff800000,32'h7f800000,32'h7fc00000,3);
run(32'h00000001,32'h00000001,32'h00000002,10);
run(32'h3f800000,32'hbf7fffff,32'h33800000,35);
$display("PASS 8 arithmetic/timing probes");$finish;
end
initial begin #100000; $fatal(1,"timeout");end
endmodule
