module SDP_X_X_alu_core_cfg_alu_shift_value_rsc_triosy_obj_cfg_alu_shift_value_rsc_triosy_wait_ctrl (
  input wire cfg_alu_shift_value_rsc_triosy_obj_oswt,
  input wire core_wen,
  input wire core_wten,
  input wire cfg_alu_shift_value_rsc_triosy_obj_iswt0,
  output wire cfg_alu_shift_value_rsc_triosy_obj_biwt,
  output wire cfg_alu_shift_value_rsc_triosy_obj_bdwt
);

  // Output wait condition is met when output wait control is enabled and core is enabled for writing
  assign cfg_alu_shift_value_rsc_triosy_obj_bdwt = cfg_alu_shift_value_rsc_triosy_obj_oswt && core_wen;

  // Input wait condition is met when input wait control is enabled and core is not in a wait state
  assign cfg_alu_shift_value_rsc_triosy_obj_biwt = cfg_alu_shift_value_rsc_triosy_obj_iswt0 && !core_wten;

endmodule
