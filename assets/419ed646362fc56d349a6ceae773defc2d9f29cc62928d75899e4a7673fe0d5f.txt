`timescale 1ns/1ps
module tb;
reg clk=0;always #5 clk=~clk;
reg rst=1,hvalid=0,valid=0,last=0;reg[7:0] data=0;
wire hready,ready,ohvalid,herror,perror;
udp_ip_rx dut(.clk(clk),.rst(rst),.s_ip_hdr_valid(hvalid),.s_ip_hdr_ready(hready),.s_ip_payload_axis_tdata(data),.s_ip_payload_axis_tvalid(valid),.s_ip_payload_axis_tready(ready),.s_ip_payload_axis_tlast(last),.s_ip_payload_axis_tuser(1'b0),.m_udp_hdr_valid(ohvalid),.m_udp_hdr_ready(1'b1),.m_udp_payload_axis_tready(1'b1),.error_header_early_termination(herror),.error_payload_early_termination(perror));
integer k;
initial begin
repeat(3) @(negedge clk);rst=0;
while(!hready) @(negedge clk);
hvalid=1;@(negedge clk);hvalid=0;
for(k=0;k<8;k=k+1) begin
while(!ready) @(negedge clk);
valid=1;data=(k==5)?8:0;last=(k==7);
@(negedge clk);
end
valid=0;last=0;
if(!herror || ohvalid) $fatal(1,"expected header-only frame rejected");
$display("CONFIRMED zero_payload_unsupported: 8-byte UDP header with length=8 and final tlast raises header error, header valid=0");
$finish;
end
initial begin #10000;$fatal(1,"probe timeout");end
endmodule
