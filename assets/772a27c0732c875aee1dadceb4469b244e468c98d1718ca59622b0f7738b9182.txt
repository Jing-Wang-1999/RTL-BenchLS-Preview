













































module uart_txfsm (
             reset_n        ,
             baud_clk_16x   ,

             cfg_tx_enable  ,
             cfg_stop_bit   ,
             cfg_pri_mod    ,

       
             fifo_empty     ,
             fifo_rd        ,
             fifo_data      ,

          
             so  
          );


input             reset_n        ; 
input             baud_clk_16x   ; 

input             cfg_tx_enable  ; 
input             cfg_stop_bit   ; 
                                   
input   [1:0]     cfg_pri_mod    ;
                                   
                                   
                                   




input             fifo_empty     ; 
output            fifo_rd        ; 
input  [7:0]      fifo_data      ; 


output            so             ;  


reg  [2:0]         txstate       ; 
reg                so            ; 
reg  [7:0]         txdata        ; 
reg                fifo_rd       ; 
reg  [2:0]         cnt           ; 
reg  [3:0]         divcnt        ; 

parameter idle_st      = 3'b000;
parameter xfr_data_st  = 3'b001;
parameter xfr_pri_st   = 3'b010;
parameter xfr_stop_st1 = 3'b011;
parameter xfr_stop_st2 = 3'b100;


always @(negedge reset_n or posedge baud_clk_16x)
begin
   if(reset_n == 1'b0) begin
      txstate  <= idle_st;
      so       <= 1'b1;
      cnt      <= 3'b0;
      txdata   <= 8'h0;
      fifo_rd  <= 1'b0;
      divcnt   <= 4'b0;
   end
   else begin
      divcnt <= divcnt+1;
      if(divcnt == 4'b0000) begin 
         case(txstate)
          idle_st      : begin
               if(!fifo_empty && cfg_tx_enable) begin
                  so       <= 1'b0 ; 
                  cnt      <= 3'b0;
                  fifo_rd  <= 1'b1;
                  txdata   <= fifo_data;
                  txstate  <= xfr_data_st;  
               end
            end

          xfr_data_st  : begin
              fifo_rd  <= 1'b0;
              so   <= txdata[cnt];
              cnt  <= cnt+1;
              if(cnt == 7) begin
                 if(cfg_pri_mod == 2'b00) begin 
                    txstate  <= xfr_stop_st1;  
                 end
                 else begin
                    txstate <= xfr_pri_st;  
                 end
              end
           end

          xfr_pri_st   : begin
               if(cfg_pri_mod == 2'b10)  
                   so <= ^txdata;
               else begin 
                   so <= ~(^txdata);
               end
               txstate  <= xfr_stop_st1;  
            end

          xfr_stop_st1  : begin 
               so <= 1;
               if(cfg_stop_bit == 0)  
                    txstate <= idle_st;
               else 
                  txstate  <= xfr_stop_st2;
            end

          xfr_stop_st2  : begin 
               so <= 1;
               txstate <= idle_st;
            end
         endcase
      end else begin
        fifo_rd  <= 1'b0;
     end
   end
end


endmodule
