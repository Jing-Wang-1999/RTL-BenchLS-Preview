module rtlpp_04677(
    input start,
    output rfd,
    input [7:0] phs,
    input iv,
    input iocs,
    input [2:0] ioaddr,
    input [15:0] din,
    output [15:0] dout,
    input iowr, iord,
    input clk, rst
);

wire first, last;
reg [3:0] z;
reg [7:0] n;
wire zce, nce;
reg dzce, dnce;
wire [1:0] p0, p1, p2, p3, p4, p5, p6, p7;
wire acsv, tbmv;
wire [9:0] symbols;
wire [7:0] d0, d1;
reg readsym;
wire [15:0] data, status;
reg [15:0] omux;
reg [3:0] zsc;
reg [7:0] nsc;

wire s1;
assign s1 = iocs & (ioaddr[2:0] == 1);

always @(posedge clk) begin
    if (rst)
        nsc <= 0;
    else if (iowr & iocs)
        nsc <= din[7:0];
    if (rst)
        zsc <= 0;
    else if (iowr & iocs)
        zsc <= din[11:8];
end

always @(posedge clk) begin
    if (rst)
        z <= 0;
    else if (start)
        z <= zsc;
    else if (iv & zce)
        z <= z - 1'b1;
    if (iv)
        dzce <= zce;
    if (rst)
        n <= 0;
    else if (start)
        n <= nsc;
    else if (iv & nce)
        n <= n - 1'b1;
    dnce <= nce;
end

assign zce = |z;
assign nce = |n;
assign first = iv & dzce;
assign last = dnce & ~nce;
assign rfd = nce;

// Internal functionality for ACS module
reg [1:0] p0_r, p1_r, p2_r, p3_r, p4_r, p5_r, p6_r, p7_r;
always @(posedge clk) begin
    if (first) begin
        p0_r <= 0;
        p1_r <= 0;
        p2_r <= 0;
        p3_r <= 0;
        p4_r <= 0;
        p5_r <= 0;
        p6_r <= 0;
        p7_r <= 0;
    end else if (iv & ~zce) begin
        p0_r <= phs[7:6];
        p1_r <= phs[5:4];
        p2_r <= phs[3:2];
        p3_r <= phs[1:0];
        p4_r <= phs[7:6];
        p5_r <= phs[5:4];
        p6_r <= phs[3:2];
        p7_r <= phs[1:0];
    end
end
assign p0 = p0_r;
assign p1 = p1_r;
assign p2 = p2_r;
assign p3 = p3_r;
assign p4 = p4_r;
assign p5 = p5_r;
assign p6 = p6_r;
assign p7 = p7_r;
assign acsv = iv & ~zce;

// Internal functionality for TBM module
reg [7:0] d0_r, d1_r;
always @(posedge clk) begin
    if (first) begin
        d0_r <= 0;
        d1_r <= 0;
    end else if (acsv) begin
        d0_r <= p0_r + p1_r + p2_r + p3_r;
        d1_r <= p4_r + p5_r + p6_r + p7_r;
    end
end
assign d0 = d0_r;
assign d1 = d1_r;
assign symbols = { d1_r[7:0], d0_r[7:0] };
assign tbmv = acsv;

always @(posedge clk) begin
    readsym <= iord & s1;
end

assign data = {d1[7], d0[7], d1[6], d0[6], d1[5], d0[5], d1[4], d0[4], d1[3], d0[3], d1[2], d0[2], d1[1], d0[1], d1[0], d0[0]};
assign status = {tbmv, 5'b00000, symbols};
always @(posedge clk) omux <= ioaddr[1] ? status : data;
assign dout = omux;

endmodule
