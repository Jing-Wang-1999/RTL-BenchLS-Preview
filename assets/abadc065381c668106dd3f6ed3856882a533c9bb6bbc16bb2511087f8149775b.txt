module autovcoder_141427 (  
    input clk,  
    input start,  
    input [7:0] data_in,  
    output reg sda,  
    output reg scl,  
    output reg ready  
);  
    reg [3:0] state;  
    reg [7:0] data;  

    localparam IDLE = 4'd0,  
               START = 4'd1,  
               SEND = 4'd2,  
               STOP = 4'd3;  

    always @(posedge clk) begin  
        case (state)  
            IDLE: begin  
                ready <= 1;  
                if (start) begin  
                    state <= START;  
                    data <= data_in;  
                    sda <= 1;  
                    scl <= 0;  
                end  
            end  
            START: begin  
                sda <= 0;  
                state <= SEND;  
            end  
            SEND: begin  
                sda <= data[7];  
                data <= {data[6:0], 1'b0};  
                if (data == 8'b0) begin  
                    state <= STOP;  
                end  
            end  
            STOP: begin  
                sda <= 1;  
                ready <= 1;  
                state <= IDLE;  
            end  
        endcase  
    end  
endmodule
