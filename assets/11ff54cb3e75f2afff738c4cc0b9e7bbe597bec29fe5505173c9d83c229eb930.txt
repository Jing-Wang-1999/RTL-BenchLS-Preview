module origen_046781 (
    input wire clock,
    input wire reset,
    input wire A,
    input wire B,
    output reg Z
);

    // State encoding
    parameter S0 = 4'b0001,
              S1 = 4'b0010,
              S2 = 4'b0100,
              S3 = 4'b1000;

    reg [3:0] fstate, next_state;

    // State transition logic
    always @(posedge clock or posedge reset) begin
        if (reset)
            fstate <= S0;
        else
            fstate <= next_state;
    end

    // Next state and output logic
    always @(*) begin
        case (fstate)
            S0: begin
                Z = 1'b0;
                next_state = A ? S1 : S2;
            end
            S1: begin
                Z = 1'b0;
                next_state = A ? S3 : S0;
            end
            S2: begin
                Z = B;
                next_state = A ? S1 : S2;
            end
            S3: begin
                Z = B;
                next_state = A ? S3 : S0;
            end
            default: begin
                Z = 1'bx;
                next_state = S0;
                $display("Error: FSM in invalid state");
            end
        endcase
    end

endmodule
