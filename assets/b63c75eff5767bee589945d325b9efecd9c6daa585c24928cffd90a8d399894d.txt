

































































































































































module vga_clkgen (
	pclk_i, rst_i, pclk_o, dvi_pclk_p_o, dvi_pclk_m_o, pclk_ena_o
);

	

	input  pclk_i;       
	input  rst_i;        

	output pclk_o;       

	output dvi_pclk_p_o; 
	output dvi_pclk_m_o; 

	output pclk_ena_o;   

	
	
	
	reg dvi_pclk_p_o;
	reg dvi_pclk_m_o;

	
	
	
	

	
	always @(posedge pclk_i)
	  if (rst_i) begin
	    dvi_pclk_p_o <=  1'b0;
	    dvi_pclk_m_o <=  1'b0;
	  end else begin
	    dvi_pclk_p_o <=  ~dvi_pclk_p_o;
	    dvi_pclk_m_o <=  dvi_pclk_p_o;
	  end





















	
	

	assign pclk_o     = pclk_i;
	assign pclk_ena_o = 1'b1;


endmodule
