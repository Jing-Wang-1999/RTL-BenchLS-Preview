always @(posedge wb_clk_i or negedge rst_i)
	  if (!rst_i)
	    cr <=  8'h0;
	  else if (wb_rst_i)
	    cr <=  8'h0;
	  else if (wb_wacc)
	    begin
	        if (core_en & (wb_adr_i == 3'b100) )
	          cr <=  wb_dat_i;
	    end
	  else
	    begin
	        if (done | i2c_al)
	          cr[7:4] <=  4'h0;           // clear command bits when done
	                                        // or when aribitration lost
	        cr[2:1] <=  2'b0;             // reserved bits
	        cr[0]   <=  1'b0;             // clear IRQ_ACK bit
	    end
