module mg_06631 #(
    parameter FB_X_MAX = 1280,
    parameter FB_Y_MAX = 1024
)(
    input wire clk,
    input wire [11:0] pixel_count,
    input wire [11:0] line_count,
    input wire fb_reset,
    input wire fb_enable,
    output reg [11:0] x,
    output reg [11:0] y
);

    always @(posedge clk) begin
        if (fb_reset) begin
            x <= 12'd0;
            y <= 12'd0;
        end else if (fb_enable) begin
            if (x == FB_X_MAX - 1) begin
                x <= 12'd0;
                if (y == FB_Y_MAX - 1) begin
                    y <= 12'd0;
                end else begin
                    y <= y + 12'd1;
                end
            end else begin
                x <= x + 12'd1;
            end
        end
    end

endmodule
