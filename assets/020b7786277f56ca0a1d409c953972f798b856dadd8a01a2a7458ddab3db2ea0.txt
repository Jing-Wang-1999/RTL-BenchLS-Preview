module origen_138496 (
    input clka,
    input wea,
    input [12:0] addra,
    input [7:0] dina,
    output reg [7:0] douta
);

reg [7:0] mem [0:8191];

integer i;
initial begin
    for (i = 0; i < 8192; i = i + 1) begin
        mem[i] = 8'b0;
    end
end

always @(posedge clka) begin
    if (wea) begin
        mem[addra] <= dina;
    end
    douta <= mem[addra];
end

endmodule
