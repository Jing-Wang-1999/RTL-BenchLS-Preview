







`timescale 1ps/1ps

module ftdi_trial #(parameter CHIP_TYPE="FTx232H", USER_EW=0)(output reg completed=0);








initial #(       100 * 1000 * 1000) begin if(u_tb_ftdi_chip_model.accepted_bytes<128) $fatal(1,"insufficient FTDI traffic"); completed=1; end






          

localparam              CHIP_EW   = (CHIP_TYPE=="FT601") ? 2 : (CHIP_TYPE=="FT600") ? 1 : 0;

wire                    ftdi_clk;
wire                    ftdi_rxf_n;
wire                    ftdi_txe_n;
wire                    ftdi_oe_n;
wire                    ftdi_rd_n;
wire                    ftdi_wr_n;
tri  [(8<<CHIP_EW)-1:0] ftdi_data;
tri  [(1<<CHIP_EW)-1:0] ftdi_be;








wire                    tready;
wire                    tvalid;
wire [(8<<USER_EW)-1:0] tdata;
wire [(1<<USER_EW)-1:0] tkeep;
wire                    tlast;






reg rstn = 1'b0;
reg clk  = 1'b0;
always #5000 clk = ~clk;           
initial begin repeat(4) @(posedge clk); rstn<=1'b1; end






tb_ftdi_chip_model # (
    .CHIP_EW        ( CHIP_EW        )
) u_tb_ftdi_chip_model (
    .ftdi_clk       ( ftdi_clk       ),
    .ftdi_rxf_n     ( ftdi_rxf_n     ),
    .ftdi_txe_n     ( ftdi_txe_n     ),
    .ftdi_oe_n      ( ftdi_oe_n      ),
    .ftdi_rd_n      ( ftdi_rd_n      ),
    .ftdi_wr_n      ( ftdi_wr_n      ),
    .ftdi_data      ( ftdi_data      ),
    .ftdi_be        ( ftdi_be        )
);






ftdi_245fifo_top #(
    .TX_EW          ( USER_EW        ),
    .TX_EA          ( 10             ),
    .RX_EW          ( USER_EW        ),
    .RX_EA          ( 10             ),
    .CHIP_TYPE      ( CHIP_TYPE      ),
    .SIMULATION     ( 0              )
) u_ftdi_245fifo_top (
    .rstn_async     ( rstn           ),
    
    .tx_clk         ( clk            ),
    .tx_tready      ( tready         ),
    .tx_tvalid      ( tvalid         ),
    .tx_tdata       ( tdata          ),
    .tx_tkeep       ( tkeep          ),
    .tx_tlast       ( tlast          ),
    
    .rx_clk         ( clk            ),
    .rx_tready      ( tready         ),
    .rx_tvalid      ( tvalid         ),
    .rx_tdata       ( tdata          ),
    .rx_tkeep       ( tkeep          ),
    .rx_tlast       ( tlast          ),
    
    .ftdi_clk       ( ftdi_clk       ),
    .ftdi_rxf_n     ( ftdi_rxf_n     ),
    .ftdi_txe_n     ( ftdi_txe_n     ),
    .ftdi_oe_n      ( ftdi_oe_n      ),
    .ftdi_rd_n      ( ftdi_rd_n      ),
    .ftdi_wr_n      ( ftdi_wr_n      ),
    .ftdi_data      ( ftdi_data      ),
    .ftdi_be        ( ftdi_be        )
);



integer rx_bytes=0, byte_index;
always @(posedge clk) if(rstn && tvalid && tready) begin
 for(byte_index=0;byte_index<(1<<USER_EW);byte_index=byte_index+1)
  if(tkeep[byte_index])begin
   if(tdata[8*byte_index+:8] !== (rx_bytes & 8'hff)) $fatal(1,"FTDI receive stream mismatch");
   rx_bytes=rx_bytes+1;
  end
end
endmodule









`timescale 1ps/1ps

module tb_ftdi_chip_model #(
    parameter                CHIP_EW = 0        
) (
    output reg               ftdi_clk,
    output reg               ftdi_rxf_n,
    output reg               ftdi_txe_n,
    input  wire              ftdi_oe_n,
    input  wire              ftdi_rd_n,
    input  wire              ftdi_wr_n,
    inout [(8<<CHIP_EW)-1:0] ftdi_data,
    inout [(1<<CHIP_EW)-1:0] ftdi_be
);






function  [31:0] randuint;
    input [31:0] min;
    input [31:0] max;
begin
    randuint = $random;
    if ( min != 0 || max != 'hFFFFFFFF )
        randuint = (randuint % (1+max-min)) + min;
end
endfunction




wire       [(8<<CHIP_EW)-1:0] DATA_HIGHZ = {(8<<CHIP_EW){1'bz}};
localparam [(8<<CHIP_EW)-1:0] DATA_ZERO  = {(8<<CHIP_EW){1'b0}};
wire       [(1<<CHIP_EW)-1:0] BE_HIGHZ   = {(1<<CHIP_EW){1'bz}};
localparam [(1<<CHIP_EW)-1:0] BE_ZERO    = {(1<<CHIP_EW){1'b0}};
localparam [(1<<CHIP_EW)-1:0] BE_ALL_ONE = ~BE_ZERO;



initial ftdi_clk = 1'b0;                    
always #8333 ftdi_clk = ~ftdi_clk;          




reg [(8<<CHIP_EW)-1:0] ftdi_r_data = DATA_ZERO;
reg [(1<<CHIP_EW)-1:0] ftdi_r_be   = BE_ZERO;

reg [(8<<CHIP_EW)-1:0] tmp_data = DATA_ZERO;
reg [(8<<CHIP_EW)-1:0] tmp_be   = (CHIP_EW==0) ? 1     : BE_ZERO;
reg              [7:0] rxbyte   = (CHIP_EW==0) ? 8'h01 : 8'h00;

integer i;


always @ (posedge ftdi_clk)                    
    if (~ftdi_rd_n & ~ftdi_rxf_n) begin
        tmp_data = ftdi_r_data;
        tmp_be   = (CHIP_EW==0) ? BE_ALL_ONE : randuint(0, 'hFFFFFFFF);
    
        for (i=0; i<(1<<CHIP_EW); i=i+1) begin
            if (tmp_be[i]) begin
                
                tmp_data[8*i +: 8] = rxbyte;
                rxbyte = rxbyte + 1;
            end
        end
        
        ftdi_r_data <= tmp_data;
        ftdi_r_be   <= tmp_be;
    end




integer accepted_bytes=0; reg [7:0] txbyte=0;

always @ (posedge ftdi_clk)                    
    if (~ftdi_wr_n & ~ftdi_txe_n) begin
        for (i=0; i<(1<<CHIP_EW); i=i+1) begin
            if (ftdi_be[i]) begin
                
                if (txbyte !== ftdi_data[8*i +: 8]) begin $fatal(1,"FTDI byte-order mismatch"); end
                txbyte = txbyte + 1; accepted_bytes=accepted_bytes+1;
            end
        end
    end


assign  ftdi_data = ftdi_oe_n ? DATA_HIGHZ : ftdi_r_data;
assign  ftdi_be   = ftdi_oe_n ? BE_HIGHZ   : ftdi_r_be;




initial begin
    ftdi_rxf_n <= 1'b1;
    while (1) begin
        repeat (randuint(1, 100)) @ (posedge ftdi_clk);
        ftdi_rxf_n <= 1'b0;
        repeat (randuint(1, 100)) @ (posedge ftdi_clk);
        ftdi_rxf_n <= 1'b1;
    end
end


initial begin
    ftdi_txe_n <= 1'b1;
    while (1) begin
        repeat (randuint(1, 100)) @ (posedge ftdi_clk);
        ftdi_txe_n <= 1'b0;
        repeat (randuint(1, 100)) @ (posedge ftdi_clk);
        ftdi_txe_n <= 1'b1;
    end
end


endmodule


module tb;
wire [1:0] done;
ftdi_trial #(.CHIP_TYPE("FTx232H"),.USER_EW(2)) t0(done[0]);
ftdi_trial #(.CHIP_TYPE("FTx232H"),.USER_EW(0)) t1(done[1]);
initial begin wait(&done);$display("RTL_BENCHLS_PASS");$finish(0);end
initial begin #110000000;$fatal(1,"FTDI watchdog");end
endmodule
