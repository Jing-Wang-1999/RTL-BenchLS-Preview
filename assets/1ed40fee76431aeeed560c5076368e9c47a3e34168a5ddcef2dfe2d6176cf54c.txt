







































































`timescale 1ns/1ps





module generic_spram(
	
	clk, rst, ce, we, oe, addr, di, \do 
);

	
	
	
	parameter aw = 6; 
	parameter dw = 8; 

	
	
	
	input           clk;  
	input           rst;  
	input           ce;   
	input           we;   
	input           oe;   
	input  [aw-1:0] addr; 
	input  [dw-1:0] di;   
	output [dw-1:0] \do ;   

	
	
	


	
	
	
	
	
	

	
	
	

	reg [dw-1:0] mem [(1<<aw) -1:0] ;
	reg [aw-1:0] ra;

	
	always @(posedge clk)
	  if (ce)
	    ra <=  addr;     

	assign \do  = mem[ra];

	
	always @(posedge clk)
	  if (we && ce)
	    mem[addr] <=  di;




































































































































































 

endmodule
