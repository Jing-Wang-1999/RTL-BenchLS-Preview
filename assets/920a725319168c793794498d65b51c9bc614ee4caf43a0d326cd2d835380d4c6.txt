def trial(index, width, address, whalf, rhalf, phase, registered=False, boundary=False):
    boundary_code="""// Fill and drain repeatedly, checking flags and rejected full/empty accesses.
for(epoch=0;epoch<4;epoch=epoch+1)begin
for(i=0;i<DEPTH;i=i+1)begin
@(negedge wclk);winc=1;wdata=(epoch*57+i*19+3);wt;
if(wfull!==(i==DEPTH-1) || awfull!==(i==DEPTH-2))$fatal(1,"fill flags %d",i);
end
@(negedge wclk);wdata='1;repeat(3)wt;winc=0;
if(writes!=(epoch+1)*DEPTH)$fatal(1,"full write accepted");settle;
if(rempty!==0)$fatal(1,"write visibility");
for(i=0;i<DEPTH;i=i+1)begin
@(negedge rclk);rinc=1;rt;
if(rempty!==(i==DEPTH-1) || arempty!==(i==DEPTH-2))$fatal(1,"drain flags %d",i);
end
repeat(3)rt;@(negedge rclk);rinc=0;settle;
if(reads!=writes || wfull!==0)$fatal(1,"drain totals");
end
// Reset a nonempty FIFO and ensure old contents are discarded.
@(negedge wclk);winc=1;wdata='h5a;repeat(DEPTH/2)wt;
@(negedge wclk);winc=0;settle;reset_both;
"""
    depth=1<<address
    return f'''`timescale 1ns/1ps
module trial_{index}(output reg completed=0);
localparam W={width}, A={address}, DEPTH={depth};
reg wclk=0,rclk=0,wrst_n=1,rrst_n=1,winc=0,rinc=0;
reg [W-1:0] wdata=0;
wire [W-1:0] rdata;wire wfull,awfull,rempty,arempty;
always #{whalf} wclk=~wclk;
initial begin #{phase};forever #{rhalf} rclk=~rclk;end
async_fifo #(.DSIZE(W),.ASIZE(A),.FALLTHROUGH("{'FALSE' if registered else 'TRUE'}")) dut
(.wclk(wclk),.wrst_n(wrst_n),.winc(winc),.wdata(wdata),.wfull(wfull),.awfull(awfull),
.rclk(rclk),.rrst_n(rrst_n),.rinc(rinc),.rdata(rdata),.rempty(rempty),.arempty(arempty));
reg [W-1:0] expected[0:8191];integer writes=0,reads=0,wcycle=0,rcycle=0;
reg active=0,automatic_traffic=0,drain=0;reg [31:0] rng=32'h2bc9137;
always @(posedge wclk)if(active && wrst_n && winc && !wfull)begin
expected[writes]=wdata;writes=writes+1;end
always @(posedge rclk)if(active && rrst_n && rinc && !rempty)begin
{'#1;' if registered else ''}
if(reads>=writes || rdata!==expected[reads])$fatal(1,"data order trial {index} read=%0d got=%h expected=%h",reads,rdata,expected[reads]);
reads=reads+1;end
always @(negedge wclk)if(automatic_traffic)begin
wcycle=wcycle+1;rng=(rng>>1)^((rng&1)?32'hd0000001:0);
winc=!drain && ((wcycle%7)!=0) && rng[0];
wdata=(rng ^ (writes*32'h9e3779b9));end
always @(negedge rclk)if(automatic_traffic)begin
rcycle=rcycle+1;rinc=drain || ((rcycle%5)!=0);end
task wt;begin @(posedge wclk);#2;end endtask
task rt;begin @(posedge rclk);#2;end endtask
task settle;begin repeat(5)wt;repeat(5)rt;end endtask
task reset_both;begin
active=0;automatic_traffic=0;winc=0;rinc=0;wrst_n=0;rrst_n=0;
settle;if(wfull!==0 || awfull!==0 || rempty!==1 || arempty!==0)$fatal(1,"reset flags");
@(negedge wclk);wrst_n=1;rrst_n=1;writes=0;reads=0;settle;active=1;end endtask
integer i,epoch,target;
initial begin
reset_both;
{boundary_code if boundary else ''}
automatic_traffic=1;repeat(1600)wt;
@(negedge wclk);drain=1;winc=0;target=writes;
repeat(DEPTH*8+30)rt;
@(negedge rclk);automatic_traffic=0;rinc=0;settle;
if(writes!=target || reads!=writes || reads<50 || rempty!==1 || wfull!==0)$fatal(1,"traffic incomplete %d %d",writes,reads);
completed=1;end
initial begin #5000000;$fatal(1,"trial timeout {index}");end
endmodule
'''


def test(configs, **kwargs):
    s=''.join(trial(i,*cfg,**kwargs) for i,cfg in enumerate(configs))
    s+=f'module tb;wire [{len(configs)-1}:0] completed;\n'
    s+=''.join(f'trial_{i} dut_{i}(completed[{i}]);\n' for i in range(len(configs)))
    return s+'initial begin wait(&completed);$display("RTL_BENCHLS_PASS");$finish(0);end endmodule\n'


INSTRUCTION='''# Implement the asynchronous FIFO project
Read the original project specification, README and testplan in specification/.
Implement the complete six-module async_fifo configuration, with the required
paths and declaration-only interfaces. Bidirectional and external RAM wrappers
are separate configurations and are not required by this case.

Reviewed document reconciliation, fixed before any evaluated model call:
- The old specification.rst uses wren/rden and a byte-depth rounding interface.
  The pinned release README and RTL interface instead use winc/rinc, DSIZE bits
  per word, and ASIZE address bits: capacity is exactly 2**ASIZE words. No byte
  rounding parameter exists. Follow this release contract when documents differ.
- Both clocks use rising edges. wrst_n and rrst_n are asynchronous active-low
  resets; assert both together before use, including when discarding queued data.
  Memory contents and read output on empty/reset are unspecified. Initial flags
  are wfull=0, awfull=0, rempty=1 and arempty=0.
- A write edge accepts wdata iff winc && !wfull. A read edge consumes the oldest
  word iff rinc && !rempty. Requests during full/empty do not advance pointers;
  writes while full must not overwrite memory. Preserve order through pointer wrap.
- Maintain local binary pointers with an extra wrap bit and Gray-code pointers
  for cross-domain transfer. The declared synchronizers transfer each pointer
  through TWO destination-clock registers, reset to zero. The source may advance
  independently; local availability flags can conservatively lag remote progress.
- With FALLTHROUGH="TRUE", rdata continuously shows the oldest addressed word;
  sample it before an accepted rising read edge. With any other value, rdata
  registers the addressed word on each rising edge with rinc; an accepted read's
  result is visible after that edge. No data correctness is required on empty.
- wfull is registered from whether the next accepted-write pointer equals the
  synchronized read pointer with the wrap relation for full. awfull registers
  the same test one additional word ahead; it is an equality flag for one free
  slot, not an inclusive threshold asserted at full. rempty registers equality of
  the next accepted-read Gray pointer and synchronized write pointer; arempty
  uses one extra read ahead, indicating exactly one available word, not empty.
- Required parameter envelope: DSIZE 8, 13 or 32; ASIZE 2, 3 or 4; both read
  modes. Preserve default parameter values in the interfaces. Internal state
  names are unrestricted; connect all six modules in the integration top.

Private points test reset, full/empty blocking, almost flags, multiple wraps,
independent clock rates/phases, intermittent and simultaneous accesses, parameter
geometry and registered reads. Tests compare end-to-end transactions with an
independent software queue; no hierarchical access to reference internals is used.
Finite RTL simulation does not establish analog metastability safety or physical
CDC timing closure. Write your own tests; public feedback checks compilation.
'''


def async_fifo_spec():
    files=['rtl/'+n+'.v' for n in ['async_fifo','fifomem','rptr_empty','sync_r2w','sync_w2r','wptr_full']]
    return dict(case_id='github_dpretet_async_fifo_v1',project='dpretet_async_fifo',top='async_fifo',
        configuration_record={'configuration_id':'async_fifo','top_module':'async_fifo','source_closure':files,
          'specification_paths':['doc/specification.rst','README.md','doc/testplan.rst']},
        instruction=INSTRUCTION,builder_source=__file__,
        tests=[('boundary_reset','async_fifo',test([(8,3,5,7,2)],boundary=True)),
          ('write_faster','async_fifo',test([(8,4,3,11,2),(13,3,5,7,1)])),
          ('read_faster','async_fifo',test([(8,3,11,3,2),(13,4,7,5,1)])),
          ('equal_rate_phases','async_fifo',test([(8,4,5,5,0),(13,3,5,5,3)])),
          ('parameter_geometry','async_fifo',test([(32,2,3,7,2),(13,2,7,3,1),(32,4,5,7,3)])),
          ('registered_read','async_fifo',test([(8,3,5,7,2),(32,4,7,3,1)],registered=True))],
        mutations=[('memory_data','rtl/fifomem.v','mem[waddr] <= wdata','mem[waddr] <= (wdata ^ 1)'),
          ('write_increment','rtl/wptr_full.v','winc & ~wfull','winc & wfull'),
          ('read_increment','rtl/rptr_empty.v','rinc & ~rempty','rinc & rempty'),
          ('write_visibility','rtl/sync_w2r.v','{rq1_wptr,wptr}','{rq1_wptr,{(ASIZE+1){1\'b0}}}'),
          ('read_visibility','rtl/sync_r2w.v','{wq1_rptr,rptr}','{wq1_rptr,{(ASIZE+1){1\'b0}}}'),
          ('almost_full','rtl/wptr_full.v','awfull <= awfull_val','awfull <= ~awfull_val')])


if __name__=='__main__':
    parser=argparse.ArgumentParser();parser.add_argument('--qualify-and-freeze',action='store_true');args=parser.parse_args()
