always @(negedge ECLK)
  begin
    if (rst_i)
    begin
      icr_m <= 5'b0;
      icr_rst <= 1'b0;
    end
    else if (sel_i & adr_icr)
    begin
      icr_rst <= ~we_i;

      if (we_i)
      begin
        if (DBi[7])
          icr_m <= icr_m |  DBi[4:0];
        else
          icr_m <= icr_m & ~DBi[4:0];
      end
    end
    else
    begin
      icr_rst <= 1'b0;
    end
  end
