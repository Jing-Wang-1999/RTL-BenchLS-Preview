#!/usr/bin/env python3
import argparse,json,shutil,subprocess,os,time,xml.etree.ElementTree as ET
from pathlib import Path
BASE=Path('/home/jwangjw/project_3rd/ICLR-RTL-BenchLS/scope2/task2_3_specification_round_trip/results/qualification_172_20260909_v1')
VENV=BASE/'environment';CONVERT=BASE/'next21_v1/environment/sv2v/sv2v-Linux';TOOLS=Path('/home/jwangjw/anaconda3/envs/securitylab/bin')
def main():
 p=argparse.ArgumentParser()
 for k in ['payload','candidate','output']:p.add_argument('--'+k,type=Path,required=True)
 a=p.parse_args();payload,candidate,out=a.payload.resolve(),a.candidate.resolve(),a.output.resolve();out.mkdir(parents=True,exist_ok=False);meta=json.load(open(payload/'metadata.json'));stage=out/'source';shutil.copytree(payload/'source',stage)
 for rel in meta['rtl_files']:shutil.copy2(candidate/rel,stage/rel)
 (stage/'build').mkdir(exist_ok=True);(stage/'test/logs').mkdir(exist_ok=True);env=dict(os.environ);env['PATH']=str(CONVERT)+':'+str(VENV/'bin')+':'+str(TOOLS)+':/usr/bin:/bin';env['PYTHONPATH']=str(stage);env['VIRTUAL_ENV']=str(VENV);env['RANDOM_SEED']='123456';points=[]
 for name in ['matadd','matmul']:
  log=out/(name+'.log');xml=out/(name+'.xml');env['COCOTB_RESULTS_FILE']=str(xml);t=time.time()
  with log.open('w') as f:
   try:r=subprocess.run(['make','test_'+name],cwd=stage,env=env,stdout=f,stderr=subprocess.STDOUT,timeout=180);rc=r.returncode
   except subprocess.TimeoutExpired:rc='timeout'
  text=log.read_text();tests=[]
  if xml.exists():tests=ET.parse(xml).getroot().findall('.//testcase')
  passed=rc==0 and len(tests)==1 and not any(x.find('failure') is not None or x.find('error') is not None or x.find('skipped') is not None for x in tests)
  points.append({'id':name,'passed':passed,'compile_valid':(stage/'build/sim.vvp').exists() and 'Running tests with cocotb' in text,'returncode':rc,'seconds':time.time()-t})
 result={'case_id':meta['case_id'],'points':points,'score':sum(x['passed'] for x in points)/2,'passed_testbenches':sum(x['passed'] for x in points),'total_testbenches':2};(out/'result.json').write_text(json.dumps(result,indent=2)+'\n');print(json.dumps(result))
if __name__=='__main__':main()
