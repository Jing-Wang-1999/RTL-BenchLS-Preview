





























module substitution(data_o,data_i); 
                                    



output wire [63:0] data_o;
input  wire [63:0] data_i;



genvar j;



generate
    for (j = 0; j < 16; j = j+1)
    begin : boxes
        sbox substitution_sbox (.data_o(data_o[j*4+3 : j*4]),.data_i(data_i[j*4+3 : j*4]));    
    end
endgenerate


endmodule
