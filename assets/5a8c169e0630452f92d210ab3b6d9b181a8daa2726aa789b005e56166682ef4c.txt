module origen_147520 #(
    parameter stimDelay = 1
)(
    output reg clk,
    output reg rst_n,
    input [3:0] out
);

    // Clock generation
    initial begin
        clk = 0;
        forever #(32*stimDelay) clk = ~clk;
    end

    // Reset generation
    initial begin
        rst_n = 0;
        #20 rst_n = 1;
        #(32*stimDelay) rst_n = 0;
    end

    // Output monitoring
    initial begin
        $monitor("Time %t: out = %b", $time, out);
    end

    // Detect changes in out
    always @(out) begin
        $display("Time %t: out changed to %b", $time, out);
    end

endmodule

// Testbench
module tb_clock_reset_generator();
    wire clk;
    wire rst_n;
    reg [3:0] out;

    // Instantiate the clock_reset_generator
    clock_reset_generator #(.stimDelay(1)) uut (
        .clk(clk),
        .rst_n(rst_n),
        .out(out)
    );

    // Stimulus
    initial begin
        out = 4'b0000;
        #10 out = 4'b0001;
        #30 out = 4'b0010;
        #50 out = 4'b0100;
        #70 out = 4'b1000;
        #90 out = 4'b1111;
        #110 $finish;
    end

endmodule
