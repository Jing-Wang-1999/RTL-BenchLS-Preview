module origen_054641 (
    input wire clk,
    input wire reset,
    input wire [10:0] posEntX,
    input wire [10:0] posEntY,
    input wire push,
    input wire pop,
    input wire obtener,
    output reg [10:0] posColaX,
    output reg [10:0] posColaY
);

    reg [5:0] ptr;
    reg [10:0] stackX [25:0];
    reg [10:0] stackY [25:0];
    integer i, j;

    initial begin
        ptr = 0;
        for (i = 0; i < 26; i = i + 1) begin
            stackX[i] = 0;
            stackY[i] = 0;
        end
    end

    always @(posedge clk or posedge reset) begin
        if (reset) begin
            ptr <= 0;
            for (i = 0; i < 26; i = i + 1) begin
                stackX[i] <= 0;
                stackY[i] <= 0;
            end
        end else begin
            if (push) begin
                stackX[ptr] <= posEntX;
                stackY[ptr] <= posEntY;
                ptr <= (ptr == 25) ? 0 : ptr + 1;
            end else if (pop) begin
                for (i = 0; i < 25; i = i + 1) begin
                    stackX[i] <= stackX[i+1];
                    stackY[i] <= stackY[i+1];
                end
                stackX[25] <= posEntX;
                stackY[25] <= posEntY;
            end else if (obtener) begin
                posColaX <= stackX[j];
                posColaY <= stackY[j];
                j <= (j == 25) ? 0 : j + 1;
            end
        end
    end

endmodule
