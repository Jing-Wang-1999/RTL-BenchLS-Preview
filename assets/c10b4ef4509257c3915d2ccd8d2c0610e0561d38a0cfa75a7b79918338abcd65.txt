module part3 (
    input [17:0] SW,
    input [2:0] KEY,
    output [17:0] LEDR,
    output [2:0] LEDG
);
    assign LEDR = SW;
    
    assign LEDG = (KEY == 3'b000) ? SW[17:15] :
                  (KEY == 3'b001) ? SW[14:12] :
                  (KEY == 3'b010) ? SW[11:9]  :
                  (KEY == 3'b011) ? SW[8:6]   :
                  (KEY == 3'b100) ? SW[5:3]   :
                  (KEY == 3'b101) ? SW[2:0]   :
                  3'b000;
endmodule
