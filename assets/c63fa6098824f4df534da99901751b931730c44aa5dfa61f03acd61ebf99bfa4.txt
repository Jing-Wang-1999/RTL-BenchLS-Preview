module rtlpp_06416 (
   RsTx, led,
   RsRx, sw, btnS, btnR, btnSend, clk
);

`define seq_dp_width 8
`define seq_in_width 8

input        RsRx;
output       RsTx;

input  [7:0] sw;
output [7:0] led;
input        btnS;                 
input        btnR;                 
input 		btnSend;			  
input        clk;                  

wire [7:0] seq_tx_data;         
wire                 seq_tx_valid;           
wire [7:0]           uart_rx_data;           
wire                 uart_rx_valid;          
wire                 uart_tx_busy;           

wire        rst;
wire        arst_i;
wire [17:0] clk_dv_inc;

reg [1:0]   arst_ff;
reg [16:0]  clk_dv;
reg         clk_en;
reg         clk_en_d;
      
reg [7:0]   inst_wd;
reg         inst_vld;
reg [2:0]   step_d;
   
wire [1:0]  sentRegister;
reg         inst_vld_send;
reg [2:0]   step_d_send;

reg [7:0]   inst_cnt;

assign arst_i = btnR;
assign rst = arst_ff[0];

always @ (posedge clk or posedge arst_i)
   if (arst_i)
     arst_ff <= 2'b11;
   else
     arst_ff <= {1'b0, arst_ff[1]};

assign clk_dv_inc = clk_dv + 1;

always @ (posedge clk)
   if (rst)
     begin
        clk_dv   <= 0;
        clk_en   <= 1'b0;
        clk_en_d <= 1'b0;
     end
   else
     begin
        clk_dv   <= clk_dv_inc[16:0];
        clk_en   <= clk_dv_inc[17];
        clk_en_d <= clk_en;
     end

always @ (posedge clk)
   if (rst)
     begin
        inst_wd[7:0]       <= 0;
        step_d[2:0]        <= 0;
        step_d_send[2:0]   <= 0;
     end
   else if (clk_en) 
     begin
        inst_wd[7:0]       <= sw[7:0];
        step_d[2:0]        <= {btnS, step_d[2:1]};
        step_d_send[2:0]   <= {btnSend, step_d_send[2:1]};
     end
    
wire is_btnS_posedge;
wire is_btnSend_posedge;
assign is_btnS_posedge = ~step_d[0] & step_d[1];
assign is_btnSend_posedge = ~step_d_send[0] & step_d_send[1];
always @ (posedge clk)
   if (rst) 
     begin
       inst_vld <= 1'b0;
       inst_vld_send <= 1'b0;
     end
   else if (clk_en_d) 
     begin
       inst_vld <= is_btnS_posedge;
       inst_vld_send <= is_btnSend_posedge;
     end
   else 
     begin
       inst_vld <= 0;
       inst_vld_send <= 0;
     end

always @ (posedge clk)
   if (rst)
     inst_cnt <= 0;
   else if (inst_vld)
     inst_cnt <= inst_cnt + 1;

assign led[7:0] = inst_cnt[7:0];

assign sentRegister = sw[5:4];

reg [7:0] o_tx_data;
reg      o_tx_valid;
reg      i_tx_busy;
reg [7:0] i_inst;
reg      i_inst_valid;
reg      i_inst_valid_send;
reg      clk;
reg      rst;

always @(posedge clk or posedge rst) begin
   if (rst) begin
       o_tx_data     <= 0;
       o_tx_valid    <= 0;
       i_tx_busy     <= 0;
       i_inst        <= 0;
       i_inst_valid  <= 0;
       i_inst_valid_send <= 0;
   end
   else begin
       if (i_inst_valid) begin
           o_tx_valid <= 1;
           o_tx_data  <= i_inst;
       end
       else begin
           o_tx_valid <= 0;
       end
   end
end

reg [7:0] o_uart_tx;
reg       o_uart_tx_busy;
reg [7:0] o_rx_data;
reg       o_rx_valid;
reg       i_rx;
reg [7:0] i_tx_data;
reg       i_tx_stb;
reg [1:0] i_tx_reg;
reg       clk;
reg       rst;

always @(posedge clk or posedge rst) begin
   if (rst) begin
       o_uart_tx      <= 0;
       o_uart_tx_busy <= 0;
       o_rx_data      <= 0;
       o_rx_valid     <= 0;
   end
   else begin
       if (i_tx_stb) begin
           o_uart_tx <= i_tx_data;
           o_uart_tx_busy <= 1;
       end
       else begin
           o_uart_tx_busy <= 0;
       end
   end
end

assign RsTx = o_uart_tx;
assign uart_tx_busy = o_uart_tx_busy;
assign uart_rx_data = o_rx_data;
assign uart_rx_valid = o_rx_valid;

endmodule
