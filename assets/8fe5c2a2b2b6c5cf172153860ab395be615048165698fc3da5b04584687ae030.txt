`timescale 1ns/1ps
module tb;reg clk=0,rst=0;always #5 clk=~clk;
task tick;begin @(posedge clk);#2;end endtask
task reset;begin @(negedge clk);rst=1;repeat(3)tick;@(negedge clk);rst=0;end endtask
initial begin #100000000;$fatal(1,"timeout");end
reg cyc=0,stb=0,we=0;reg [7:0] adr=0;reg [31:0] di=0;reg [3:0] sel=15;
wire [31:0] data;wire ack,err,irq;reg [30:0] aux=0,pins=0;reg eclk=0;wire [30:0] outputs,oe;
gpio_top dut(clk,rst,cyc,adr,di,sel,we,stb,data,ack,err,irq,aux,pins,outputs,oe,eclk);
task wr;input [7:0] a;input [31:0] d;input [3:0] mask;begin
@(negedge clk);adr=a;di=d;sel=mask;cyc=1;stb=1;we=1;tick;
if(ack!==1 || err!==0)$fatal(1,"write ack");
@(negedge clk);cyc=0;stb=0;we=0;repeat(3)tick;end endtask
task rd;input [7:0] a;input [31:0] expected;begin
@(negedge clk);adr=a;sel=15;cyc=1;stb=1;we=0;tick;
if(ack!==1 || err!==0 || data!==expected)$fatal(1,"read %h: %h != %h",a,data,expected);
@(negedge clk);cyc=0;stb=0;repeat(3)tick;end endtask
initial begin reset;repeat(5)tick;
@(negedge clk);adr=1;cyc=1;stb=1;we=0;tick;if(err!==1 || ack!==0)$fatal(1,"address error");@(negedge clk);cyc=0;stb=0;repeat(3)tick;
@(negedge clk);adr=2;cyc=1;stb=1;we=0;tick;if(err!==1 || ack!==0)$fatal(1,"address error");@(negedge clk);cyc=0;stb=0;repeat(3)tick;
@(negedge clk);adr=3;cyc=1;stb=1;we=0;tick;if(err!==1 || ack!==0)$fatal(1,"address error");@(negedge clk);cyc=0;stb=0;repeat(3)tick;
@(negedge clk);adr=64;cyc=1;stb=1;we=0;tick;if(err!==1 || ack!==0)$fatal(1,"address error");@(negedge clk);cyc=0;stb=0;repeat(3)tick;
@(negedge clk);adr=128;cyc=1;stb=1;we=0;tick;if(err!==1 || ack!==0)$fatal(1,"address error");@(negedge clk);cyc=0;stb=0;repeat(3)tick;
@(negedge clk);adr=196;cyc=1;stb=1;we=0;tick;if(err!==1 || ack!==0)$fatal(1,"address error");@(negedge clk);cyc=0;stb=0;repeat(3)tick;
@(negedge clk);adr=0;cyc=1;stb=1;repeat(4)begin tick;if(ack!==1)$fatal(1,"ack high");tick;if(ack!==0)$fatal(1,"ack low");end
$display("RTL_BENCHLS_PASS");$finish(0);end endmodule
