module gpio_top(
	// WISHBONE Interface
	wb_clk_i, wb_rst_i, wb_cyc_i, wb_adr_i, wb_dat_i, wb_sel_i, wb_we_i, wb_stb_i,
	wb_dat_o, wb_ack_o, wb_err_o, wb_inta_o,


	// Auxiliary inputs interface
	aux_i,
 //  GPIO_AUX_IMPLEMENT

	// External GPIO Interface
	ext_pad_i, ext_pad_o, ext_padoe_o

  , clk_pad_i 

);

parameter dw = 32;
parameter aw = 7+1;
parameter gw = 31;
//
// WISHBONE Interface
//
input             wb_clk_i;	// Clock
input             wb_rst_i;	// Reset
input             wb_cyc_i;	// cycle valid input
input   [aw-1:0]	wb_adr_i;	// address bus inputs
input   [dw-1:0]	wb_dat_i;	// input data bus
input	  [3:0]     wb_sel_i;	// byte select inputs
input             wb_we_i;	// indicates write transfer
input             wb_stb_i;	// strobe input
output  [dw-1:0]  wb_dat_o;	// output data bus
output            wb_ack_o;	// normal termination
output            wb_err_o;	// termination w/ error
output            wb_inta_o;	// Interrupt request output


// Auxiliary Inputs Interface
input	  [gw-1:0]  aux_i;		// Auxiliary inputs
 // GPIO_AUX_IMPLEMENT

//
// External GPIO Interface
//
input   [gw-1:0]  ext_pad_i;	// GPIO Inputs

input             clk_pad_i;	// GPIO Eclk
 //  GPIO_CLKPAD
output  [gw-1:0]  ext_pad_o;	// GPIO Outputs
output  [gw-1:0]  ext_padoe_o;	// GPIO output drivers enables



//
// GPIO Input Register (or no register)
//

reg	[gw-1:0]	rgpio_in;	// RGPIO_IN register




//
// GPIO Output Register (or no register)
//

reg	[gw-1:0]	rgpio_out;	// RGPIO_OUT register




//
// GPIO Output Driver Enable Register (or no register)
//

reg	[gw-1:0]	rgpio_oe;	// RGPIO_OE register




//
// GPIO Interrupt Enable Register (or no register)
//

reg	[gw-1:0]	rgpio_inte;	// RGPIO_INTE register




//
// GPIO Positive edge Triggered Register (or no register)
//

reg	[gw-1:0]	rgpio_ptrig;	// RGPIO_PTRIG register




//
// GPIO Auxiliary select Register (or no register)
//

reg	[gw-1:0]	rgpio_aux;	// RGPIO_AUX register




//
// GPIO Control Register (or no register)
//

reg	[1:0]		rgpio_ctrl;	// RGPIO_CTRL register




//
// GPIO Interrupt Status Register (or no register)
//

reg	[gw-1:0]	rgpio_ints;	// RGPIO_INTS register




//
// GPIO Enable Clock  Register (or no register)
//

reg	[gw-1:0]	rgpio_eclk;	// RGPIO_ECLK register




//
// GPIO Active Negative Edge  Register (or no register)
//

reg	[gw-1:0]	rgpio_nec;	// RGPIO_NEC register





//
// Synchronization flops for input signals
//
 
reg  [gw-1:0]  sync      ,
               ext_pad_s ;






//
// Internal wires & regs
//
wire            rgpio_out_sel;  // RGPIO_OUT select
wire            rgpio_oe_sel; // RGPIO_OE select
wire            rgpio_inte_sel; // RGPIO_INTE select
wire            rgpio_ptrig_sel;// RGPIO_PTRIG select
wire            rgpio_aux_sel;  // RGPIO_AUX select
wire            rgpio_ctrl_sel; // RGPIO_CTRL select
wire            rgpio_ints_sel; // RGPIO_INTS select
wire            rgpio_eclk_sel ;
wire            rgpio_nec_sel ;
wire            full_decoding;  // Full address decoding qualification
wire  [gw-1:0]  in_muxed; // Muxed inputs
wire            wb_ack;   // WB Acknowledge
wire            wb_err;   // WB Error
wire            wb_inta;  // WB Interrupt
reg   [dw-1:0]  wb_dat;   // WB Data out

reg             wb_ack_o; // WB Acknowledge
reg             wb_err_o; // WB Error
reg             wb_inta_o;  // WB Interrupt
reg   [dw-1:0]  wb_dat_o; // WB Data out

wire  [gw-1:0]  out_pad;  // GPIO Outputs

reg   [gw-1:0]  ext_pad_o;  // GPIO Outputs


wire  [gw-1:0]  extc_in;  // Muxed inputs sampled by external clock
wire  [gw-1:0]  pext_clk; // External clock for posedge flops
reg   [gw-1:0]  pextc_sampled;  // Posedge external clock sampled inputs






reg   [gw-1:0]  nextc_sampled;  // Negedge external clock sampled inputs

 //  GPIO_CLKPAD


//
// All WISHBONE transfer terminations are successful except when:
// a) full address decoding is enabled and address doesn't match
//    any of the GPIO registers
// b) wb_sel_i evaluation is enabled and one of the wb_sel_i inputs is zero
//

//
// WB Acknowledge
//
assign wb_ack = wb_cyc_i & wb_stb_i & !wb_err_o;

//
// Optional registration of WB Ack
//

always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		wb_ack_o <=  1'b0;
	else
		wb_ack_o <=  wb_ack & ~wb_ack_o & (!wb_err) ;




//
// WB Error
//




assign wb_err = wb_cyc_i & wb_stb_i & !full_decoding;









//
// Optional registration of WB error
//

always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		wb_err_o <=  1'b0;
	else
		wb_err_o <=  wb_err & ~wb_err_o;




//
// Full address decoder
//

assign full_decoding = (wb_adr_i[7:6] == {7-6+1{1'b0}}) &
			(wb_adr_i[1:0] == {1-0+1{1'b0}});




//
// GPIO registers address decoder
//

assign rgpio_out_sel = wb_cyc_i & wb_stb_i & (wb_adr_i[6-1:1+1] == 4'h1) & full_decoding;


assign rgpio_oe_sel = wb_cyc_i & wb_stb_i & (wb_adr_i[6-1:1+1] == 4'h2) & full_decoding;


assign rgpio_inte_sel = wb_cyc_i & wb_stb_i & (wb_adr_i[6-1:1+1] == 4'h3) & full_decoding;


assign rgpio_ptrig_sel = wb_cyc_i & wb_stb_i & (wb_adr_i[6-1:1+1] == 4'h4) & full_decoding;


assign rgpio_aux_sel = wb_cyc_i & wb_stb_i & (wb_adr_i[6-1:1+1] == 4'h5) & full_decoding;


assign rgpio_ctrl_sel = wb_cyc_i & wb_stb_i & (wb_adr_i[6-1:1+1] == 4'h6) & full_decoding;


assign rgpio_ints_sel = wb_cyc_i & wb_stb_i & (wb_adr_i[6-1:1+1] == 4'h7) & full_decoding;


assign rgpio_eclk_sel = wb_cyc_i & wb_stb_i & (wb_adr_i[6-1:1+1] == 4'h8) & full_decoding;


assign rgpio_nec_sel = wb_cyc_i & wb_stb_i & (wb_adr_i[6-1:1+1] == 4'h9) & full_decoding;



//
// Write to RGPIO_CTRL or update of RGPIO_CTRL[INT] bit
//

always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		rgpio_ctrl <=  2'b0;
	else if (rgpio_ctrl_sel && wb_we_i)
		rgpio_ctrl <=  wb_dat_i[1:0];
	else if (rgpio_ctrl[0])
		rgpio_ctrl[1] <=  rgpio_ctrl[1] | wb_inta_o;




//
// Write to RGPIO_OUT
//

always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		rgpio_out <=  {gw{1'b0}};
	else if (rgpio_out_sel && wb_we_i)
    begin





     if ( wb_sel_i [3] == 1'b1 )
       rgpio_out [gw-1:24] <=  wb_dat_i [gw-1:24] ;
     if ( wb_sel_i [2] == 1'b1 )
       rgpio_out [23:16] <=  wb_dat_i [23:16] ;
     if ( wb_sel_i [1] == 1'b1 )
       rgpio_out [15:8] <=  wb_dat_i [15:8] ;
     if ( wb_sel_i [0] == 1'b1 )
       rgpio_out [7:0] <=  wb_dat_i [7:0] ;



















   end





//
// Write to RGPIO_OE.
//

always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		rgpio_oe <=  {gw{1'b0}};
	else if (rgpio_oe_sel && wb_we_i)
  begin





     if ( wb_sel_i [3] == 1'b1 )
       rgpio_oe [gw-1:24] <=  wb_dat_i [gw-1:24] ;
     if ( wb_sel_i [2] == 1'b1 )
       rgpio_oe [23:16] <=  wb_dat_i [23:16] ;
     if ( wb_sel_i [1] == 1'b1 )
       rgpio_oe [15:8] <=  wb_dat_i [15:8] ;
     if ( wb_sel_i [0] == 1'b1 )
       rgpio_oe [7:0] <=  wb_dat_i [7:0] ;



















   end





//
// Write to RGPIO_INTE
//

always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		rgpio_inte <=  {gw{1'b0}};
	else if (rgpio_inte_sel && wb_we_i)
  begin





     if ( wb_sel_i [3] == 1'b1 )
       rgpio_inte [gw-1:24] <=  wb_dat_i [gw-1:24] ;
     if ( wb_sel_i [2] == 1'b1 )
       rgpio_inte [23:16] <=  wb_dat_i [23:16] ;
     if ( wb_sel_i [1] == 1'b1 )
       rgpio_inte [15:8] <=  wb_dat_i [15:8] ;
     if ( wb_sel_i [0] == 1'b1 )
       rgpio_inte [7:0] <=  wb_dat_i [7:0] ;



















   end






//
// Write to RGPIO_PTRIG
//

always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		rgpio_ptrig <=  {gw{1'b0}};
	else if (rgpio_ptrig_sel && wb_we_i)
  begin





     if ( wb_sel_i [3] == 1'b1 )
       rgpio_ptrig [gw-1:24] <=  wb_dat_i [gw-1:24] ;
     if ( wb_sel_i [2] == 1'b1 )
       rgpio_ptrig [23:16] <=  wb_dat_i [23:16] ;
     if ( wb_sel_i [1] == 1'b1 )
       rgpio_ptrig [15:8] <=  wb_dat_i [15:8] ;
     if ( wb_sel_i [0] == 1'b1 )
       rgpio_ptrig [7:0] <=  wb_dat_i [7:0] ;



















   end
    




//
// Write to RGPIO_AUX
//

always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		rgpio_aux <=  {gw{1'b0}};
	else if (rgpio_aux_sel && wb_we_i)
  begin





     if ( wb_sel_i [3] == 1'b1 )
       rgpio_aux [gw-1:24] <=  wb_dat_i [gw-1:24] ;
     if ( wb_sel_i [2] == 1'b1 )
       rgpio_aux [23:16] <=  wb_dat_i [23:16] ;
     if ( wb_sel_i [1] == 1'b1 )
       rgpio_aux [15:8] <=  wb_dat_i [15:8] ;
     if ( wb_sel_i [0] == 1'b1 )
       rgpio_aux [7:0] <=  wb_dat_i [7:0] ;



















   end






//
// Write to RGPIO_ECLK
//

always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		rgpio_eclk <=  {gw{1'b0}};
	else if (rgpio_eclk_sel && wb_we_i)
  begin





     if ( wb_sel_i [3] == 1'b1 )
       rgpio_eclk [gw-1:24] <=  wb_dat_i [gw-1:24] ;
     if ( wb_sel_i [2] == 1'b1 )
       rgpio_eclk [23:16] <=  wb_dat_i [23:16] ;
     if ( wb_sel_i [1] == 1'b1 )
       rgpio_eclk [15:8] <=  wb_dat_i [15:8] ;
     if ( wb_sel_i [0] == 1'b1 )
       rgpio_eclk [7:0] <=  wb_dat_i [7:0] ;



















   end








//
// Write to RGPIO_NEC
//

always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		rgpio_nec <=  {gw{1'b0}};
	else if (rgpio_nec_sel && wb_we_i)
  begin





     if ( wb_sel_i [3] == 1'b1 )
       rgpio_nec [gw-1:24] <=  wb_dat_i [gw-1:24] ;
     if ( wb_sel_i [2] == 1'b1 )
       rgpio_nec [23:16] <=  wb_dat_i [23:16] ;
     if ( wb_sel_i [1] == 1'b1 )
       rgpio_nec [15:8] <=  wb_dat_i [15:8] ;
     if ( wb_sel_i [0] == 1'b1 )
       rgpio_nec [7:0] <=  wb_dat_i [7:0] ;



















   end






//
// synchronize inputs to systam clock
//

always @(posedge wb_clk_i or posedge wb_rst_i)
  if (wb_rst_i) begin
    sync      <=  {gw{1'b0}} ; 
    ext_pad_s <=  {gw{1'b0}} ; 
  end else begin
    sync      <=  ext_pad_i  ; 
    ext_pad_s <=  sync       ; 
  end


 // GPIO_SYNC_IN_WB
  
//
// Latch into RGPIO_IN
//

always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		rgpio_in <=  {gw{1'b0}};
	else
		rgpio_in <=  in_muxed;







//
// external clock enabled
// synchronized to system clock
// (one clock domain)
//

reg  sync_clk,
     clk_s   ,
     clk_r   ;
wire pedge   ,
     nedge   ;
wire [gw-1:0] pedge_vec   ,
              nedge_vec   ;
wire [gw-1:0] in_lach     ;

assign pedge =  clk_s & !clk_r ; 
assign nedge = !clk_s &  clk_r ; 
assign pedge_vec = {gw{pedge}} ;   
assign nedge_vec = {gw{nedge}} ;   

assign in_lach = (~rgpio_nec & pedge_vec) | (rgpio_nec & nedge_vec) ;
assign extc_in = (in_lach & ext_pad_s) | (~in_lach & pextc_sampled) ;

always @(posedge wb_clk_i or posedge wb_rst_i)
  if (wb_rst_i) begin
    sync_clk <=  1'b0 ;
    clk_s    <=  1'b0 ;
    clk_r    <=  1'b0 ;
  end else begin
    sync_clk <=  clk_pad_i ;
    clk_s    <=  sync_clk  ;
    clk_r    <=  clk_s     ;
  end

always @(posedge wb_clk_i or posedge wb_rst_i)
  if (wb_rst_i) begin
    pextc_sampled <=  {gw{1'b0}};
  end else begin
    pextc_sampled <=  extc_in ;
  end

assign in_muxed = (rgpio_eclk & pextc_sampled) | (~rgpio_eclk & ext_pad_s) ;








































































































































 //  GPIO_SYNC_CLK_WB






 //  GPIO_CLKPAD



//
// Mux all registers when doing a read of GPIO registers
//
always @(wb_adr_i or rgpio_in or rgpio_out or rgpio_oe or rgpio_inte or
		rgpio_ptrig or rgpio_aux or rgpio_ctrl or rgpio_ints or rgpio_eclk or rgpio_nec)
	case (wb_adr_i[6-1:1+1])	// synopsys full_case parallel_case

  
  	4'h1: begin
			wb_dat[dw-1:0] = rgpio_out;
		end
  
  
		4'h2: begin
			wb_dat[dw-1:0] = rgpio_oe;
		end
  
  
		4'h3: begin
			wb_dat[dw-1:0] = rgpio_inte;
		end
  
  
		4'h4: begin
			wb_dat[dw-1:0] = rgpio_ptrig;
		end
  
  
		4'h9: begin
			wb_dat[dw-1:0] = rgpio_nec;
		end
  
  
		4'h8: begin
			wb_dat[dw-1:0] = rgpio_eclk;
		end
  
  
		4'h5: begin
			wb_dat[dw-1:0] = rgpio_aux;
		end
  
  
		4'h6: begin
			wb_dat[1:0] = rgpio_ctrl;
			wb_dat[dw-1:2] = {dw-2{1'b0}};
		end
  

  
		4'h7: begin
			wb_dat[dw-1:0] = rgpio_ints;
		end
  
		default: begin
			wb_dat[dw-1:0] = rgpio_in;
		end
	endcase

//
// WB data output
//

always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		wb_dat_o <=  {dw{1'b0}};
	else
		wb_dat_o <=  wb_dat;




//
// RGPIO_INTS
//

always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		rgpio_ints <=  {gw{1'b0}};
	else if (rgpio_ints_sel && wb_we_i)
		rgpio_ints <=  wb_dat_i[gw-1:0];
	else if (rgpio_ctrl[0])
		rgpio_ints <=  (rgpio_ints | ((in_muxed ^ rgpio_in) & (in_muxed ^ rgpio_ptrig)) & rgpio_inte);




//
// Generate interrupt request
//
assign wb_inta = |rgpio_ints ? rgpio_ctrl[0] : 1'b0;

//
// Optional registration of WB interrupt
//

always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		wb_inta_o <=  1'b0;
	else
		wb_inta_o <=  wb_inta;


 // GPIO_REGISTERED_WB_OUTPUTS

//
// Output enables are RGPIO_OE bits
//
assign ext_padoe_o = rgpio_oe;

//
// Generate GPIO outputs
//

assign out_pad = rgpio_out & ~rgpio_aux | aux_i & rgpio_aux;


 //  GPIO_AUX_IMPLEMENT

//
// Optional registration of GPIO outputs
//

always @(posedge wb_clk_i or posedge wb_rst_i)
	if (wb_rst_i)
		ext_pad_o <=  {gw{1'b0}};
	else
		ext_pad_o <=  out_pad;


 // GPIO_REGISTERED_IO_OUTPUTS



















 //  GPIO_IMPLEMENTED

endmodule
