module rtlpp_08957 (
    address,
    address2,
    byteenable,
    byteenable2,
    chipselect,
    chipselect2,
    clk,
    clk2,
    clken,
    clken2,
    write,
    write2,
    writedata,
    writedata2,
    readdata,
    readdata2
);

  output [31:0] readdata;
  output [31:0] readdata2;
  input [11:0] address;
  input [11:0] address2;
  input [3:0] byteenable;
  input [3:0] byteenable2;
  input chipselect;
  input chipselect2;
  input clk;
  input clk2;
  input clken;
  input clken2;
  input write;
  input write2;
  input [31:0] writedata;
  input [31:0] writedata2;
  
  reg [31:0] memory_a [0:4095];
  reg [31:0] memory_b [0:4095];
  reg [31:0] readdata;
  reg [31:0] readdata2;

  wire wren;
  wire wren2;
  
  assign wren = chipselect & write;
  assign wren2 = chipselect2 & write2;
  
  initial begin
    // Load memory contents from the file
    $readmemh("ddr2_v10_1_0002_sequencer_rom_ddr2.hex", memory_a);
    $readmemh("ddr2_v10_1_0002_sequencer_rom_ddr2.hex", memory_b);
  end
  
  always @(posedge clk) begin
    if (clken) begin
      if (wren) begin
        if (byteenable[0]) memory_a[address][7:0] <= writedata[7:0];
        if (byteenable[1]) memory_a[address][15:8] <= writedata[15:8];
        if (byteenable[2]) memory_a[address][23:16] <= writedata[23:16];
        if (byteenable[3]) memory_a[address][31:24] <= writedata[31:24];
      end
      else begin
        readdata <= memory_a[address];
      end
    end
  end
  
  always @(posedge clk2) begin
    if (clken2) begin
      if (wren2) begin
        if (byteenable2[0]) memory_b[address2][7:0] <= writedata2[7:0];
        if (byteenable2[1]) memory_b[address2][15:8] <= writedata2[15:8];
        if (byteenable2[2]) memory_b[address2][23:16] <= writedata2[23:16];
        if (byteenable2[3]) memory_b[address2][31:24] <= writedata2[31:24];
      end
      else begin
        readdata2 <= memory_b[address2];
      end
    end
  end
  
endmodule
