assign max_count = (1 << width) - 1;

assign ones_5lsb = (width < 5) ? 0 :

// Calculating the next state
assign next_state = (ld_n == 0) ? data_in :
                    ((cnt_en == 0) ? current_state : current_state + 1);

// State transition

    if (!rst_n)
        current_state <= 0;
    else
        current_state <= next_state;
