            i <= 0;
            Next_state <= 2'b01;
        end

        2'b01: begin // State S1 (Multiplication)
            out_array[i] <= Activation[i] * Weight[i];
            i <= i + 1;
            Next_state <= 2'b10;
        end

        2'b10: begin // State S2 (Summation)
            if (i <= 19) begin
                y <= y + out_array[i];
                i <= i + 1;
                Next_state <= 2'b10;
            end else begin
                Next_state <= 2'b11;
            end
        end

        2'b11: begin // State S3 (Output)
            out_neuron <= y[12:0];
            Pres_state <= 2'b00;
        end

        default: Pres_state <= 2'b00;
    endcase
end

endmodule
