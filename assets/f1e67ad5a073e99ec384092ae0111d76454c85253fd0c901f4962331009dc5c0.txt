    output reg Jump,

            ALUsrc = 1'b1;
            RegWrite = 1'b1;
        end

            #5;
            RegDst = 2'b00;

            Branch = 1'b1;
            MemRead = 1'b0;
            MemtoReg = 2'bxx;
            ALUop = 2'b01;
            MemWrite = 1'b0;
            ALUsrc = 1'b0;
            RegWrite = 1'b0;
        end

            #5;

        end
        default: begin
            #5;
            RegDst = 2'bxx;
            Jump = 1'bx;
            Branch = 1'bx;
            MemRead = 1'bx;
            MemtoReg = 2'bxx;
            ALUop = 2'bxx;
            MemWrite = 1'bx;
            ALUsrc = 1'bx;
            RegWrite = 1'bx;
        end
    endcase
end

endmodule
