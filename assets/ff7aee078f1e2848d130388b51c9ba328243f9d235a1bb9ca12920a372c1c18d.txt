    parameter WIDTH = 32
) (
    input [WIDTH-1:0] a,
    input [WIDTH-1:0] b,
    output [WIDTH-1:0] result
);
    assign result = a - b;
endmodule

module fp_div #(
    parameter WIDTH = 32
) (
    input [WIDTH-1:0] a,
    input [WIDTH-1:0] b,
    output reg [WIDTH-1:0] result
);
    reg [WIDTH-1:0] abs_a, abs_b;
    reg sign;

    always @* begin
        abs_a = a[WIDTH-1] ? -a : a;
        abs_b = b[WIDTH-1] ? -b : b;
        sign = a[WIDTH-1] ^ b[WIDTH-1];
        result = sign ? -(abs_a / abs_b) : (abs_a / abs_b);
    end
endmodule

module adder (
    input [31:0] a,
    input [31:0] b,
    output [31:0] result
);
    fp_add #(.WIDTH(32)) add_inst (
        .a(a),
        .b(b),
        .result(result)

    parameter IN_WIDTH = 64,
    parameter OUT_WIDTH = 32
) (
    input [IN_WIDTH-1:0] in,
    output [OUT_WIDTH-1:0] out
);
    assign out = {in[IN_WIDTH-1], in[IN_WIDTH-2:IN_WIDTH-OUT_WIDTH]};
endmodule
