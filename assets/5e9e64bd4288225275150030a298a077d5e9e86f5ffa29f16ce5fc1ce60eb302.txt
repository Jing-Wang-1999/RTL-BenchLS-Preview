module origen_057857 (
    input clk,
    input reset,
    input state_alu,
    input state_fetch,
    input state_read,
    input state_write,
    input [31:0] ob,
    input destpdlx,
    input destpdlp,
    input pdlcnt,
    input srcpdlpop,
    input srcpdlidx,
    input srcpdlptr,
    output reg [9:0] pdlidx,
    output reg [9:0] pdlptr,
    output pidrive,
    output ppdrive
);

    // Drive enables
    assign pidrive = srcpdlidx && (state_alu || state_write || state_fetch);
    assign ppdrive = srcpdlptr && (state_alu || state_write || state_fetch);

    // Pointer update logic
    always @(posedge clk) begin
        if (reset) begin
            pdlidx <= 10'b0;
            pdlptr <= 10'b0;
        end else begin
            // pdlidx update
            if (state_write && destpdlx) begin
                pdlidx <= ob[9:0];
            end

            // pdlptr update
            if (state_read) begin
                if (!destpdlp && pdlcnt && !srcpdlpop) begin
                    pdlptr <= pdlptr + 1'b1;
                end
            end else if (state_fetch) begin
                if (destpdlp) begin
                    pdlptr <= ob[9:0];
                end else if (!destpdlp && pdlcnt && srcpdlpop) begin
                    pdlptr <= pdlptr - 1'b1;
                end
            end
        end
    end

endmodule
