module origen_183408(
    input wire En,
    input wire [2:0] S,
    input wire [7:0] D,
    output reg Y,
    output reg Yn
);
    always @(*) begin
        if (En) begin
            Y = 1'b0;
            Yn = 1'b1;
        end else begin
            Y = D[S];
            Yn = ~Y;
        end
    end
endmodule
