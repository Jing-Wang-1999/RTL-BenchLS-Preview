module origen_125357(
    input wire clk,
    input wire [3:0] parallel_in,
    input wire select,
    output wire [4:0] sum_out
);

    wire [3:0] shift_out, count_out;

    shift_register sr (
        .clk(clk),
        .parallel_in(parallel_in),
        .shift_out(shift_out)
    );

    counter cnt (
        .clk(clk),
        .count_out(count_out)
    );

    wire [3:0] active_out;
    assign active_out = select ? count_out : shift_out;

    adder add (
        .a(shift_out),
        .b(count_out),
        .sum(sum_out)
    );

endmodule

module shift_register(
    input wire clk,
    input wire [3:0] parallel_in,
    output reg [3:0] shift_out
);

    always @(posedge clk) begin
        shift_out <= {shift_out[2:0], parallel_in[3]};
    end

endmodule

module counter(
    input wire clk,
    output reg [3:0] count_out
);

    always @(posedge clk) begin
        count_out <= count_out + 1;
    end

endmodule

module adder(
    input wire [3:0] a,
    input wire [3:0] b,
    output wire [4:0] sum
);

    assign sum = a + b;

endmodule
