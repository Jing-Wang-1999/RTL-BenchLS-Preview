module clkdivider(
    input wire clk,
    input wire reset,
    output reg clk_out
);
    reg [6:0] counter;

    always @(posedge clk or posedge reset) begin
        if (reset) begin
            counter <= 7'b0;
            clk_out <= 1'b0;
        end else begin
            if (counter == 7'd127) begin
                counter <= 7'b0;
                clk_out <= ~clk_out;
            end else begin
                counter <= counter + 1;
            end
        end
    end
endmodule
