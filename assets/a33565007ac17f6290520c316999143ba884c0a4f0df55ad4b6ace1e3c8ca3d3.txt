module origen_116131 (
    input wire i_clk,
    output wire p_clk,
    output reg p_cs,
    output reg p_in,
    input wire p_out,
    output reg [9:0] o_data0,
    output reg [9:0] o_data1,
    output reg o_done
);

reg [4:0] counter;
reg channel;
reg [9:0] shift_reg;

assign p_clk = i_clk;

always @(negedge i_clk) begin
    counter <= counter + 1;
end

always @(posedge i_clk) begin
    case (counter)
        5'd0: begin
            p_cs <= 0;
            p_in <= 1;
        end
        5'd1: p_in <= 1;
        5'd2: p_in <= channel;
        5'd3: p_in <= 1;
        5'd4: p_in <= 0;
        5'd6, 5'd7, 5'd8, 5'd9, 5'd10, 5'd11, 5'd12, 5'd13, 5'd14, 5'd15:
            shift_reg <= {shift_reg[8:0], p_out};
        5'd15: begin
            if (channel)
                o_data1 <= shift_reg;
            else
                o_data0 <= shift_reg;
            o_done <= 1;
        end
        5'd16: o_done <= 0;
        5'd31: begin
            p_cs <= 1;
            channel <= ~channel;
        end
        default: begin
            p_cs <= 0;
            p_in <= 0;
        end
    endcase
end

endmodule
