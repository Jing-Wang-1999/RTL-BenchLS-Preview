module Alu (
    input [0:15] x,
    input [0:15] y,
    output [0:15] out,
    input zx, zy, nx, ny, f, no,
    output zr, ng
);
    // Internal signals
    wire [0:15] x1, y1, out1, out2;
    wire nout;

    // Zero processing
    assign x1 = zx ? 16'b0 : x;
    assign y1 = zy ? 16'b0 : y;

    // Negate processing
    assign x1 = nx ? ~x1 : x1;
    assign y1 = ny ? ~y1 : y1;

    // Function
    assign out1 = f ? (x1 + y1) : (x1 & y1);

    // Negate output
    assign out2 = no ? ~out1 : out1;

    // Zero flag
    assign zr = (out2 == 16'b0);

    // Negative flag
    assign ng = out2[0];

    // Output
    assign out = out2;
endmodule
