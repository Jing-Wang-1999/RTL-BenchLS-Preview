"""SDRAM controller: independent Wishbone transactions against the upstream device model."""
import random
import re
from pathlib import Path

SOURCE=Path('/home/jwangjw/project_3rd/ICLR-RTL-BenchLS-backup-jul-29/sources/opencores_git_mirror/memory_core_8-16-32_bit_sdram_controller')
TB=SOURCE/'verif/tb/tb_top.sv'
MODEL=SOURCE/'verif/model/mt48lc2m32b2.v'


def test(mode):
    prefix=TB.read_text().split('int dfifo[$]')[0].replace('module tb_top;','module tb;')
    prefix='`define SDR_32BIT\n'+prefix
    s=prefix+'''
integer refreshes=0,reads=0,writes=0,activates=0,base_refresh=0;
always @(posedge sdram_clk_d)if(RESETN && sdr_cke && !sdr_cs_n)begin
case({sdr_ras_n,sdr_cas_n,sdr_we_n})
3'b001:refreshes=refreshes+1;
3'b011:activates=activates+1;
3'b101:reads=reads+1;
3'b100:writes=writes+1;
endcase end
initial begin #50000000;$fatal(1,"test timeout");end
task access;input writing;input [31:0] address,value;input [3:0] mask;integer count;begin
@(negedge sys_clk);wb_stb_i=1;wb_cyc_i=1;wb_we_i=writing;wb_addr_i=address[27:2];wb_dat_i=value;wb_sel_i=mask;wb_cti_i=0;
count=0;@(posedge sys_clk);
while(wb_ack_o!==1 && count<5000)begin @(posedge sys_clk);count=count+1;end
if(wb_ack_o!==1)$fatal(1,"Wishbone timeout");
if(!writing && wb_dat_o!==value)$fatal(1,"memory %h: %h != %h",address,wb_dat_o,value);
@(negedge sys_clk);wb_stb_i=0;wb_cyc_i=0;wb_we_i=0;repeat(4)@(posedge sys_clk);end endtask
initial begin
wb_addr_i=0;wb_dat_i=0;wb_sel_i=0;wb_we_i=0;wb_stb_i=0;wb_cyc_i=0;wb_cti_i=0;
RESETN=1;#100;RESETN=0;#10000;RESETN=1;
wait(sdr_init_done===1);repeat(40)@(posedge sdram_clk);
if(refreshes<2)$fatal(1,"initialization refresh missing");base_refresh=refreshes;
'''
    rng=random.Random(9060637+mode);expected={}
    if mode==0:
        addresses=[0,4,0x3fc,0x400,0xffc,0x1000,0x40000,0x400000]
        addresses += [(rng.randrange(0x1fffff)<<2) for _ in range(48)]
    elif mode==1:addresses=[0x100+4*i for i in range(24)]
    elif mode==2:addresses=[base+4*i for base in [0x3e0,0xfe0,0x1fe0,0x40fe0] for i in range(16)]
    else:addresses=[0x5000+4*i for i in range(16)]
    for a in addresses:
        d=rng.getrandbits(32);expected[a]=d;s+=f"access(1,32'h{a:x},32'h{d:x},15);\n"
    if mode==1:
        for a in addresses:
            d=rng.getrandbits(32);mask=rng.randrange(1,16);bits=sum(255<<(8*i) for i in range(4) if mask>>i&1)
            expected[a]=(expected[a]&~bits)|(d&bits)
            s+=f"access(1,32'h{a:x},32'h{d:x},{mask});\n"
    s+=f'repeat({6000 if mode==3 else 200})@(posedge sdram_clk);\n'
    rng.shuffle(addresses)
    for a in addresses:s+=f"access(0,32'h{a:x},32'h{expected[a]:x},15);\n"
    s+='repeat(100)@(posedge sdram_clk);if(reads<1 || writes<1 || activates<1)$fatal(1,"missing physical SDRAM traffic");\n'
    if mode==3:s+='if(refreshes<=base_refresh)$fatal(1,"periodic refresh missing");\n'
    s+='$display("RTL_BENCHLS_PASS");$finish(0);end endmodule\n'
    model=MODEL.read_text(encoding='cp1252')
    model=re.sub(r'(wire\s+Debug\s*=\s*)1\'b1',r"\g<1>1'b0",model)
    return s+model

- Wishbone wb_addr_i is a WORD address (byte address divided by four). Hold
  cyc/stb, direction, address, data, byte enables until ACK. Classical accesses
