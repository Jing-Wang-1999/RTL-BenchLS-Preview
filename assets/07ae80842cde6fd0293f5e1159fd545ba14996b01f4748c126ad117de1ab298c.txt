module origen_071564 (
    input wire clk,
    input wire [63:0] data_in,
    output reg [63:0] data_out
);

    // Initialize data_out to zero
    initial begin
        data_out = 64'b0;
    end

    // Capture data_in on the rising edge of clk
    always @(posedge clk) begin
        data_out <= data_in;
    end

endmodule
