module origen_147786 (
    input A, B, C, D,
    input VPWR, VGND, VPB, VNB,
    output Y
);

    wire nor_out, pwrgood_out;

    // Instantiate 4-input NOR gate
    assign nor_out = ~(A | B | C | D);

    // Instantiate power good module
    assign pwrgood_out = (VPWR & VPB & ~VGND & ~VNB) ? nor_out : 1'bz;

    // Buffer the power good output
    buf buffer (Y, pwrgood_out);

endmodule

    end
