









































module uart_core (  
        line_reset_n ,
        line_clk ,

	
        cfg_tx_enable  , 
        cfg_rx_enable  , 
        cfg_stop_bit   , 
        cfg_pri_mod    , 
	cfg_baud_16x   ,

    
        tx_data_avail,
        tx_rd,
        tx_data,
         

    
        rx_ready,
        rx_wr,
        rx_data,

       
        frm_error,
	par_error,

	baud_clk_16x,

       
        rxd,
        txd

     );







input        line_reset_n         ; 
input        line_clk             ; 




input         cfg_tx_enable        ; 
input         cfg_rx_enable        ; 
input         cfg_stop_bit         ; 
input   [1:0] cfg_pri_mod          ; 
input   [11:0] cfg_baud_16x        ; 




input         tx_data_avail        ; 
input [7:0]   tx_data              ; 
output        tx_rd                ; 





input         rx_ready            ; 
output [7:0]  rx_data             ; 
output        rx_wr               ; 





output        frm_error            ; 
output        par_error            ; 

output        baud_clk_16x         ; 





input         rxd                  ; 
output        txd                  ; 



wire [1  : 0]   error_ind          ;







clk_ctl #(11) u_clk_ctl (
   
       .clk_o          (baud_clk_16x),

   
       .mclk           (line_clk),
       .reset_n        (line_reset_n), 
       .clk_div_ratio  (cfg_baud_16x)
   );


uart_txfsm u_txfsm (
               . reset_n           ( line_reset_n      ),
               . baud_clk_16x      ( baud_clk_16x      ),

               . cfg_tx_enable     ( cfg_tx_enable     ),
               . cfg_stop_bit      ( cfg_stop_bit      ),
               . cfg_pri_mod       ( cfg_pri_mod       ),

       
               . fifo_empty        ( !tx_data_avail    ),
               . fifo_rd           ( tx_rd             ),
               . fifo_data         ( tx_data           ),

          
               . so                ( txd                )
          );


uart_rxfsm u_rxfsm (
               . reset_n           (  line_reset_n     ),
               . baud_clk_16x      (  baud_clk_16x     ) ,

               . cfg_rx_enable     (  cfg_rx_enable    ),
               . cfg_stop_bit      (  cfg_stop_bit     ),
               . cfg_pri_mod       (  cfg_pri_mod      ),

               . error_ind         (  error_ind        ),

       
               .  fifo_aval        ( rx_ready          ),
               .  fifo_wr          ( rx_wr             ),
               .  fifo_data        ( rx_data           ),

          
               .  si               (rxd              )
          );


wire   frm_error          = (error_ind == 2'b01);
wire   par_error          = (error_ind == 2'b10);



endmodule
