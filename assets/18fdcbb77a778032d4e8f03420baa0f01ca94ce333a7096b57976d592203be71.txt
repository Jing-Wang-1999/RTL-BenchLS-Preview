module mg_00175 (
    input  wire       rst_n,
    input  wire       clk,
    input  wire       clken,
    input  wire       d,
    output reg        valid
);
    reg [4:0] r;
    wire [4:0] next_r;
    wire msb_match;

    // Compute next state
    assign msb_match = (r[4] == d);
    assign next_r = msb_match ? {r[3:1], 1'b0} : {r[3], ~r[2], r[1:0], 1'b1};

    // Combinatorial valid output
    always @(*) begin
        valid = (next_r == 5'b01100);
    end

    // Sequential logic
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            r <= 5'b11111;
        end else if (clken) begin
            r <= next_r;
        end
    end
endmodule
