module origen_034521 (
    input clk,
    input resetn,
    input x,
    input y,
    output reg f,
    output reg g
);

    // State encoding
    localparam IDLE = 4'b0000;
    localparam F0   = 4'b0001;
    localparam F1   = 4'b0010;
    localparam X0   = 4'b0011;
    localparam X1   = 4'b0100;
    localparam X2   = 4'b0101;
    localparam Y0   = 4'b0110;
    localparam Y1   = 4'b0111;
    localparam Y00  = 4'b1000;
    localparam Y01  = 4'b1001;

    reg [3:0] state, next_state;

    // State transition logic
    always @(posedge clk or negedge resetn) begin
        if (!resetn)
            state <= IDLE;
        else
            state <= next_state;
    end

    // Next state logic
    always @(*) begin
        case (state)
            IDLE: next_state = F0;
            F0: next_state = F1;
            F1: next_state = x ? X0 : F1;
            X0: next_state = x ? X0 : X1;
            X1: next_state = x ? X2 : F1;
            X2: next_state = y ? Y1 : Y0;
            Y0: next_state = y ? Y01 : Y00;
            Y1: next_state = Y1;
            Y00: next_state = Y00;
            Y01: next_state = Y01;
            default: next_state = IDLE;
        endcase
    end

    // Output logic
    always @(*) begin
        f = (state == F0);
        g = (state == X2 || state == Y1 || state == Y0 || state == Y01);
    end

endmodule
