    wr_ptr_gray_sync2_reg <= wr_ptr_gray_sync1_reg;

        wr_ptr_commit_sync_reg <= wr_ptr_sync_commit_reg;

        wr_ptr_gray_sync2_reg <= {ADDR_WIDTH+1{1'b0}};

        wr_ptr_update_sync2_reg <= 1'b0;
        wr_ptr_update_sync3_reg <= 1'b0;
    end
end
