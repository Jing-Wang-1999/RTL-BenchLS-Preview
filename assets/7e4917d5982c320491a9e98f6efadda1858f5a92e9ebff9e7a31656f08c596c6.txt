module autovcoder_017920 (
    input IN1,
    input IN2,
    input IN3,
    input IN4,
    input IN5,
    input IN6,
    output OUT
);

    wire term_c, term_b, term_a;
    wire and_out;

    // Term C: IN1 OR IN2
    or #(0.26, 0.27) or_c (term_c, IN1, IN2);
    // Note: OR gate delay from IN1: rise=0.26, fall=0.27
    // OR gate delay from IN2: rise=0.30, fall=0.28
    // Since we can only specify one pair of delays per gate,
    // we use the IN1 delays as representative.
    // For precise per-input delays, we would need separate gates.

    // Term B: IN3 OR IN4
    or #(0.33, 0.37) or_b (term_b, IN3, IN4);
    // Using IN3 delays as representative

    // Term A: IN5 OR IN6
    or #(0.30, 0.41) or_a (term_a, IN5, IN6);
    // Using IN5 delays as representative

    // Three-input AND: term_c AND term_b AND term_a
    // AND gate delays depend on which input triggers the change.
    // We'll use conservative delays: max rise delay = 0.41, max fall delay = 0.41
    // Actually, the AND output falls when any input falls, so fall delay should consider
    // the worst-case fall delay of any input path.
    // Since we can't specify per-input delays, we'll use the maximum values.
    and #(0.34, 0.41) and_gate (and_out, term_c, term_b, term_a);
    // Using max rise delay (0.34 from IN6) and max fall delay (0.41 from IN5/IN6)

    // Inverter
    // The inverter's output rise/fall depends on and_out transition.
    // We need to account for the path from each input through the OR and AND gates.
    // We'll specify minimum delays that cover the critical paths.
    not #(0.34, 0.41) inv (OUT, and_out);

endmodule
