module mg_03551(
    input [15:0] x,
    input [15:0] y,
    input zx,
    input nx,
    input zy,
    input ny,
    input f,
    input no,
    output [15:0] out,
    output zr,
    output ng
);
    wire [15:0] x_processed, y_processed;
    wire [15:0] alu_result;
    wire [15:0] final_result;

    // Process x
    assign x_processed = (zx ? 16'b0 : x) ^ {16{nx}};

    // Process y
    assign y_processed = (zy ? 16'b0 : y) ^ {16{ny}};

    // ALU operation
    assign alu_result = f ? (x_processed + y_processed) : (x_processed & y_processed);

    // Final output processing
    assign final_result = alu_result ^ {16{no}};

    // Output assignment
    assign out = final_result;

    // Status flags
    assign zr = (out == 16'b0);
    assign ng = out[15];

endmodule
