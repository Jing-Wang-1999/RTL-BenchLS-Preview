module origen_111750 #(
    parameter DEPTH = 16,
    parameter WIDTH = 32,
    parameter ADDRESS_WIDTH = 8
) (
    input wire clk,
    input wire reset,
    input wire [ADDRESS_WIDTH-1:0] address,
    input wire [WIDTH-1:0] data_wr,
    output reg [WIDTH-1:0] data_rd,
    input wire wr_enable,
    output reg data_valid
);

    // Cache entry structure
    reg [ADDRESS_WIDTH-1:0] keys [0:DEPTH-1];
    reg [WIDTH-1:0] values [0:DEPTH-1];
    reg [DEPTH-1:0] cell_valid;
    reg [3:0] count [0:DEPTH-1];

    // PRNG and state variables
    reg [7:0] lfsr;
    reg [ADDRESS_WIDTH-1:0] old_key;
    reg [1:0] subcount;

    integer i;
    reg write_done;

    // PRNG function
    function [7:0] next_lfsr;
        input [7:0] current;
        begin
            next_lfsr = {current[6:0], current[7] ^ current[5] ^ current[4] ^ current[3]};
        end
    endfunction

    always @(posedge clk or posedge reset) begin
        if (reset) begin
            for (i = 0; i < DEPTH; i = i + 1) begin
                keys[i] <= 0;
                values[i] <= 0;
                count[i] <= 0;
            end
            cell_valid <= 0;
            data_rd <= 0;
            data_valid <= 0;
            lfsr <= 8'hFF;
            old_key <= 0;
            subcount <= 0;
        end else begin
            lfsr <= next_lfsr(lfsr);
            subcount <= subcount + 1;

            if (wr_enable) begin
                // Write operation
                data_valid <= 0;
                write_done = 0;
                if (cell_valid[lfsr[3:0]] == 0 || keys[lfsr[3:0]] == address) begin
                    keys[lfsr[3:0]] <= address;
                    values[lfsr[3:0]] <= data_wr;
                    cell_valid[lfsr[3:0]] <= 1;
                    count[lfsr[3:0]] <= 4'hF;
                    write_done = 1;
                end else begin
                    for (i = 0; i < DEPTH && !write_done; i = i + 1) begin
                        if (!cell_valid[i] || keys[i] == address) begin
                            keys[i] <= address;
                            values[i] <= data_wr;
                            cell_valid[i] <= 1;
                            count[i] <= 4'hF;
                            write_done = 1;
                        end else if (count[i] == 0) begin
                            keys[i] <= address;
                            values[i] <= data_wr;
                            cell_valid[i] <= 1;
                            count[i] <= 4'hF;
                            write_done = 1;
                        end else begin
                            count[i] <= count[i] - 1;
                        end
                    end
                end
            end else begin
                // Read operation
                data_valid <= 0;
                for (i = 0; i < DEPTH; i = i + 1) begin
                    if (cell_valid[i] && keys[i] == address) begin
                        data_rd <= values[i];
                        data_valid <= 1;
                        count[i] <= 4'hF;
                    end else if (count[i] != 0) begin
                        count[i] <= count[i] - 1;
                    end
                end
            end

            if (old_key != address) begin
                old_key <= address;
                subcount <= 0;
            end
        end
    end

endmodule
