module origen_001360 #(
    parameter NCH_ADC = 20,
    parameter ADC_CYC = 20,
    parameter NCH_SDM = 19,
    parameter SDM_CYC = 4,
    parameter FIFO_DIN64_CYC = 9
)(
    input wire RESET,
    input wire CLK,
    input wire USER_CLK,
    input wire [63:0] M_AXI_RX_TDATA,
    input wire M_AXI_RX_TVALID,
    output reg [511:0] DOUT,
    output reg DOUT_VALID,
    output wire FIFO_FULL
);

    reg [3:0] cnt;
    reg [511:0] sdm_adc_v;
    reg sdm_adc_v_valid;

    always @(posedge USER_CLK or posedge RESET) begin
        if (RESET) begin
            cnt <= 4'd0;
            sdm_adc_v <= 512'd0;
            sdm_adc_v_valid <= 1'b0;
        end else begin
            if (M_AXI_RX_TVALID) begin
                sdm_adc_v <= {sdm_adc_v[448:0], M_AXI_RX_TDATA[62:0]};
                
                if (M_AXI_RX_TDATA[63] || cnt == FIFO_DIN64_CYC-1) begin
                    cnt <= 4'd0;
                    sdm_adc_v_valid <= 1'b1;
                end else begin
                    cnt <= cnt + 1'b1;
                    sdm_adc_v_valid <= 1'b0;
                end
            end else begin
                sdm_adc_v_valid <= 1'b0;
            end
        end
    end

    always @(posedge USER_CLK or posedge RESET) begin
        if (RESET) begin
            DOUT <= 512'd0;
            DOUT_VALID <= 1'b0;
        end else begin
            DOUT <= {2'b00, sdm_adc_v[509:0]};
            DOUT_VALID <= sdm_adc_v_valid;
        end
    end

    assign FIFO_FULL = 1'b0; // Not implemented in this module

endmodule
