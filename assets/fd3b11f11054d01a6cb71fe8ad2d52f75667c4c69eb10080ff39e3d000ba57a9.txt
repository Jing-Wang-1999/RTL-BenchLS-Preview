#!/usr/bin/env python3
"""Replay the frozen upstream bench against an alternate RTL source directory.
Usage: python3 evaluate_opencores_project_172.py --case-result PATH --candidate-dir PATH --output-dir PATH
One upstream bench yields one Boolean point; no internal-vector points.
"""
import argparse,hashlib,json,pathlib,subprocess,os,re,shutil

def main():
 a=argparse.ArgumentParser();a.add_argument('--case-result',type=pathlib.Path,required=True);a.add_argument('--candidate-dir',type=pathlib.Path,required=True);a.add_argument('--output-dir',type=pathlib.Path,required=True);args=a.parse_args()
 d=json.loads(args.case_result.read_text());gold=args.case_result.resolve().parent/'source';cand=args.candidate_dir.resolve();out=args.output_dir.resolve();out.mkdir(parents=True,exist_ok=True)
 design=set(d['step1']['rtl_files']);points=[]
 for n,tb in enumerate(d['step4']['scored_testbench_paths']):
  e=next(e for e in d['testbenches'] if e['path']==tb);cmd=e['compile_command'][:];cmd[cmd.index('-o')+1]=str(out/f'point_{n}.vvp')
  for asset in e.get('runtime_input_files',[]):
   destination=out/asset['path'];destination.parent.mkdir(parents=True,exist_ok=True);shutil.copyfile(gold/asset['path'],destination)
  sim_cwd=out/e.get('simulation_cwd_relative','.');sim_cwd.mkdir(parents=True,exist_ok=True)
  i=1
  while i<len(cmd):
   if cmd[i]=='-I':cmd[i+1]=str(gold/cmd[i+1]);i+=2;continue
   if cmd[i] in design:cmd[i]=str(cand/cmd[i])
   elif (gold/cmd[i]).is_file():cmd[i]=str(gold/cmd[i])
   i+=1
  record={'testbench':tb,'compile_command':cmd,'passed':False}
  with (out/f'point_{n}_compile.log').open('w') as log:
   try:r=subprocess.run(cmd,cwd=out,stdout=log,stderr=subprocess.STDOUT,timeout=60);rc=r.returncode
   except subprocess.TimeoutExpired:rc='timeout'
  record['compile_returncode']=rc
  if rc==0:
   vvp=str(pathlib.Path(cmd[0]).with_name('vvp')) if '/' in cmd[0] else 'vvp';run=[vvp,'-n',str(out/f'point_{n}.vvp')];record['simulation_command']=run
   with (out/f'point_{n}.log').open('w') as log:
    try:r=subprocess.run(run,cwd=sim_cwd,stdout=log,stderr=subprocess.STDOUT,timeout=180,env={**os.environ,'IVERILOG_DUMPER':'none'});rc=r.returncode
    except subprocess.TimeoutExpired:rc='timeout'
   text=(out/f'point_{n}.log').read_text(errors='replace');record['simulation_returncode']=rc
   if d['selected_tops']==['fpu']:passed=rc==0 and text.count('Answer is correct')==e['pass_fail_rule']['expected_correct_answers'] and 'Error!' not in text
   elif d['selected_tops']==['pit_top']:passed=rc==0 and 'Simulation Passed' in text and 'Simulation Finished!!' in text and 'Simulation Failed' not in text
   elif d['selected_tops']==['aes']:
    if tb.endswith('/tb_kat.v'):passed=rc==0 and text.count('Test finished OK!')==2078 and 'ERROR:' not in text
    else:passed=rc==0 and 'Simulation Done !!!' in text and [x.lower().zfill(32) for x in re.findall(r"Output Data \d+: 128'h([0-9a-fA-FxXzZ]+)",text)]==e['pass_fail_rule']['expected_output_values']
   else:raise ValueError('No reviewed parser for selected project')
   record['passed']=passed
  points.append(record)
 result={'project_id':d['project_id'],'passed_testbenches':sum(p['passed'] for p in points),'total_testbenches':len(points),'points':points};result['score']=result['passed_testbenches']/result['total_testbenches'];(out/'result.json').write_text(json.dumps(result,indent=2)+'\n');print(json.dumps(result));return 0
if __name__=='__main__':main()
