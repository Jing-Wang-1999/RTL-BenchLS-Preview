always @(posedge clk or posedge reset) begin
        if (reset) begin
            count0 <= 0;
            count1 <= 0;
        end else if (d_clr) begin
            count0 <= 0;
            count1 <= 0;
        end else if (d_inc) begin
            if (count0 == 9) begin
                count0 <= 0;
                if (count1 == 9)
                    count1 <= 0;
                else
                    count1 <= count1 + 1;
            end else
                count0 <= count0 + 1;
        end
    end
