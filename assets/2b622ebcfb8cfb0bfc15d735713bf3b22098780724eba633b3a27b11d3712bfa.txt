module origen_178419(
    input clk,
    input reset,
    input ready,
    input [7:0] datain,
    output reg [7:0] dataout
);

    reg [1:0] state;
    reg [1:0] next_state;

    // State machine
    always @(posedge clk or posedge reset) begin
        if (reset) begin
            state <= 2'b00;
            dataout <= 8'b0;
        end else begin
            state <= next_state;
            
            // Capture data in State 2 when ready is asserted
            if (state == 2'b10 && ready) begin
                dataout <= datain;
            end
        end
    end

    // Next state logic
    always @(*) begin
        case (state)
            2'b00: begin  // State 0 (Idle)
                if (ready)
                    next_state = 2'b01;
                else
                    next_state = 2'b00;
            end
            2'b01: begin  // State 1 (Pattern Detection)
                if (datain == 8'hF0)
                    next_state = 2'b10;
                else
                    next_state = 2'b00;
            end
            2'b10: begin  // State 2 (Capture)
                if (ready)
                    next_state = 2'b00;
                else
                    next_state = 2'b10;
            end
            2'b11: begin  // State 3 (Unused)
                next_state = 2'b00;
            end
            default:
                next_state = 2'b00;
        endcase
    end

endmodule
