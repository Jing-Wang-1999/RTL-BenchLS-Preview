module origen_054772(
    input [3:0] seconds_ones,
    input [3:0] seconds_tens,
    input [3:0] minutes,
    output reg [6:0] seconds_ones_out,
    output reg [6:0] seconds_tens_out,
    output reg [6:0] minutes_out
);

    // Function to convert BCD to seven-segment encoding
    function [6:0] bcd_to_seven_segment;
        input [3:0] bcd;
        begin
            case (bcd)
                4'b0000: bcd_to_seven_segment = 7'b0000001; // 0
                4'b0001: bcd_to_seven_segment = 7'b1001111; // 1
                4'b0010: bcd_to_seven_segment = 7'b0010010; // 2
                4'b0011: bcd_to_seven_segment = 7'b0000110; // 3
                4'b0100: bcd_to_seven_segment = 7'b1001100; // 4
                4'b0101: bcd_to_seven_segment = 7'b0100100; // 5
                4'b0110: bcd_to_seven_segment = 7'b0100000; // 6
                4'b0111: bcd_to_seven_segment = 7'b0001101; // 7
                4'b1000: bcd_to_seven_segment = 7'b0000000; // 8
                4'b1001: bcd_to_seven_segment = 7'b0000100; // 9
                default: bcd_to_seven_segment = 7'b1111111; // Error/blank
            endcase
        end
    endfunction

    // Assign outputs based on BCD inputs
    always @(*) begin
        seconds_ones_out = bcd_to_seven_segment(seconds_ones);
        seconds_tens_out = (seconds_tens <= 4'b0101) ? bcd_to_seven_segment(seconds_tens) : 7'b1111111;
        minutes_out = bcd_to_seven_segment(minutes);
    end

endmodule
