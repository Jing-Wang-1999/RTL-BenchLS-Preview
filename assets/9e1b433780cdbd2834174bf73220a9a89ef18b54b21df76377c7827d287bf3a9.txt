module mg_01532 (
    input wire clk,
    input wire reset,
    input wire in,
    output reg p
);
    reg [7:0] r;

    always @(posedge clk or posedge reset) begin
        if (reset) begin
            r <= 8'b0;
        end else if (in) begin
            r <= 8'b1;
        end else if (r != 8'b0) begin
            r <= r + 1;
        end
    end

    always @(*) begin
        p = (r == 8'd140);
    end
endmodule
