module origen_066995(
    input CLOCK,
    input [3:0] SWITCHES,
    output reg [3:0] LEDS
);

    reg [31:0] counter;
    reg clk;
    parameter peak = 32'd10000000;

    reg [2:0] state;
    parameter S1 = 3'd1, S2 = 3'd2, S3 = 3'd3, S4 = 3'd4, S5 = 3'd5;

    // Clock divider
    always @(posedge CLOCK) begin
        if (counter == peak - 1) begin
            counter <= 32'd0;
            clk <= ~clk;
        end else begin
            counter <= counter + 1;
        end
    end

    // State machine and LED control
    always @(posedge clk) begin
        case (state)
            S1: begin
                LEDS <= 4'b0001;
                if (SWITCHES == 4'b0001) state <= S5;
                else state <= S2;
            end
            S2: begin
                LEDS <= 4'b0010;
                if (SWITCHES == 4'b0010) state <= S5;
                else state <= S3;
            end
            S3: begin
                LEDS <= 4'b0100;
                if (SWITCHES == 4'b0100) state <= S5;
                else state <= S4;
            end
            S4: begin
                LEDS <= 4'b1000;
                if (SWITCHES == 4'b1000) state <= S5;
                else state <= S1;
            end
            S5: begin
                LEDS <= 4'b1111;
                if (SWITCHES == 4'b0000) state <= S1;
            end
            default: state <= S1;
        endcase
    end

endmodule
