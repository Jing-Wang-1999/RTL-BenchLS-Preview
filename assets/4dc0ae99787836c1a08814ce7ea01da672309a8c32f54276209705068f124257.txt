always @(*) begin
        case (op)
            ALU_AND: y = a & b;
            ALU_OR:  y = a | b;
            ALU_ADD: y = sum;
            ALU_SUB: y = diff;
            ALU_SLT: y = ($signed(a) < $signed(b)) ? 32'b1 : 32'b0;
            ALU_XOR: y = a ^ b;
            ALU_SLL: y = b << shamt;
            ALU_SRL: y = b >> shamt;
            ALU_LUI: y = {b[15:0], 16'b0};
            default: y = 32'b0;
        endcase
    end
