module autovcoder_128283 (
    input clk,
    input reset,
    input [3:0] green_time,
    input [3:0] yellow_time,
    input [3:0] red_time,
    output reg [1:0] state
);

    reg [3:0] counter;
    reg [1:0] next_state;

    // State encodings
    localparam RED    = 2'b00;
    localparam GREEN  = 2'b01;
    localparam YELLOW = 2'b10;

    // State transition and counter logic
    always @(posedge clk or posedge reset) begin
        if (reset) begin
            state <= RED;
            counter <= 4'b0;
        end else begin
            state <= next_state;
            if (next_state != state) begin
                counter <= 4'b0;
            end else begin
                counter <= counter + 1;
            end
        end
    end

    // Next state logic
    always @(*) begin
        case (state)
            RED: begin
                if (counter >= red_time) begin
                    next_state = GREEN;
                end else begin
                    next_state = RED;
                end
            end
            GREEN: begin
                if (counter >= green_time) begin
                    next_state = YELLOW;
                end else begin
                    next_state = GREEN;
                end
            end
            YELLOW: begin
                if (counter >= yellow_time) begin
                    next_state = RED;
                end else begin
                    next_state = YELLOW;
                end
            end
            default: next_state = RED;
        endcase
    end

endmodule
