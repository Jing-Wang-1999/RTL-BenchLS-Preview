SOURCE=Path('/home/jwangjw/project_3rd/ICLR-RTL-BenchLS-backup-jul-29/sources/opencores_git_mirror/communication_controller_i2c_controller_core')
SLAVE=SOURCE/'bench/verilog/i2c_slave_model.v'
BUS=HEAD+'''reg cyc=0,stb=0,we=0;reg [2:0] adr=0;reg [7:0] di=0;wire [7:0] data;wire ack,irq;
wire scl_o,scl_en,sda_o,sda_en;tri1 scl,sda;reg stretch=0,competitor=0;
assign scl=scl_en?1'bz:scl_o;assign scl=stretch?1'b0:1'bz;
assign sda=sda_en?1'bz:sda_o;assign sda=competitor?1'b0:1'bz;
i2c_master_top dut(clk,rst,~rst,adr,di,data,we,stb,cyc,ack,irq,scl,scl_o,scl_en,sda,sda_o,sda_en);
task wr;input [2:0] a;input [7:0] d;begin
@(negedge clk);cyc=1;stb=1;we=1;adr=a;di=d;tick;if(ack!==1)$fatal(1,"write ack");
tick;@(negedge clk);cyc=0;stb=0;we=0;repeat(2)tick;end endtask
task rd;input [2:0] a;output [7:0] d;begin
@(negedge clk);cyc=1;stb=1;we=0;adr=a;tick;if(ack!==1)$fatal(1,"read ack");d=data;
tick;@(negedge clk);cyc=0;stb=0;repeat(2)tick;end endtask
task check;input [2:0] a;input [7:0] expected;reg [7:0] d;begin rd(a,d);if(d!==expected)$fatal(1,"register %d %h != %h",a,d,expected);end endtask
task setup;begin reset;repeat(8)tick;wr(0,31);wr(1,0);wr(2,8'hc0);end endtask
task done;integer n;reg [7:0] status;begin
n=0;status=0;while((!status[0] || status[1]) && n<3000)begin rd(4,status);n=n+1;end
if(!status[0] || status[1] || status[5])$fatal(1,"completion %h",status);
repeat(3)tick;if(irq!==1)$fatal(1,"interrupt");wr(4,1);repeat(3)tick;if(irq!==0)$fatal(1,"irq ack");end endtask
task send;input [7:0] d,command;reg [7:0] status;begin
wr(3,d);wr(4,command);done;rd(4,status);if(status[7])$fatal(1,"slave NACK %h",d);end endtask
reg [7:0] value;
'''


def slave_model():
    s=SLAVE.read_text().replace('`include "timescale.v"','`timescale 1ns/1ps')
    # Upstream diagnostic messages are not pass/fail assertions. Preserve the
    # original privately and suppress them for the exact pass-marker protocol.
    return re.sub(r'\$display\(.*?\);',';',s,flags=re.S)


def registers():
    s=BUS+'initial begin reset;repeat(8)tick;check(0,255);check(1,255);check(2,0);check(5,0);check(6,0);\n'
    rng=random.Random(9060627)
    for a in [0,1,2,3]:
        for _ in range(24):
            v=rng.randrange(256)
            if a==2:v&=0x7f
            s+=f'wr({a},{v});check({5 if a==3 else a},{v});\n'
    s+='wr(4,255);repeat(4)tick;check(6,0);reset;repeat(8)tick;check(0,255);check(1,255);check(2,0);\n'
    return s+END


def transactions(stretching=False):
    s=BUS+"i2c_slave_model #(7'h20) slave(scl,sda);\ninitial begin setup;\n"
    rng=random.Random(9060628)
    for _ in range(5):
        values=[rng.randrange(256) for _ in range(4)]
        s+="send(8'h40,8'h90);send(0,8'h10);\n"
        for i,v in enumerate(values):
            cmd=0x50 if i==3 else 0x10
            if stretching and i==1:
                s+=f"fork begin send(8'h{v:02x},8'h{cmd:02x});end begin @(negedge scl);stretch=1;repeat(500)tick;if(irq!==0)$fatal(1,\"completed while stretched\");@(negedge clk);stretch=0;end join\n"
            else:s+=f"send(8'h{v:02x},8'h{cmd:02x});\n"
        s+="repeat(60)tick;send(8'h40,8'h90);send(0,8'h10);send(8'h41,8'h90);\n"
        for i,v in enumerate(values):
            s+=f"wr(4,8'h{0x68 if i==3 else 0x20:02x});done;check(3,8'h{v:02x});\n"
        s+='repeat(60)tick;\n'
    return s+END+slave_model()


def arbitration():
    return BUS+'''integer timeout;initial begin setup;wr(3,255);
fork begin wr(4,8'h90);end begin @(negedge scl);competitor=1;end join
timeout=0;value=0;while(!value[5] && timeout<3000)begin rd(4,value);timeout=timeout+1;end
if(!value[5] || !value[0])$fatal(1,"arbitration status %h",value);
repeat(20)tick;if(sda_en!==1 || scl_en!==1 || irq!==1)$fatal(1,"lost bus release");
competitor=0;reset;repeat(10)tick;check(4,0);
'''+END


INSTRUCTION='''# Implement the Wishbone I2C master project
Read the original PDF, the output contract and the declaration-only interfaces.
Implement i2c_master_top, i2c_master_byte_ctrl and i2c_master_bit_ctrl in their
required paths. Headers are preprocessed privately to freeze the original default
configuration; provide synthesizable modules without includes or simulation delays.

Clarifications for this pinned release:
- Top wb_rst_i is active-high synchronous. arst_i has configurable active level
  ARST_LVL (default 0, active-low asynchronous). Child nReset is active-low
  asynchronous and rst active-high synchronous. Reset initializes released bus
  lines; pad_o outputs are always zero, pad_oen=1 means release/open-drain high.
- Register addresses 0/1 prescaler low/high (reset FFFF), 2 control (reset zero),
  3 transmit write / receive read, 4 command write / status read. Address 5 reads
  the stored transmit byte, 6 reads the command register, 7 returns zero. Read
  data registers every clock. Wishbone ACK alternates while cyc/stb are held;
  hold writes for two rising edges because write acceptance uses old ACK.
- Control bit 7 enables the engine, bit 6 enables IRQ. Other control bits are
  stored/read. Command writes are ignored while disabled. Command bits are
  START[7], STOP[6], READ[5], WRITE[4], ACK-to-send[3] (1=NACK), IACK[0].
  Bits 7:4 clear on done/arbitration loss, bits 2:1 clear, IACK self-clears.
  Bit 3 persists. Use only valid commands, clearing enable only when idle.
- Status: RXACK[7] (1=NACK), bus-busy[6], arbitration-lost[5], zeros[4:2],
  transfer-in-progress[1], interrupt-flag[0]. Completion/arbitration sets sticky
  flag; command IACK clears it. IRQ is the registered flag AND control[6].
- Serial transfers are 8 bits MSB first plus ACK, with START, repeated START,
  STOP, slave clock stretching and bus arbitration. Prescaler gives approximately
  SCL = wb_clk/(5*(prescaler+1)); this release synchronizes and filters SCL/SDA.
  Losing arbitration releases both lines and clears the command. Busy tracks
  observed START/STOP. A transmitted one observed as zero loses arbitration.
- Byte controller starts a transfer on read or write, optionally emits START,
  shifts eight bits and receives/sends ACK, optionally STOP, then pulses cmd_ack.
  ack_out contains received ACK and dout the received/shifted byte. Hold command
  inputs until completion, then clear them before requesting another operation.
- Bit-controller 4-bit commands: NOP=0, START=1, STOP=2, READ=4, WRITE=8.
  A command emits its bus waveform and pulses cmd_ack. din selects transmitted
  data; dout samples SDA. Handshake/timing follows the byte controller and PDF;
  internal state encodings and pipeline organization need not copy the reference.

Private points check register behavior, full slave memory transactions with
repeated START and ACK/NACK, clock stretching, and arbitration loss. The finite
tests are not exhaustive formal equivalence. Public checks give compilation only.
'''


def i2c_spec():
    return dict(case_id='opencores_i2c_master_v1',project='communication_controller_i2c_controller_core',
        top='i2c_master_top',preprocess=True,instruction=INSTRUCTION,builder_source=__file__,
        original_tests=[str(SLAVE)],
        tests=[('registers','i2c_master_top',registers()),('transactions','i2c_master_top',transactions()),
