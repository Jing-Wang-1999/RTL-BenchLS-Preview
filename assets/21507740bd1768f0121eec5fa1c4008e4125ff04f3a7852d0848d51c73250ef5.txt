module origen_032418 (
    input wire clk,
    input wire reset,
    input wire EXC_flush,
    input wire [31:0] V2_EX,
    input wire [4:0] A3_EX,
    input wire [31:0] ALU_O_EX,
    input wire [31:0] PC4_EX,
    input wire [31:0] PC8_EX,
    output reg [31:0] V2_M,
    output reg [4:0] A3_M,
    output reg [31:0] ALU_O_M,
    output reg [31:0] PC4_M,
    output reg [31:0] PC8_M
);

    reg [31:0] V2_EX_reg;
    reg [4:0] A3_EX_reg;
    reg [31:0] ALU_O_EX_reg;
    reg [31:0] PC4_EX_reg;
    reg [31:0] PC8_EX_reg;

    always @(posedge clk or posedge reset) begin
        if (reset || EXC_flush) begin
            V2_EX_reg <= 32'b0;
            A3_EX_reg <= 5'b0;
            ALU_O_EX_reg <= 32'b0;
            PC4_EX_reg <= 32'b0;
            PC8_EX_reg <= 32'b0;
        end else begin
            V2_EX_reg <= V2_EX;
            A3_EX_reg <= A3_EX;
            ALU_O_EX_reg <= ALU_O_EX;
            PC4_EX_reg <= PC4_EX;
            PC8_EX_reg <= PC8_EX;
        end
    end

    always @(*) begin
        V2_M = V2_EX_reg;
        A3_M = A3_EX_reg;
        ALU_O_M = ALU_O_EX_reg;
        PC4_M = PC4_EX_reg;
        PC8_M = PC8_EX_reg;
    end

    initial begin
        V2_EX_reg = 32'b0;
        A3_EX_reg = 5'b0;
        ALU_O_EX_reg = 32'b0;
        PC4_EX_reg = 32'b0;
        PC8_EX_reg = 32'b0;
    end

endmodule
