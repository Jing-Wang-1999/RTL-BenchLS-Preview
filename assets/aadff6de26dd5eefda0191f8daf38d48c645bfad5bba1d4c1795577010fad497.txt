module origen_027682 (
    input clk,
    input rst_n,
    input [3:0] minute1,
    input [3:0] minute0,
    input [3:0] second1,
    input [3:0] second0,
    output alarm_radio
);

    // Internal signals
    reg F500HZ;
    reg F1KHZ;
    reg [15:0] counter_500hz;
    reg [14:0] counter_1khz;
    
    // Constants for clock division (assuming 50MHz system clock)
    localparam CNT_500HZ_MAX = 16'd49999;  // 50MHz / (500Hz * 2) - 1
    localparam CNT_1KHZ_MAX = 15'd24999;   // 50MHz / (1kHz * 2) - 1
    
    // Clock division for 500Hz
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            counter_500hz <= 16'd0;
            F500HZ <= 1'b0;
        end else begin
            if (counter_500hz == CNT_500HZ_MAX) begin
                counter_500hz <= 16'd0;
                F500HZ <= ~F500HZ;
            end else begin
                counter_500hz <= counter_500hz + 16'd1;
            end
        end
    end
    
    // Clock division for 1kHz
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            counter_1khz <= 15'd0;
            F1KHZ <= 1'b0;
        end else begin
            if (counter_1khz == CNT_1KHZ_MAX) begin
                counter_1khz <= 15'd0;
                F1KHZ <= ~F1KHZ;
            end else begin
                counter_1khz <= counter_1khz + 15'd1;
            end
        end
    end
    
    // Alarm trigger logic
    reg alarm_enable_500hz;
    reg alarm_enable_1khz;
    
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            alarm_enable_500hz <= 1'b0;
            alarm_enable_1khz <= 1'b0;
        end else begin
            // Check for 59:56 or 59:58 (500Hz trigger)
            if (minute1 == 4'd5 && minute0 == 4'd9 && 
                second1 == 4'd5 && (second0 == 4'd6 || second0 == 4'd8)) begin
                alarm_enable_500hz <= 1'b1;
                alarm_enable_1khz <= 1'b0;
            end
            // Check for 00:00 (1kHz trigger)
            else if (minute1 == 4'd0 && minute0 == 4'd0 && 
                     second1 == 4'd0 && second0 == 4'd0) begin
                alarm_enable_500hz <= 1'b0;
                alarm_enable_1khz <= 1'b1;
            end
            // All other times
            else begin
                alarm_enable_500hz <= 1'b0;
                alarm_enable_1khz <= 1'b0;
            end
        end
    end
    
    // Output multiplexer
    assign alarm_radio = (alarm_enable_500hz & F500HZ) | 
                         (alarm_enable_1khz & F1KHZ);

endmodule
