module origen_007818(
    input [1:0] sel,
    input [1:0] a,
    input [1:0] b,
    output reg [1:0] res
);

always @(*) begin
    case(sel)
        2'b00: res = a & b;
        2'b01: res = a | b;
        2'b10: res = a ^ b;
        2'b11: res = ~a;
    endcase
end

endmodule
