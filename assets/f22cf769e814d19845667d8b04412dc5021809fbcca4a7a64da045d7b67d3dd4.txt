module origen_046133(
    input [5:0] instruction,
    output reg [1:0] ALUOp,
    output reg MemRead,
    output reg MemtoReg,
    output reg RegDst,
    output reg Branch,
    output reg BNE,
    output reg ALUSrc,
    output reg MemWrite,
    output reg Jump,
    output reg RegWrite
);

always @(*) begin
    // Default values for control signals
    ALUOp = 2'b00;
    MemRead = 0;
    MemtoReg = 0;
    RegDst = 0;
    Branch = 0;
    BNE = 0;
    ALUSrc = 0;
    MemWrite = 0;
    Jump = 0;
    RegWrite = 0;

    case (instruction)
        6'b000000: begin // R-type
            ALUOp = 2'b10;
            RegDst = 1;
            RegWrite = 1;
        end
        6'b000100: begin // beq
            ALUOp = 2'b01;
            Branch = 1;
        end
        6'b101011: begin // sw
            ALUSrc = 1;
            MemWrite = 1;
        end
        6'b100011: begin // lw
            MemRead = 1;
            MemtoReg = 1;
            ALUSrc = 1;
            RegWrite = 1;
        end
        6'b001000: begin // addi
            ALUSrc = 1;
            RegWrite = 1;
        end
        6'b000010: begin // j
            Jump = 1;
        end
        6'b000101: begin // bne
            ALUOp = 2'b01;
            BNE = 1;
        end
        6'b001111: begin // lui
            ALUSrc = 1;
            RegWrite = 1;
        end
        default: begin
            // All signals are already set to 0 by default
        end
    endcase
end

endmodule
