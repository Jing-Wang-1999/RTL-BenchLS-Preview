def rnd(n,shift):
 if shift<=0:return n<<-shift
 q,r=divmod(n,1<<shift)
 return q+int(r>(1<<(shift-1)) or (r==(1<<(shift-1)) and q&1))
def reference(a,b):
 sa,sb=a>>31,b>>31;ea,eb=(a>>23)&255,(b>>23)&255;ma,mb=a&0x7fffff,b&0x7fffff
 if (ea==255 and ma) or (eb==255 and mb):return 0xffc00000
 if ea==255:
  return ((sb<<31)|0x7fc00000) if eb==255 and sa!=sb else a
 if eb==255:return b
 if a&0x7fffffff==0 and b&0x7fffffff==0:return (sa&sb)<<31
 if a&0x7fffffff==0:return b
 if b&0x7fffffff==0:return a
 xa=(ea-150) if ea else -149;xb=(eb-150) if eb else -149
 if ea:ma|=1<<23
 if eb:mb|=1<<23
 e=min(xa,xb);n=((-1 if sa else 1)*ma<<(xa-e))+((-1 if sb else 1)*mb<<(xb-e))
 if n==0:return 0
 s=int(n<0);n=abs(n);E=n.bit_length()-1+e
 if E < -126:
  q=rnd(n,-149-e)
  return (s<<31)|q
 q=rnd(n,n.bit_length()-24)
 if q==1<<24:q>>=1;E+=1
 if E>127:return (s<<31)|0x7f800000
 return (s<<31)|((E+127)<<23)|(q&0x7fffff)
