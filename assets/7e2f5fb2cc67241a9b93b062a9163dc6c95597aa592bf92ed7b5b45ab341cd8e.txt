def evaluate(rule, output, execution):
    if execution['timeout'] or execution['exit_code']!=0:return False
    if re.search(r'error|failed|fatal|timeout',output,re.I):return False
    if rule=='des':return bool(re.search(r'^PASS$',output,re.M)) and not re.search(r'mismatch|not as expected|FAIL',output,re.I)
    if rule=='cia':return len(re.findall(r'^OK @',output,re.M))==86
    if rule=='core':return len(re.findall(r'Test\s+.*?passed!',output))==65
    if rule=='alu':return bool(re.search(r'All tests passed\.',output,re.I))
    if rule=='ftdi':
        output=re.sub(r'^VCD info:.*\n','',output)
        # Icarus 12 appends source:line $finish diagnostics without a newline
        # after the last upstream $write byte; do not parse its line number.
        prefix=re.match(r'((?:\s+[0-9a-fA-F]{2})+)',output)
        values=prefix.group(1).split() if prefix else []
        return bool(values) and all(int(v,16)==i%256 for i,v in enumerate(values))
    raise ValueError(rule)
