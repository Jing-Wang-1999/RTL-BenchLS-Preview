module origen_005795(
    input pbdiv,
    input clock,
    input [7:0] x,
    input [7:0] y,
    output reg [7:0] q
);

    reg s;
    reg [3:0] C;
    reg [7:0] aq, bq, qq;
    wire [8:0] sum;
    wire go, BLA;

    assign go = (x > 0) & pbdiv;
    assign sum = {1'b0, ~aq} + {1'b0, bq} + 9'b1;
    assign BLA = ~sum[8];

    always @(posedge clock) begin
        if (go & ~s) begin
            s <= 1'b1;
            C <= 4'b0001;
        end
        else if (s) begin
            case (C)
                4'b0001: C <= 4'b0010;
                4'b0010: C <= 4'b0100;
                4'b0100: begin
                    if (BLA) C <= 4'b1000;
                    else C <= 4'b0010;
                end
                4'b1000: begin
                    s <= 1'b0;
                    C <= 4'b0000;
                end
            endcase
        end
    end

    always @(posedge clock) begin
        if (C[0]) begin
            aq <= x;
            qq <= 8'b0;
        end
        else if (C[1]) begin
            if (s == 1'b1 & C == 4'b0010) bq <= y;
            else aq <= sum[7:0];
        end
        else if (C[2] & ~BLA) qq <= qq + 1'b1;
        else if (C[3]) q <= qq;
    end

endmodule
