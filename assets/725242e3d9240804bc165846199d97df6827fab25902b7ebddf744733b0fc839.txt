module mg_02947 (
    input signed [27:0] b_cost_0_i,
    input signed [27:0] b_cost_1_i,
    input signed [27:0] b_cost_2_i,
    input signed [27:0] b_cost_3_i,
    input signed [27:0] b_cost_4_i,
    output reg signed [27:0] b_cost_o,
    output reg [2:0] b_band_o
);

always @(*) begin
    // Initialize minimum cost and band
    b_cost_o = b_cost_0_i;
    b_band_o = 3'd0;

    // Compare with b_cost_1_i
    if (b_cost_1_i < b_cost_o) begin
        b_cost_o = b_cost_1_i;
        b_band_o = 3'd1;
    end

    // Compare with b_cost_2_i
    if (b_cost_2_i < b_cost_o) begin
        b_cost_o = b_cost_2_i;
        b_band_o = 3'd2;
    end

    // Compare with b_cost_3_i
    if (b_cost_3_i < b_cost_o) begin
        b_cost_o = b_cost_3_i;
        b_band_o = 3'd3;
    end

    // Compare with b_cost_4_i
    if (b_cost_4_i < b_cost_o) begin
        b_cost_o = b_cost_4_i;
        b_band_o = 3'd4;
    end
end

endmodule
