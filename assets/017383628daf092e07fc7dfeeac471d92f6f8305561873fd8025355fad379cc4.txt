

































`timescale 1ns / 100ps


 


module dividor (clk, inp, rst, out);
  
  input clk;
  input [32-1:0] inp;
  input rst;



  output  reg [32-1:0] out;


  
 wire [32-1:0]DIVISOR = 32'd360 ;
 
  
  always @ (posedge clk)



   if(rst)
     out <= 8'd0 ;
   else 
    out <= inp % DIVISOR ;
    
    

   
     
 endmodule
