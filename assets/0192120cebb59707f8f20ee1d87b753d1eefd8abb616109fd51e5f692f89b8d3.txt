module origen_092256 (
    input clk,
    input reset,
    input FlushE,
    input [1:0] RegDstD,
    input MemtoRegD,
    input MemWriteD,
    input [1:0] ALUOpD,
    input ALUsrcD,
    input RegWriteD,
    input [4:0] RsD,
    input [4:0] RtD,
    input [4:0] RdD,
    input [31:0] PC8D,
    input [31:0] RD1D,
    input [31:0] RD2D,
    input [31:0] extD,
    input cal_rD,
    input cal_iD,
    input ldD,
    input stD,
    input jalD,
    output reg [1:0] RegDstE,
    output reg MemtoRegE,
    output reg MemWriteE,
    output reg [1:0] ALUOpE,
    output reg ALUsrcE,
    output reg RegWriteE,
    output reg [4:0] RsE,
    output reg [4:0] RtE,
    output reg [4:0] RdE,
    output reg [31:0] PC8E
);

always @(posedge clk or posedge reset) begin
    if (reset) begin
        RegDstE <= 2'b0;
        MemtoRegE <= 1'b0;
        MemWriteE <= 1'b0;
        ALUOpE <= 2'b0;
        ALUsrcE <= 1'b0;
        RegWriteE <= 1'b0;
        RsE <= 5'b0;
        RtE <= 5'b0;
        RdE <= 5'b0;
        PC8E <= 32'b0;
    end else if (FlushE) begin
        RegDstE <= 2'b0;
        MemtoRegE <= 1'b0;
        MemWriteE <= 1'b0;
        ALUOpE <= 2'b0;
        ALUsrcE <= 1'b0;
        RegWriteE <= 1'b0;
        RsE <= 5'b0;
        RtE <= 5'b0;
        RdE <= 5'b0;
        PC8E <= 32'b0;
    end else begin
        RegDstE <= RegDstD;
        MemtoRegE <= MemtoRegD;
        MemWriteE <= MemWriteD;
        ALUOpE <= ALUOpD;
        ALUsrcE <= ALUsrcD;
        RegWriteE <= RegWriteD;
        RsE <= RsD;
        RtE <= RtD;
        RdE <= RdD;
        PC8E <= PC8D;
    end
end

endmodule
