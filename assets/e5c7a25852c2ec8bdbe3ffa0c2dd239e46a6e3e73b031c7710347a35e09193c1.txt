#!/usr/bin/env python3
"""Replay all checked upstream testpoints of an exported candidate case.

--rtl-dir optionally replaces the listed RTL files, preserving relative paths.
Verification/configuration files remain the pinned upstream files. --output must
be a new directory, keeping every invocation's artifacts separate.
"""
import argparse
import hashlib
import json
from pathlib import Path
import shutil
import os
import subprocess
import signal
import time
from review_qualified import evaluate

IV='/home/jwangjw/anaconda3/envs/securitylab/bin/iverilog'
VVP='/home/jwangjw/anaconda3/envs/securitylab/bin/vvp'
VER='/home/jwangjw/anaconda3/envs/securitylab/bin/verilator'

def execute(command,cwd,log,timeout):
    start=time.monotonic()
    with log.open('wb') as f:
        p=subprocess.Popen(command,cwd=cwd,stdout=f,stderr=subprocess.STDOUT,start_new_session=True,
                           env={**os.environ,'IVERILOG_DUMPER':'none'})
        try:rc=p.wait(timeout=timeout);timed=False
        except subprocess.TimeoutExpired:os.killpg(p.pid,signal.SIGKILL);p.wait();rc=None;timed=True
    return {'command':command,'cwd':str(cwd),'log':str(log),'exit_code':rc,'timeout':timed,'seconds':round(time.monotonic()-start,3)}

def main():
    p=argparse.ArgumentParser(description=__doc__);p.add_argument('case',type=Path);p.add_argument('--rtl-dir',type=Path);p.add_argument('--output',type=Path,required=True);args=p.parse_args()
    meta=json.loads((args.case/'meta.json').read_text());source=(args.case/'source').resolve();out=args.output.resolve();out.mkdir(parents=True,exist_ok=False)
    for f,expected in meta['source_sha256'].items():
        if f not in meta['rtl_files'] and hashlib.sha256((source/f).read_bytes()).hexdigest()!=expected:raise RuntimeError('Pinned source checksum mismatch: '+f)
    stage=out/'source';shutil.copytree(source,stage)
    if args.rtl_dir:
        for f in meta['rtl_files']:shutil.copy2(args.rtl_dir/f,stage/f)
    results=[]
    for i,t in enumerate(meta['testpoints']):
        run=out/str(i);run.mkdir();includes=sorted({str(q.parent) for q in stage.rglob('*') if q.is_file()});files=[str(stage/f) for f in t['compiled_files']]
        if t['rule_id']=='alu':
            cmd=[VER,'--binary','--timing','--assert','-Wno-fatal','--top-module',t['tb_top'],'--Mdir',str(run/'obj_dir'),*['-I'+d for d in includes],*files]
            executable=run/'obj_dir'/('V'+t['tb_top']);simcmd=[str(executable)]
        else:
            executable=run/'sim.vvp';cmd=[IV,'-g2001' if t['rule_id']=='ftdi' else '-g2012','-s',t['tb_top'],'-o',str(executable),*[x for d in includes for x in ['-I',d]],*files];simcmd=[VVP,'-n',str(executable)]
        comp=execute(cmd,stage,run/'compile.log',60);res={'testpoint':t['id'],'compile':comp,'passed':False}
        if comp['exit_code']==0 and executable.is_file():
            sim=execute(simcmd,(stage/t['testbench']).parent,run/'simulation.log',20);res.update(simulation=sim,passed=evaluate(t['rule_id'],(run/'simulation.log').read_text(errors='replace'),sim))
        results.append(res)
    report={'case_id':meta['case_id'],'passed_testpoints':sum(t['passed'] for t in results),'total_testpoints':len(results),'testpoints':results};report['score']=report['passed_testpoints']/report['total_testpoints']
    (out/'result.json').write_text(json.dumps(report,indent=2)+'\n');print(json.dumps({k:v for k,v in report.items() if k!='testpoints'}))
    return 0  # completed evaluation; DUT failure is reported in per-point results

if __name__=='__main__':raise SystemExit(main())
