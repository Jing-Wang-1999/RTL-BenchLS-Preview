CLOCKED_BUS_CHECKER = r"""
module wb_b3_check(input clk_i,cyc_i,stb_i,we_i,input[2:0]cti_i,input[1:0]bte_i,input ack_i,err_i,rty_i);
reg previous_cyc=0,previous_stb=0,previous_ack=0,previous_we=0;
reg[2:0]previous_cti=0;reg[1:0]previous_bte=0;
always @(posedge clk_i)begin
if(cyc_i && stb_i)begin
if((ack_i && err_i)||(ack_i && rty_i)||(err_i && rty_i))$fatal(1,"multiple bus terminations");
if(cti_i!==0 && cti_i!==2 && cti_i!==7)$fatal(1,"unsupported CTI");
if(previous_cyc && previous_stb && !previous_ack && (cti_i!==previous_cti || bte_i!==previous_bte || we_i!==previous_we))$fatal(1,"control changed during stalled request");
end
// Pinned master can terminate a partial burst at image completion (ImDone).
// CYC deassertion ends that transfer; this is a compatibility, not conformance case.
previous_cyc<=cyc_i;previous_stb<=stb_i;previous_ack<=ack_i|err_i|rty_i;
previous_cti<=cti_i;previous_bte<=bte_i;previous_we<=we_i;
end
endmodule
"""

def expand(text):
 def inc(m):
  name=m[1]
  p=BENCH/name
  if not p.exists():p=SOURCE/'rtl/verilog'/name
  return expand(p.read_text())
 return re.sub(r'`include\s+"([^"]+)"',inc,text)

def quiet(text):
 # Strip comments before scanning displays; replacing each call with a statement
 # preserves if/else branches. Any original ERROR becomes fatal immediately.
 text=re.sub(r'/\*.*?\*/|//[^\n]*','',text,flags=re.S)
 def display(m):
  args=m[1]
  return '$fatal(1,'+args+');' if 'error' in args.lower() and 'total errors' not in args.lower() else 'begin end'
 text=re.sub(r'\$display\s*\((.*?)\)\s*;',display,text,flags=re.S)
 text=re.sub(r'\$(?:finish|stop)\s*;', '$fatal(1,"upstream early stop");',text)
 return text

def test(task):
 original=(BENCH/'test_bench_top.v').read_text()
 prefix=original.split('initial\n')[0]
 suffix=original[original.index('// Sync Monitor')-len('/////////////////////////////////////////////////////////////////////\n//\n'):]
 # Prefix/suffix keep only declarations, instantiated DUT/models, monitors/tasks.
 body=prefix+f'''initial begin
scen=0;error_cnt=0;clk=0;pclk_i=0;rst=0;int_warn=0;wd_cnt=0;
repeat(20)@(posedge clk);rst=1;repeat(20)@(posedge clk);
{task};
if(error_cnt!=0)$fatal(1,"accumulated errors");
$display("RTL_BENCHLS_PASS");$finish(0);
end
initial begin #500000000;$fatal(1,"test timeout");end
'''+suffix
 # Temporarily protect our success sentinel from quieting original diagnostics.
 body=body.replace('$display("RTL_BENCHLS_PASS");','BENCH_SUCCESS;')
 if task=='ur_test':body=body.replace('#500000000;', '#5000000;')
 body=expand(body)+'\n'+'\n'.join(expand((BENCH/n).read_text()) for n in HELPERS[:-1] if n!='wb_model_defines.v')
 # Old TB packs buffer switches one bit too low; match the pinned register map.
 body=re.sub(r"1'b0,\s*// Reserved\s*1'b1,\s*// CBSWE\s*1'b1,\s*// VBSWE","1'b1, 1'b1, 1'b0,",body)
 # Explicitly type the legacy task port; Icarus rejects scalar input + vector reg.
 body=body.replace('input dly;', 'input [5:0] dly;').replace('reg [5:0] dly;', '')
 # Sample blank only at the settled half-cycle, avoiding obsolete #1 edge races.
 body=body.replace('always @(bval or blanc)\n\t#0.01', 'always @(negedge pclk)')
 # Compact frames preserve all format/polarity/burst cases and FIFO crossings.
 for signal,value in [('thgate','159'),('tvgate','15'),('thlen','thsync+thgdel+thgate+17'),('tvlen','tvsync+tvgdel+tvgate+9')]:
  body=re.sub(r'\b'+signal+r'\s*=\s*\d+\s*;',signal+'='+value+';',body)
 # Cover every format/burst pair once, with distributed ACK delays0..5,
 # instead of the upstream sixfold full cross-product.
 body=body.replace('for(delay=0;delay<6;delay=delay+1)', 'for(delay=0;delay<1;delay=delay+1)')
 body=re.sub(r'(for\(mode=0;mode<4;mode=mode\+1\)\s*begin)',r'\1 s0.set_delay((mode+vbl)%6);',body)
 body=quiet(body).replace('BENCH_SUCCESS;', '$display("RTL_BENCHLS_PASS");')
 # The old event-triggered bus checker compares delayed snapshots against NBA
 # transitions; use settled clock-edge protocol samples instead.
 body=re.sub(r'module wb_b3_check\b.*?endmodule',lambda m:CLOCKED_BUS_CHECKER,body,flags=re.S)
 body=re.sub(r'\btest\b','tb',body);body=re.sub(r'\bint\b','vga_irq',body)
 # Compact buffer-B base0x40000 fits17 address bits; preserve distinct banks.
 body=body.replace('wb_slv #(24)', 'wb_slv #(17)').replace("32'h0040_0000","32'h0004_0000")
 body=body.replace('if(pd !== {red, green, blue} )', 'if((^pd)===1\'bx || (^{red,green,blue})===1\'bx || pd !== {red, green, blue})')
 assert '`include' not in body
 return body
