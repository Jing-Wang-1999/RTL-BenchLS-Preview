module origen_123696 (
    input [6:0] i0,
    input [6:0] i1,
    output [6:0] s
);

    wire [7:0] cig_s;
    reg [7:0] cig_si;
    reg [7:0] cig_di;
    wire [7:0] cig_co;
    wire [6:0] si;
    wire [1:0] ci;

    // Bitwise XOR of i0 and i1
    assign si = i0 ^ i1;

    // Instantiate the two 4-bit full adders
    full_adder_4bit u_cry0 (
        .a(i0[3:0]),
        .b(i1[3:0]),
        .ci(1'b0),
        .s(cig_s[3:0]),
        .co(ci[0])
    );

    full_adder_4bit u_cry1 (
        .a(i0[6:4]),
        .b(i1[6:4]),
        .ci(ci[0]),
        .s(cig_s[6:4]),
        .co(ci[1])
    );

    // Assign the sum output
    assign s = cig_s[6:0];

    // Additional logic for cig_si and cig_di
    always @(*) begin
        cig_si = {1'b0, si};
        cig_di = {7'b0, ci[0]};
    end

    // Generate carry-out signals
    assign cig_co = cig_si + cig_di;

endmodule

// 4-bit Full Adder module
module full_adder_4bit (
    input [3:0] a,
    input [3:0] b,
    input ci,
    output [3:0] s,
    output co
);
    assign {co, s} = a + b + ci;
endmodule
