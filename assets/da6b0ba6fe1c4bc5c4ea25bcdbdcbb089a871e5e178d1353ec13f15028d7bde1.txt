`timescale 1ns/1ps
module tb;
reg clk=0,rst=1,cyc=0,stb=0,we=0,rx=1; always #5 clk=~clk;
reg [1:0]addr=0;reg[31:0]wd=0;reg[3:0]sel=15;
wire stall,ack,tx,rts,ri,ti,rfi,tfi;wire[31:0]rd;
wbuart #(.INITIAL_SETUP(31'h40000010),.LGFLEN(4),.HARDWARE_FLOW_CONTROL_PRESENT(0)) dut(
.i_clk(clk),.i_reset(rst),.i_wb_cyc(cyc),.i_wb_stb(stb),.i_wb_we(we),.i_wb_addr(addr),.i_wb_data(wd),.i_wb_sel(sel),
.o_wb_stall(stall),.o_wb_ack(ack),.o_wb_data(rd),.i_uart_rx(rx),.o_uart_tx(tx),.i_cts_n(1'b0),.o_rts_n(rts),
.o_uart_rx_int(ri),.o_uart_tx_int(ti),.o_uart_rxfifo_int(rfi),.o_uart_txfifo_int(tfi));
task bus;input wr;input[1:0]a;input[31:0]d;input[3:0]s;output[31:0]q;integer n;begin
@(negedge clk);cyc=1;stb=1;we=wr;addr=a;wd=d;sel=s;
@(posedge clk);#1;if(stall!==0)$fatal(1,"unexpected stall");
@(negedge clk);stb=0;n=0;
while(ack!==1 && n<8)begin @(posedge clk);#1;n=n+1;end
if(ack!==1)$fatal(1,"missing ack");q=rd;
@(negedge clk);cyc=0;we=0;repeat(2)@(negedge clk);
end endtask
task receive_tx;input[7:0]expected;integer b;reg[7:0]v;begin
@(negedge tx);#240;for(b=0;b<8;b=b+1)begin v[b]=tx;#160;end
if(v!==expected || tx!==1)$fatal(1,"TX data or stop mismatch");
end endtask
task send_rx;input[7:0]v;integer b;begin
@(negedge clk);rx=0;#160;for(b=0;b<8;b=b+1)begin rx=v[b];#160;end
rx=1;#320;
end endtask
reg[31:0]q;
initial begin
repeat(5)@(negedge clk);rst=0;repeat(200)@(negedge clk);
bus(0,0,0,15,q);if(q!==32'h40000010)$fatal(1,"setup initial mismatch");
bus(1,0,32'h00000014,1,q);bus(0,0,0,15,q);if(q!==32'h40000014)$fatal(1,"setup byte write mismatch");
bus(1,0,32'h40000010,15,q);repeat(200)@(negedge clk);
fork receive_tx(8'ha5);bus(1,3,32'ha5,1,q);join
repeat(20)@(negedge clk);send_rx(8'h69);repeat(20)@(negedge clk);
if(ri!==1)$fatal(1,"RX interrupt absent");bus(0,2,0,15,q);
if(q[7:0]!==8'h69 || q[11:8]!==0)$fatal(1,"RX data/status mismatch");
repeat(10)@(negedge clk);if(ri!==0)$fatal(1,"RX FIFO not popped");
$display("RTL_BENCHLS_PASS");$finish(0);
end
initial begin #2000000;$fatal(1,"timeout");end
endmodule
