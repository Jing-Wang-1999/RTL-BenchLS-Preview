module origen_097092 (
    input [6:0] opcode,
    input reset,
    input invert,
    input [2:0] funct,
    output reg [1:0] im_sel,
    output reg jump,
    output reg branch,
    output reg Alusrc1,
    output reg Alusrc2,
    output reg regWrite,
    output reg MemRead,
    output reg MemWrite,
    output reg jal,
    output reg csr,
    output reg [1:0] wr_sel,
    output reg [2:0] ALUop,
    output wire fence
);

    // FENCE instruction detection (combinatorial)
    assign fence = (opcode == 7'b0001111);

    // Main control logic
    always @(*) begin
        if (reset) begin
            // Reset all outputs to inactive state
            im_sel = 2'b00;
            jump = 1'b0;
            branch = 1'b0;
            Alusrc1 = 1'b0;
            Alusrc2 = 1'b0;
            regWrite = 1'b0;
            MemRead = 1'b0;
            MemWrite = 1'b0;
            jal = 1'b0;
            csr = 1'b0;
            wr_sel = 2'b00;
            ALUop = 3'b000;
        end else begin
            // Default values
            im_sel = 2'b00;
            jump = 1'b0;
            branch = 1'b0;
            Alusrc1 = 1'b0;
            Alusrc2 = 1'b0;
            regWrite = 1'b0;
            MemRead = 1'b0;
            MemWrite = 1'b0;
            jal = 1'b0;
            csr = 1'b0;
            wr_sel = 2'b00;
            ALUop = 3'b000;

            // Decode opcode
            case (opcode)
                // R-type: 0110011
                7'b0110011: begin
                    regWrite = 1'b1;
                    Alusrc2 = 1'b0;
                    im_sel = 2'b00;
                    ALUop = {1'b0, funct[2:1]};
                    wr_sel = 2'b00;
                end

                // I-type: 0010011
                7'b0010011: begin
                    regWrite = 1'b1;
                    Alusrc2 = 1'b1;
                    im_sel = 2'b00;
                    ALUop = {1'b0, funct[2:1]};
                    wr_sel = 2'b00;
                end

                // S-type: 0100011
                7'b0100011: begin
                    Alusrc2 = 1'b1;
                    MemWrite = 1'b1;
                    im_sel = 2'b01;
                    ALUop = 3'b000;
                end

                // B-type: 1100011
                7'b1100011: begin
                    branch = 1'b1;
                    im_sel = 2'b10;
                    ALUop = 3'b001;
                end

                // U-type (LUI): 0110111
                7'b0110111: begin
                    regWrite = 1'b1;
                    Alusrc1 = 1'b1;
                    Alusrc2 = 1'b1;
                    im_sel = 2'b11;
                    ALUop = 3'b000;
                    wr_sel = 2'b00;
                end

                // U-type (AUIPC): 0010111
                7'b0010111: begin
                    regWrite = 1'b1;
                    Alusrc1 = 1'b1;
                    Alusrc2 = 1'b1;
                    im_sel = 2'b11;
                    ALUop = 3'b000;
                    wr_sel = 2'b00;
                end

                // J-type (JAL): 1101111
                7'b1101111: begin
                    regWrite = 1'b1;
                    jump = 1'b1;
                    jal = 1'b1;
                    im_sel = 2'b11;
                    wr_sel = 2'b10;
                end

                // Load instructions (I-type with memory): 0000011
                7'b0000011: begin
                    regWrite = 1'b1;
                    Alusrc2 = 1'b1;
                    MemRead = 1'b1;
                    im_sel = 2'b00;
                    ALUop = 3'b000;
                    wr_sel = 2'b01;
                end

                // CSR instructions: 1110011
                7'b1110011: begin
                    csr = 1'b1;
                    regWrite = 1'b1;
                    wr_sel = 2'b01;
                end

                // Default: maintain all zeros
                default: begin
                    im_sel = 2'b00;
                    jump = 1'b0;
                    branch = 1'b0;
                    Alusrc1 = 1'b0;
                    Alusrc2 = 1'b0;
                    regWrite = 1'b0;
                    MemRead = 1'b0;
                    MemWrite = 1'b0;
                    jal = 1'b0;
                    csr = 1'b0;
                    wr_sel = 2'b00;
                    ALUop = 3'b000;
                end
            endcase
        end
    end

endmodule
