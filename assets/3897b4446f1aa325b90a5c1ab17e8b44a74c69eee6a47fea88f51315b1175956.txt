module autovcoder_112075 (
    a,
    b,
    out,
    carry
);

  parameter WIDTH = 64;

  input [WIDTH-1:0] a, b;
  output [WIDTH-1:0] out;
  output carry;

  wire [WIDTH-1:0] carry_vec;

  genvar i;
  generate
    if (WIDTH == 1) begin
      full_addr fa (
          a,
          b,
          out,
          1'b0,
          carry
      );
    end else begin
      for (i = 0; i < WIDTH; i = i + 1) begin
        if (i == 0) begin
          full_addr fa (
              a[i],
              b[i],
              out[i],
              1'b0,
              carry_vec[i]
          );
        end else begin
          full_addr fa (
              a[i],
              b[i],
              out[i],
              carry_vec[i-1],
              carry_vec[i]
          );
        end
      end
    end
  endgenerate

  assign carry = carry_vec[WIDTH-1];
endmodule
