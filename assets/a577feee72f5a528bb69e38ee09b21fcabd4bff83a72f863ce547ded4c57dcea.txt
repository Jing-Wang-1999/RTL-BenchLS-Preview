module mg_03025 (
    input  wire signed [19:0] i_data,
    output wire signed [26:0] o_data_36,
    output wire signed [26:0] o_data_83
);

    // Multiply by 36 = 32 + 4 = 2^5 + 2^2
    assign o_data_36 = (i_data << 5) + (i_data << 2);

    // Multiply by 83 = 64 + 16 + 2 + 1 = 2^6 + 2^4 + 2^1 + 2^0
    assign o_data_83 = (i_data << 6) + (i_data << 4) + (i_data << 1) + i_data;

endmodule
