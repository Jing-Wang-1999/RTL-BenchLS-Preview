module mg_07533 (
    input wire [4:0] in,
    output reg [5:0] out
);
    reg [3:0] lsb;
    reg [1:0] msb;
    integer i;

    always @* begin
        lsb = 4'b0;
        msb = 2'b0;
        for (i = 4; i >= 0; i = i - 1) begin
            if (lsb >= 5)
                lsb = lsb + 3;
            {msb, lsb} = {msb, lsb, in[i]};
        end
        out = {msb, lsb};
    end
endmodule
