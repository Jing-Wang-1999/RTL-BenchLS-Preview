module mg_03272 (
    input IN0,
    input IN1,
    input CIN,
    output SUM,
    output COUT
);
    // SUM is the XOR of all three inputs
    assign SUM = IN0 ^ IN1 ^ CIN;
    // COUT is the carry-out, which is true if at least two inputs are true
    assign COUT = (IN0 & IN1) | (IN1 & CIN) | (IN0 & CIN);
endmodule
