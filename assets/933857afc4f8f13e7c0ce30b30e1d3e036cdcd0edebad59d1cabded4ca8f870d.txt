module origen_121558 #(
    parameter KER_SIZE = 3,
    parameter BITWIDTH = 8,
    parameter STRIDE = 1,
    parameter NFMAPS = 3,
    parameter INPUT_X_DIM = 28,
    parameter PAD = 1,
    parameter AW = 5
)(
    input wire clk,
    input wire rstn,
    input wire valid,
    output reg [AW-1:0] addr,
    output reg [KER_SIZE-1:0] write_en,
    output reg [KER_SIZE-1:0] read_en,
    output reg ready,
    output reg [KER_SIZE-1:0] top_pad_mask,
    output reg bottom_pad_mask,
    output reg row_is_complete
);

    localparam PADDED_DIM = INPUT_X_DIM + 2*PAD;
    
    // Internal counters
    reg [$clog2(KER_SIZE):0] row_ptr;
    reg [$clog2(PADDED_DIM):0] global_row_ptr;
    reg [$clog2(PADDED_DIM):0] col_ptr;
    reg [$clog2(KER_SIZE):0] init_row_counter;
    
    // Address calculation
    wire [$clog2(PADDED_DIM*PADDED_DIM):0] addr_calc;
    assign addr_calc = (global_row_ptr + row_ptr) * PADDED_DIM + col_ptr;
    
    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            addr <= 0;
            write_en <= 0;
            read_en <= 0;
            ready <= 1;
            top_pad_mask <= {KER_SIZE{1'b1}};
            bottom_pad_mask <= 0;
            row_is_complete <= 0;
            row_ptr <= 0;
            global_row_ptr <= 0;
            col_ptr <= 0;
            init_row_counter <= 0;
        end else begin
            row_is_complete <= 0;
            
            if (valid && ready) begin
                // Generate address
                addr <= addr_calc[AW-1:0];
                
                // Generate write enable - one-hot based on row_ptr
                write_en <= (1 << row_ptr);
                
                // Generate read enable - all active after initialization
                if (init_row_counter >= KER_SIZE - 1) begin
                    read_en <= {KER_SIZE{1'b1}};
                end else begin
                    read_en <= 0;
                end
                
                // Update column pointer
                if (col_ptr == PADDED_DIM - 1) begin
                    col_ptr <= 0;
                    row_is_complete <= 1;
                    
                    // Update row pointer (circular)
                    if (row_ptr == KER_SIZE - 1) begin
                        row_ptr <= 0;
                        // Increment global row pointer
                        if (global_row_ptr < PADDED_DIM - KER_SIZE) begin
                            global_row_ptr <= global_row_ptr + 1;
                        end
                    end else begin
                        row_ptr <= row_ptr + 1;
                    end
                    
                    // Update initialization counter
                    if (init_row_counter < KER_SIZE - 1) begin
                        init_row_counter <= init_row_counter + 1;
                    end
                    
                end else begin
                    col_ptr <= col_ptr + 1;
                end
                
                // Update top_pad_mask
                if (global_row_ptr == 0 && col_ptr == PADDED_DIM - 1) begin
                    if (row_ptr < KER_SIZE - 1) begin
                        top_pad_mask <= top_pad_mask & ~(1 << (row_ptr + 1));
                    end
                end
                
                // Update bottom_pad_mask
                if ((global_row_ptr + row_ptr) >= (PADDED_DIM - 1)) begin
                    bottom_pad_mask <= 1;
                end else begin
                    bottom_pad_mask <= 0;
                end
                
            end else begin
                write_en <= 0;
                if (init_row_counter < KER_SIZE - 1) begin
                    read_en <= 0;
                end
            end
        end
    end

endmodule
