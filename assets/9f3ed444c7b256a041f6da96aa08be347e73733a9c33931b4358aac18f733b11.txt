// Multiplier module
module origen_109386 (
    input [63:0] A,
    input [63:0] B,
    input doSigned,
    output [63:0] mult_low,
    output [63:0] mult_high
);
    wire signed [63:0] A_signed, B_signed;
    wire [127:0] unsigned_result, signed_result;

    assign A_signed = A;
    assign B_signed = B;

    assign unsigned_result = A * B;
    assign signed_result = A_signed * B_signed;

    assign {mult_high, mult_low} = doSigned ? signed_result : unsigned_result;
endmodule

// Shifter module
module shifter (
    input [63:0] value,
    input direction,
    input [5:0] distance,
    output [63:0] result
);
    assign result = direction ? (value >> distance) : (value << distance);
endmodule

// Shifter testbench
module shifter_testbench();
    reg [63:0] value;
    reg direction;
    reg [5:0] distance;
    wire [63:0] result;

    shifter dut (
        .value(value),
        .direction(direction),
        .distance(distance),
        .result(result)
    );

    initial begin
        value = 64'hDEADBEEFDECAFBAD;
        
        // Test left shifts
        direction = 0;
        for (distance = 0; distance < 64; distance = distance + 1) begin
            #10;
            $display("Left shift %d: %h", distance, result);
        end

        // Test right shifts
        direction = 1;
        for (distance = 0; distance < 64; distance = distance + 1) begin
            #10;
            $display("Right shift %d: %h", distance, result);
        end

        $finish;
    end
endmodule

// Multiplier testbench
module mult_testbench();
    reg [63:0] A, B;
    reg doSigned;
    wire [63:0] mult_low, mult_high;

    mult dut (
        .A(A),
        .B(B),
        .doSigned(doSigned),
        .mult_low(mult_low),
        .mult_high(mult_high)
    );

    initial begin
        // Test case 1: Unsigned multiplication
        A = 64'h0000000000000002;
        B = 64'h0000000000000003;
        doSigned = 0;
        #10;
        $display("Unsigned 2 * 3: %h%h", mult_high, mult_low);

        // Test case 2: Signed multiplication
        A = 64'hFFFFFFFFFFFFFFFF; // -1
        B = 64'h0000000000000002;
        doSigned = 1;
        #10;
        $display("Signed -1 * 2: %h%h", mult_high, mult_low);

        // Test case 3: Large unsigned values
        A = 64'hFFFFFFFFFFFFFFFF;
        B = 64'hFFFFFFFFFFFFFFFF;
        doSigned = 0;
        #10;
        $display("Unsigned max * max: %h%h", mult_high, mult_low);

        // Test case 4: Large signed values
        A = 64'h7FFFFFFFFFFFFFFF; // Max positive signed
        B = 64'h8000000000000000; // Min negative signed
        doSigned = 1;
        #10;
        $display("Signed max * min: %h%h", mult_high, mult_low);

        // Test case 5: Zero
        A = 64'h0000000000000000;
        B = 64'hFFFFFFFFFFFFFFFF;
        doSigned = 1;
        #10;
        $display("Signed 0 * -1: %h%h", mult_high, mult_low);

        $finish;
    end
endmodule
