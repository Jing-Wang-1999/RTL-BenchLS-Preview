module mux4to1(
    input [3:0] in,
    input [1:0] select,
    output out
);

    wire sel0_not, sel1_not;
    not(sel0_not, select[0]);
    not(sel1_not, select[1]);
    
    wire and0, and1, and2, and3;
    and(and0, sel1_not, sel0_not, in[0]);
    and(and1, sel1_not, select[0], in[1]);
    and(and2, select[1], sel0_not, in[2]);
    and(and3, select[1], select[0], in[3]);

    or(out, and0, and1, and2, and3);
    
endmodule
