module rtlpp_00240 (
    input ff_clk,
    input ff_D,
    input clk,
    input set,
    input reset,
    output reg ff_Q
);
    always @(posedge clk or posedge set or posedge reset) begin
        if (reset) begin
            ff_Q <= 1'b0;
        end else if (set) begin
            ff_Q <= 1'b1;
        end else begin
            ff_Q <= ff_D;
        end
    end
endmodule
