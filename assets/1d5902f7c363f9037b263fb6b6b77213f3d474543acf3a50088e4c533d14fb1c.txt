module rtlpp_10457 (
    input wire clk,
    input wire reset_n,
    input wire [31:0] A_mul_src1,
    input wire [31:0] A_mul_src2,
    output wire [31:0] A_mul_cell_result
);

    // Split operands into upper and lower 16 bits
    wire [15:0] src1_lower = A_mul_src1[15:0];
    wire [15:0] src1_upper = A_mul_src1[31:16];
    wire [15:0] src2_lower = A_mul_src2[15:0];
    
    // Lower partial product: lower 16 bits of src1 * lower 16 bits of src2
    wire [31:0] part1 = src1_lower * src2_lower;
    
    // Upper partial product: upper 16 bits of src1 * lower 16 bits of src2
    wire [31:0] part2 = src1_upper * src2_lower;
    
    // Combine results:
    // - Lower 16 bits come from part1[15:0]
    // - Upper 16 bits come from part1[31:16] + part2[15:0]
    assign A_mul_cell_result = {part1[31:16] + part2[15:0], part1[15:0]};

endmodule
