module origen_018345 #(
    parameter DBIT = 8,    // Number of data bits
    parameter SB_TICK = 16 // Number of ticks for stop bits
)(
    input wire clk,        // Clock signal
    input wire reset,      // Reset signal
    input wire tx_start,   // Start transmission signal
    input wire s_tick,     // Synchronization tick
    input wire [7:0] din,  // Data input
    output reg tx_done_tick, // Transmission done signal
    output reg tx          // Transmitted data bit
);

    // State encoding
    localparam [1:0]
        IDLE  = 2'b00,
        START = 2'b01,
        DATA  = 2'b10,
        STOP  = 2'b11;

    // Internal registers
    reg [1:0] state_reg, state_next;
    reg [3:0] s_reg, s_next; // Tick counter
    reg [2:0] n_reg, n_next; // Data bit counter
    reg [7:0] b_reg, b_next; // Data buffer
    reg tx_next;

    // State and data registers
    always @(posedge clk or posedge reset) begin
        if (reset) begin
            state_reg <= IDLE;
            s_reg <= 0;
            n_reg <= 0;
            b_reg <= 0;
            tx <= 1'b1;
        end else begin
            state_reg <= state_next;
            s_reg <= s_next;
            n_reg <= n_next;
            b_reg <= b_next;
            tx <= tx_next;
        end
    end

    // Next-state logic
    always @* begin
        // Default assignments
        state_next = state_reg;
        s_next = s_reg;
        n_next = n_reg;
        b_next = b_reg;
        tx_next = tx;
        tx_done_tick = 1'b0;

        case (state_reg)
            IDLE: begin
                tx_next = 1'b1; // Line idle high
                if (tx_start) begin
                    state_next = START;
                    s_next = 0;
                    b_next = din;
                end
            end

            START: begin
                tx_next = 1'b0; // Start bit
                if (s_tick) begin
                    if (s_reg == 15) begin
                        state_next = DATA;
                        s_next = 0;
                        n_next = 0;
                    end else begin
                        s_next = s_reg + 1;
                    end
                end
            end

            DATA: begin
                tx_next = b_reg[0];
                if (s_tick) begin
                    if (s_reg == 15) begin
                        s_next = 0;
                        b_next = b_reg >> 1;
                        if (n_reg == (DBIT - 1)) begin
                            state_next = STOP;
                        end else begin
                            n_next = n_reg + 1;
                        end
                    end else begin
                        s_next = s_reg + 1;
                    end
                end
            end

            STOP: begin
                tx_next = 1'b1; // Stop bit
                if (s_tick) begin
                    if (s_reg == (SB_TICK - 1)) begin
                        state_next = IDLE;
                        tx_done_tick = 1'b1;
                    end else begin
                        s_next = s_reg + 1;
                    end
                end
            end
        endcase
    end

endmodule
