module mg_00577(
    input [4:0] number,
    output reg [6:0] code
);

always @(*) begin
    case (number)
        5'b00000: code = 7'b1000000; // Inverted 7-segment for '0'
        5'b00001: code = 7'b1111001; // Inverted 7-segment for '1'
        5'b00010: code = 7'b0100100; // Inverted 7-segment for '2'
        5'b00011: code = 7'b0110000; // Inverted 7-segment for '3'
        5'b00100: code = 7'b0011001; // Inverted 7-segment for '4'
        5'b00101: code = 7'b0010010; // Inverted 7-segment for '5'
        5'b00110: code = 7'b0000010; // Inverted 7-segment for '6'
        5'b00111: code = 7'b1111000; // Inverted 7-segment for '7'
        5'b01000: code = 7'b0000000; // Inverted 7-segment for '8'
        5'b01001: code = 7'b0010000; // Inverted 7-segment for '9'
        5'b01010: code = 7'b0001000; // Inverted 7-segment for 'A'
        5'b01011: code = 7'b0000011; // Inverted 7-segment for 'b'
        5'b01100: code = 7'b1000110; // Inverted 7-segment for 'C'
        5'b01101: code = 7'b0100001; // Inverted 7-segment for 'd'
        5'b01110: code = 7'b0000110; // Inverted 7-segment for 'E'
        5'b01111: code = 7'b0001110; // Inverted 7-segment for 'F'
        5'b11111: code = 7'b1111111; // Inverted 7-segment for blank (all off)
        default:  code = 7'b0111111; // Inverted 7-segment for top segment on
    endcase
end

endmodule
