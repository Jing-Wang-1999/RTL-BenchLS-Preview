module mg_02274 (
    input [2:0] RX_CLK_COR_CNT,
    input USER_CLK,
    output reg CHANNEL_BOND_LOAD
);

    // Predefined code for comparison
    localparam CHANNEL_BOND_LOAD_CODE = 3'b101;

    always @(posedge USER_CLK) begin
        if (RX_CLK_COR_CNT == CHANNEL_BOND_LOAD_CODE) begin
            CHANNEL_BOND_LOAD <= 1'b1;
        end else begin
            CHANNEL_BOND_LOAD <= 1'b0;
        end
    end

endmodule
