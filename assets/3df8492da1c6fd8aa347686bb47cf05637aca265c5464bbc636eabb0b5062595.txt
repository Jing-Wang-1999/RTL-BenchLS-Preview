module Avg #(parameter BITWIDTH = 8, parameter LENGTH = 4) (
    input signed [BITWIDTH*LENGTH-1:0] data,
    output reg signed [BITWIDTH-1:0] result
);
    integer i;
    reg signed [BITWIDTH-1:0] data_array [0:LENGTH-1];
    reg signed [(BITWIDTH+LENGTH-1):0] sum; // Extra bits to prevent overflow during summation

    always @(*) begin
        // Split the input data into individual integers
        for (i = 0; i < LENGTH; i = i + 1) begin
            data_array[i] = data[i*BITWIDTH +: BITWIDTH];
        end

        // Sum all integers
        sum = 0;
        for (i = 0; i < LENGTH; i = i + 1) begin
            sum = sum + data_array[i];
        end

        // Calculate the average
        result = sum / LENGTH;
    end
endmodule
