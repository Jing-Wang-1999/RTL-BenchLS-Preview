module origen_181245 (
    input clock,
    input reset,
    inout SDA,
    input SCL,
    input [6:0] my_address,
    input [7:0] data_buffer_init
);

    // Synchronize SDA and SCL signals
    reg SDA_sync, SCL_sync;
    reg SDA_last, SCL_last;
    always @(posedge clock or negedge reset) begin
        if (!reset) begin
            SDA_sync <= 1'b1;
            SCL_sync <= 1'b1;
            SDA_last <= 1'b1;
            SCL_last <= 1'b1;
        end else begin
            SDA_sync <= SDA;
            SCL_sync <= SCL;
            SDA_last <= SDA_sync;
            SCL_last <= SCL_sync;
        end
    end

    // Detect start and stop conditions
    wire start_detect = SDA_last & ~SDA_sync & SCL_sync;
    wire stop_detect = ~SDA_last & SDA_sync & SCL_sync;

    // I2C slave state machine
    reg [2:0] state;
    parameter IDLE = 3'b000, START = 3'b001, ADDRESS = 3'b010,
              MODE = 3'b011, START_ACK = 3'b100, END_ACK = 3'b101,
              TX_DATA = 3'b110, GET_ACK = 3'b111, RX_DATA = 3'b101;

    reg [6:0] address;
    reg [7:0] data_buffer;
    reg [7:0] shift_reg;
    reg data_direction;
    reg ack_out;

    always @(posedge clock or negedge reset) begin
        if (!reset) begin
            state <= IDLE;
            address <= 7'b0;
            data_buffer <= data_buffer_init;
            shift_reg <= 8'b0;
            data_direction <= 1'b0;
            ack_out <= 1'b0;
        end else begin
            case (state)
                IDLE: begin
                    if (start_detect) begin
                        state <= START;
                    end
                end
                START: begin
                    state <= ADDRESS;
                end
                ADDRESS: begin
                    if (SCL_sync & ~SCL_last) begin
                        address[6:0] <= {address[5:0], SDA_sync};
                    end
                    if (SCL_last & ~SCL_sync) begin
                        if (address == my_address) begin
                            state <= START_ACK;
                            ack_out <= 1'b0;
                        end else begin
                            state <= IDLE;
                        end
                    end
                end
                START_ACK: begin
                    if (SCL_last & ~SCL_sync) begin
                        state <= MODE;
                        data_direction <= SDA_sync;
                    end
                end
                MODE: begin
                    if (SCL_last & ~SCL_sync) begin
                        if (data_direction) begin
                            state <= TX_DATA;
                            shift_reg <= data_buffer;
                        end else begin
                            state <= RX_DATA;
                            shift_reg <= 8'b0;
                        end
                    end
                end
                TX_DATA: begin
                    if (SCL_sync & ~SCL_last) begin
                        shift_reg <= {shift_reg[6:0], 1'b0};
                    end
                    if (SCL_last & ~SCL_sync) begin
                        state <= GET_ACK;
                        ack_out <= shift_reg[7];
                    end
                end
                GET_ACK: begin
                    if (SCL_last & ~SCL_sync) begin
                        if (SDA_sync) begin
                            state <= END_ACK;
                        end else begin
                            state <= TX_DATA;
                        end
                    end
                end
                RX_DATA: begin
                    if (SCL_sync & ~SCL_last) begin
                        shift_reg <= {shift_reg[6:0], SDA_sync};
                    end
                    if (SCL_last & ~SCL_sync) begin
                        data_buffer <= shift_reg;
                        state <= END_ACK;
                        ack_out <= 1'b0;
                    end
                end
                END_ACK: begin
                    if (SCL_last & ~SCL_sync) begin
                        if (stop_detect) begin
                            state <= IDLE;
                        end else begin
                            state <= START;
                        end
                    end
                end
                default: begin
                    state <= IDLE;
                end
            endcase
        end
    end

    // Drive SDA output
    assign SDA = (state == TX_DATA || state == END_ACK) ? ack_out : 1'bz;

endmodule
