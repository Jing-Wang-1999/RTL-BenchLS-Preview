module mg_00802 (
    input wire Clk,
    input wire x,
    input wire y,
    output reg [1:0] state
);
    reg [1:0] next_state;
    
    always @(posedge Clk) begin
        state <= next_state;
    end
    
    always @* begin
        case ({x, y})
            2'b00: begin
                case (state)
                    2'b00: next_state = 2'b00;
                    2'b01: next_state = 2'b10;
                    2'b10: next_state = 2'b00;
                    2'b11: next_state = 2'b10;
                    default: next_state = 2'b00;
                endcase
            end
            2'b01: begin
                case (state)
                    2'b00: next_state = 2'b00;
                    2'b01: next_state = 2'b11;
                    2'b10: next_state = 2'b00;
                    2'b11: next_state = 2'b11;
                    default: next_state = 2'b00;
                endcase
            end
            2'b10: begin
                case (state)
                    2'b00: next_state = 2'b01;
                    2'b01: next_state = 2'b10;
                    2'b10: next_state = 2'b10;
                    2'b11: next_state = 2'b00;
                    default: next_state = 2'b00;
                endcase
            end
            2'b11: begin
                case (state)
                    2'b00: next_state = 2'b01;
                    2'b01: next_state = 2'b11;
                    2'b10: next_state = 2'b11;
                    2'b11: next_state = 2'b00;
                    default: next_state = 2'b00;
                endcase
            end
            default: next_state = 2'b00;
        endcase
    end
    
    initial begin
        state = 2'b00;
    end
endmodule
