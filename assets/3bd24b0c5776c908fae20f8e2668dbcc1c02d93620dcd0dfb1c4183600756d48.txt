module ALU(ctrl, A, B, shamt, cout, ovf, ze, R);
	input [2:0] ctrl;
	input [31:0] A, B;
	input [4:0] shamt;
	output cout, ovf, ze;
	output [31:0] R;
	
	// ALU operation logic
	reg [31:0] res;
	assign cout = 1'b0;
	assign ovf = 1'b0;
	assign ze = (res == 32'b0);
	assign R = res;

	always @(ctrl or A or B or shamt) begin
		case (ctrl)
			// Example operations
			3'b000: res = A + B; // ADD
			3'b001: res = A - B; // SUB
			3'b010: res = A & B; // AND
			3'b011: res = A | B; // OR
			3'b100: res = A << shamt; // SLL
			3'b101: res = (A < B) ? 32'b1 : 32'b0; // SLT
			default: res = 32'b0;
		endcase
	end
endmodule
