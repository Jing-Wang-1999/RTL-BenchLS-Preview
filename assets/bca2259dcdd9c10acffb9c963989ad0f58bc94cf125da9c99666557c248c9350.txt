module twotofour(a1, en1, f1);
    input en1;
    input [1:0] a1;
    output reg [3:0] f1;

    always @(a1 or en1) begin
        case ({en1, a1})
            3'b000: f1 = 4'b1000;
            3'b001: f1 = 4'b0100;
            3'b010: f1 = 4'b0010;
            3'b011: f1 = 4'b0001;
            default: f1 = 4'b0000;
        endcase
    end
endmodule
