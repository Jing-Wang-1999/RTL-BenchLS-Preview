module autovcoder_028902 (
    input  CLK,
    input  RST,
    input  input_data,
    output output_data
);

  //Load other module(s)

  //Definition for Variables in the module
  reg dff[3:0];

  //Logical
  assign output_data = dff[3];

  always @(posedge CLK, negedge RST) begin
    if (!RST)
    //Reset
    begin
      dff[0] <= 1'b0;
    end else begin
      dff[0] <= input_data;
    end
  end

  always @(posedge CLK, negedge RST) begin
    if (!RST)
    //Reset
    begin
      dff[1] <= 1'b0;
    end else begin
      dff[1] <= dff[0];
    end
  end

  always @(posedge CLK, negedge RST) begin
    if (!RST)
    //Reset
    begin
      dff[2] <= 1'b0;
    end else begin
      dff[2] <= dff[1];
    end
  end

  always @(posedge CLK, negedge RST) begin
    if (!RST)
    //Reset
    begin
      dff[3] <= 1'b0;
    end else begin
      dff[3] <= dff[2];
    end
  end

endmodule
