module origen_064019 (
    input wire clk,
    input wire outsig,
    input wire oe,
    inout wire pad
);

reg sampled_outsig, sampled_oe;
reg next_outsig, next_oe;

always @(posedge clk) begin
    sampled_outsig <= next_outsig;
    sampled_oe <= next_oe;
end

always @* begin
    #1;
    next_outsig = outsig;
    next_oe = oe;
end

assign pad = sampled_oe ? sampled_outsig : 1'bz;

endmodule
