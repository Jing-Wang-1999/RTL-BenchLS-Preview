always @(*) begin
      if (RESET) begin
         next_state = STATE_IDLE;
         ADDRESS_REG = 0;
         READ_REG = 0;
         WRITE_REG = 0;
         WRITEDATA_REG = 0;
         data_read_REG = 0;
         LOCK_REG = 0;
         BEGINTRANSFER_REG = 0;
         BURSTCOUNT_REG = 0;
      end else begin
         case (state)
            STATE_IDLE: begin
               if (start) begin
                  ADDRESS_REG = address_to_access;
                  BYTE_ENABLE_REG = bytes;
                  if (!rnw) begin
                     WRITE_REG = 1'b1;
                     READ_REG = 1'b0;
                     WRITEDATA_REG = data_to_write;
                  end else begin
                     READ_REG = 1'b1;
                     WRITE_REG = 1'b0;
                  end
                  if (WAITREQUEST) begin
                     next_state = STATE_WAIT;
                  end else begin
                     next_state = STATE_DONE;
                  end
               end else begin 
                  next_state = STATE_IDLE;
                  ADDRESS_REG = 0;
                  READ_REG = 0;
                  WRITE_REG = 0;
                  WRITEDATA_REG = 0;
                  data_read_REG = 0;
                  LOCK_REG = 0;
                  BEGINTRANSFER_REG = 0;
                  BURSTCOUNT_REG = 0;
               end
            end 
            
            STATE_WAIT: begin
               if (WAITREQUEST) begin
                  next_state = STATE_WAIT;
               end else begin
                  next_state = STATE_DONE;
               end
            end

            STATE_DONE: begin
               if (rnw) begin
                  data_read_REG = READDATA;
               end
               next_state = STATE_IDLE;
            end

            default: begin
               next_state = STATE_IDLE;
               ADDRESS_REG = 0;
               READ_REG = 0;
               WRITE_REG = 0;
               WRITEDATA_REG = 0;
               data_read_REG = 0;
               LOCK_REG = 0;
               BEGINTRANSFER_REG = 0;
               BURSTCOUNT_REG = 0;
            end
         endcase
      end
   end
