module origen_013659(
    input wire Clk,
    input wire Start,
    input wire [7:0] X,
    output reg [7:0] R,
    output reg Ack
);

    reg [7:0] sir [4:0];
    reg [7:0] transResult;
    reg [1:0] state;
    reg [2:0] counter;
    reg [2:0] transCounter;
    reg [7:0] powerOfFour;

    always @(posedge Clk) begin
        case (state)
            2'b00: begin // Idle state
                if (Start) begin
                    state <= 2'b01;
                    counter <= 3'b000;
                    Ack <= 1'b0;
                end
            end

            2'b01: begin // Input collection state
                sir[counter] <= X;
                counter <= counter + 1;
                if (counter == 3'b011) begin
                    state <= 2'b10;
                    transCounter <= 3'b000;
                    transResult <= 8'b0;
                    powerOfFour <= 8'b1;
                end
            end

            2'b10: begin // Conversion state
                transResult <= transResult + (sir[transCounter] * powerOfFour);
                powerOfFour <= powerOfFour << 2;
                transCounter <= transCounter + 1;
                if (transCounter == 3'b011) begin
                    state <= 2'b11;
                end
            end

            2'b11: begin // Output state
                R <= transResult;
                Ack <= 1'b1;
                state <= 2'b00;
            end
        endcase
    end

endmodule
