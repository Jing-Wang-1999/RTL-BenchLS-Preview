module origen_132439 #(
    parameter id = 0,
    parameter nreads = 128,
    parameter nwrites = 128,
    parameter p = 4
)(
    input sys_clk,
    input sys_rst,
    input [31:0] dat_r,
    input ack,
    output reg [31:0] dat_w,
    output reg [31:0] adr,
    output reg [2:0] cti,
    output reg we,
    output reg [3:0] sel,
    output reg cyc,
    output reg stb,
    output reg tend
);

reg [7:0] read_counter;
reg [7:0] write_counter;
reg [2:0] state;
reg [31:0] rng;

localparam IDLE = 3'b000;
localparam READ = 3'b001;
localparam WRITE = 3'b010;
localparam WAIT_ACK = 3'b011;
localparam DONE = 3'b100;

always @(posedge sys_clk or posedge sys_rst) begin
    if (sys_rst) begin
        read_counter <= 0;
        write_counter <= 0;
        state <= IDLE;
        tend <= 0;
        cyc <= 0;
        stb <= 0;
        we <= 0;
        adr <= 0;
        dat_w <= 0;
        sel <= 0;
        cti <= 0;
        rng <= $random;
    end else begin
        case (state)
            IDLE: begin
                if (read_counter < nreads || write_counter < nwrites) begin
                    rng <= {rng[30:0], rng[31] ^ rng[21] ^ rng[1] ^ rng[0]};
                    if (rng[7:0] < p) begin
                        if (rng[8] && read_counter < nreads) begin
                            state <= READ;
                        end else if (write_counter < nwrites) begin
                            state <= WRITE;
                        end
                    end
                end else begin
                    state <= DONE;
                end
            end

            READ: begin
                cyc <= 1;
                stb <= 1;
                we <= 0;
                adr <= {28'b0, rng[1:0], 2'b00};
                sel <= 4'b1111;
                cti <= 3'b000;
                read_counter <= read_counter + 1;
                state <= WAIT_ACK;
            end

            WRITE: begin
                cyc <= 1;
                stb <= 1;
                we <= 1;
                adr <= {28'b0, rng[1:0], 2'b00};
                dat_w <= rng;
                sel <= 4'b1111;
                cti <= 3'b000;
                write_counter <= write_counter + 1;
                state <= WAIT_ACK;
            end

            WAIT_ACK: begin
                if (ack) begin
                    cyc <= 0;
                    stb <= 0;
                    we <= 0;
                    state <= IDLE;
                end
            end

            DONE: begin
                tend <= 1;
            end

            default: state <= IDLE;
        endcase
    end
end

endmodule
