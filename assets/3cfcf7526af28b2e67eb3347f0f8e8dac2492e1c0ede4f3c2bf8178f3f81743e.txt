"""Pinned 31-pin GPIO configuration, with independent register/pin scoreboards."""
import random
from case_specs_uartbus import HEAD, END

BUS=HEAD+'''reg cyc=0,stb=0,we=0;reg [7:0] adr=0;reg [31:0] di=0;reg [3:0] sel=15;
wire [31:0] data;wire ack,err,irq;reg [30:0] aux=0,pins=0;reg eclk=0;wire [30:0] outputs,oe;
gpio_top dut(clk,rst,cyc,adr,di,sel,we,stb,data,ack,err,irq,aux,pins,outputs,oe,eclk);
task wr;input [7:0] a;input [31:0] d;input [3:0] mask;begin
@(negedge clk);adr=a;di=d;sel=mask;cyc=1;stb=1;we=1;tick;
if(ack!==1 || err!==0)$fatal(1,"write ack");
@(negedge clk);cyc=0;stb=0;we=0;repeat(3)tick;end endtask
task rd;input [7:0] a;input [31:0] expected;begin
@(negedge clk);adr=a;sel=15;cyc=1;stb=1;we=0;tick;
if(ack!==1 || err!==0 || data!==expected)$fatal(1,"read %h: %h != %h",a,data,expected);
@(negedge clk);cyc=0;stb=0;repeat(3)tick;end endtask
'''


def registers():
    s=BUS+'initial begin reset;repeat(5)tick;\n'
    for a in range(0,40,4):s+=f'rd({a},0);\n'
    rng=random.Random(9060623)
    for a in [4,8,12,16,20,32,36]:
        v=0
        for _ in range(20):
            d=rng.getrandbits(32);mask=rng.randrange(16)
            bits=sum(255<<(8*i) for i in range(4) if mask>>i&1)
            v=((v&~bits)|(d&bits))&0x7fffffff
            s+=f"wr({a},32'h{d:08x},4'h{mask:x});rd({a},32'h{v:08x});\n"
    s+='wr(28,32\'h76543210,0);rd(28,32\'h76543210);wr(24,2,0);rd(24,2);reset;repeat(5)tick;\n'
    for a in range(0,40,4):s+=f'rd({a},0);\n'
    return s+END


def outputs():
    s=BUS+'initial begin reset;repeat(5)tick;\n';rng=random.Random(9060624)
    for _ in range(80):
        o,a,m,e=[rng.getrandbits(31) for _ in range(4)]
        expected=(o&~m)|(a&m)
        s+=f"wr(4,31'h{o:x},15);wr(8,31'h{e:x},15);wr(20,31'h{m:x},15);@(negedge clk);aux=31'h{a:x};tick;"
        s+=f"if(outputs!==31'h{expected:x} || oe!==31'h{e:x})$fatal(1,\"pin outputs\");\n"
    return s+END


def inputs():
    s=BUS+'initial begin reset;repeat(5)tick;\n';rng=random.Random(9060625)
    for _ in range(60):
        d=rng.getrandbits(31)
        s+=f"@(negedge clk);pins=31'h{d:x};repeat(4)tick;rd(0,31'h{d:x});\n"
    return s+END


def interrupts():
    s=BUS+'initial begin reset;repeat(5)tick;\n'
    for positive in [True,False]:
        for bit in [0,7,8,15,23,24,30]:
            mask=1<<bit;before=0 if positive else mask;after=mask if positive else 0
            s+=f"wr(24,0,15);@(negedge clk);pins=31'h{before:x};repeat(6)tick;wr(28,0,15);wr(12,31'h{mask:x},15);wr(16,31'h{mask if positive else 0:x},15);wr(24,1,15);"
            s+='if(irq!==0)$fatal(1,"premature interrupt");'
            s+=f"@(negedge clk);pins=31'h{after:x};repeat(8)tick;rd(28,31'h{mask:x});rd(24,3);if(irq!==1)$fatal(1,\"interrupt missing\");"
            s+='wr(28,0,15);wr(24,1,15);repeat(4)tick;rd(28,0);rd(24,1);if(irq!==0)$fatal(1,"interrupt clear");\n'
    return s+END


def external_clock():
    s=BUS+'initial begin reset;repeat(5)tick;wr(32,32\'h7fffffff,15);\n'
    rng=random.Random(9060626)
    for negative in [False,True]:
        s+=f"wr(36,31'h{0x7fffffff if negative else 0:x},15);\n"
        for _ in range(16):
            d=rng.getrandbits(31)
            s+=f"@(negedge clk);eclk={int(negative)};repeat(8)tick;@(negedge clk);pins=31'h{d:x};repeat(8)tick;"
            s+=f"@(negedge clk);eclk={int(not negative)};repeat(8)tick;rd(0,31'h{d:x});"
            s+=f"@(negedge clk);pins=~pins;repeat(8)tick;rd(0,31'h{d:x});\n"
    return s+END


def bus_errors():
    s=BUS+'initial begin reset;repeat(5)tick;\n'
    for a in [1,2,3,0x40,0x80,0xc4]:
        s+=f'@(negedge clk);adr={a};cyc=1;stb=1;we=0;tick;if(err!==1 || ack!==0)$fatal(1,"address error");@(negedge clk);cyc=0;stb=0;repeat(3)tick;\n'
    s+='@(negedge clk);adr=0;cyc=1;stb=1;repeat(4)begin tick;if(ack!==1)$fatal(1,"ack high");tick;if(ack!==0)$fatal(1,"ack low");end\n'
    return s+END


INSTRUCTION='''# Implement the configured GPIO project
Read the original PDF and the fixed interface declarations. This case uses the
pinned default 31-pin configuration: gw=31, dw=32, aw=8; one required gpio_top
module. The original defines header is resolved into numeric ports/parameters;
implement the configured functionality in the required RTL file without includes.
This complete peripheral has one RTL module; it is not a missing-module task.

Configuration and timing clarifications:
- Active-high asynchronous reset clears all registers, output enables and
  synchronized samples. Inputs and external clock each use two synchronizer
  flops. All functional state in this configuration uses rising wb_clk_i.
- Byte addresses: IN=0x00, OUT=0x04, OE=0x08, INTE=0x0C, PTRIG=0x10,
  AUX=0x14, CTRL=0x18, INTS=0x1C, ECLK=0x20, NEC=0x24. Bit 31 reads zero
  for 31-bit registers; CTRL is two bits. Other aligned addresses within 0x00..3F
  read IN. Address bits 7:6 and 1:0 must be zero; invalid addresses return error.
- Registered read data updates every clock. A valid cyc/stb access gets registered
  ACK on the next edge; held accesses alternate ACK high/low. Errors are also
  registered alternating pulses. Register writes occur on valid cyc/stb/we,
  regardless of old ACK. OUT/OE/INTE/PTRIG/AUX/ECLK/NEC use wb_sel byte masks.
  CTRL and INTS writes replace all implemented bits regardless of byte mask.
- OUT bits drive pins when corresponding AUX=0, else aux_i drives them.
  ext_pad_o is registered; ext_padoe_o directly reflects OE, one=output enabled.
  IN samples synchronized pad inputs every clock when ECLK=0 for that bit.
- ECLK=1 selects samples latched at the synchronized external-clock edge:
  NEC=0 selects rising, NEC=1 falling. Selected external-clock samples hold
  between edges. Allow synchronization latency; external clock is much slower
  than the Wishbone clock. IN itself adds a register after that selection.
- CTRL[0] globally enables interrupts; INTE enables each bit. PTRIG=1 selects
  a rising input transition, zero a falling transition. With global enable,
  matching input events set sticky INTS bits; writes replace INTS (zero clears).
  IRQ is registered (any INTS) AND CTRL[0]. CTRL[1] latches IRQ while enabled;
  software clears it by writing CTRL. CTRL[1] does not itself drive IRQ.

Private points test byte masks/reset, output/auxiliary selection, synchronized
input sampling, positive/negative interrupt behavior, external-clock sampling,
and bus acknowledgement/errors. The original header and implementation are private.
'''
