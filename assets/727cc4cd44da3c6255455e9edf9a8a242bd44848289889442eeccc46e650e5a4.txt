module autovcoder_088563
#(
    parameter DATA_BITWIDTH = 8,
    parameter CODE_BITWIDTH = 16,
    parameter ADDR_BITWIDTH = 16,
    parameter CADDR_NOP = 2'b00,
    parameter CADDR_INC = 2'b01,
    parameter CADDR_MOD = 2'b10,
    parameter CADDR_SET = 2'b11
)
(
    input  wire                         clk,
    input  wire                         rst_n,
    input  wire [1:0]                   flag_op_caddr,
    input  wire [CODE_BITWIDTH-1:0]     code,
    input  wire [DATA_BITWIDTH-1:0]     data,
    input  wire                         dbg_clk,
    output reg  [ADDR_BITWIDTH-1:0]     code_addr,
    output wire                         dbg_local_f_pn,
    output wire                         dbg_local_f_mem,
    output wire                         dbg_local_f_lh
);

    wire [11:0] _inst12;
    wire [10:0] _inst11;
    wire [7:0]  _inst8;
    wire        _f_pn;
    wire        _f_mem;
    wire        _f_lh;

    assign _inst12 = code[15:4];
    assign _inst11 = code[14:4];
    assign _inst8  = code[11:4];
    assign _f_pn   = code[15];
    assign _f_mem  = code[13];
    assign _f_lh   = code[12];

    assign dbg_local_f_pn  = _f_pn;
    assign dbg_local_f_mem = _f_mem;
    assign dbg_local_f_lh  = _f_lh;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            code_addr <= {ADDR_BITWIDTH{1'b0}};
        end else begin
            case (flag_op_caddr)
                CADDR_NOP: begin
                    code_addr <= code_addr;
                end
                CADDR_INC: begin
                    code_addr <= code_addr + 1'b1;
                end
                CADDR_MOD: begin
                    code_addr <= code_addr + {{(ADDR_BITWIDTH-8){_inst8[7]}}, _inst8};
                end
                CADDR_SET: begin
                    code_addr <= {{(ADDR_BITWIDTH-12){_inst12[11]}}, _inst12};
                end
                default: begin
                    code_addr <= code_addr;
                end
            endcase
        end
    end

endmodule
