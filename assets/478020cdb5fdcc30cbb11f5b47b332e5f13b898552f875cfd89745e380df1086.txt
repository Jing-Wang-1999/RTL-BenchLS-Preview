module origen_148613 #(
    parameter PIPE_PIPELINE_STAGES = 0
)(
    input wire pipe_tx_rcvr_det_i,
    input wire pipe_tx_reset_i,
    input wire pipe_tx_rate_i,
    input wire pipe_tx_deemph_i,
    input wire [2:0] pipe_tx_margin_i,
    input wire pipe_tx_swing_i,
    input wire pipe_clk,
    input wire rst_n,
    output reg pipe_tx_rcvr_det_o,
    output reg pipe_tx_reset_o,
    output reg pipe_tx_rate_o,
    output reg pipe_tx_deemph_o,
    output reg [2:0] pipe_tx_margin_o,
    output reg pipe_tx_swing_o

    // Internal registers for pipeline stages
    reg pipe_tx_rcvr_det_stage1;
    reg pipe_tx_reset_stage1;
    reg pipe_tx_rate_stage1;
    reg pipe_tx_deemph_stage1;
    reg [2:0] pipe_tx_margin_stage1;
    reg pipe_tx_swing_stage1;

    reg pipe_tx_rcvr_det_stage2;
    reg pipe_tx_reset_stage2;
    reg pipe_tx_rate_stage2;
    reg pipe_tx_deemph_stage2;
    reg [2:0] pipe_tx_margin_stage2;
    reg pipe_tx_swing_stage2;

    always @(posedge pipe_clk or negedge rst_n) begin
        if (!rst_n) begin
            // Reset all pipeline stages
            pipe_tx_rcvr_det_stage1 <= 0;
            pipe_tx_reset_stage1 <= 0;
            pipe_tx_rate_stage1 <= 0;
            pipe_tx_deemph_stage1 <= 0;
            pipe_tx_margin_stage1 <= 3'b000;
            pipe_tx_swing_stage1 <= 0;

            pipe_tx_rcvr_det_stage2 <= 0;
            pipe_tx_reset_stage2 <= 0;
            pipe_tx_rate_stage2 <= 0;
            pipe_tx_deemph_stage2 <= 0;
            pipe_tx_margin_stage2 <= 3'b000;
            pipe_tx_swing_stage2 <= 0;
        end else begin
            // Pipeline stage 1
            pipe_tx_rcvr_det_stage1 <= pipe_tx_rcvr_det_i;
            pipe_tx_reset_stage1 <= pipe_tx_reset_i;
            pipe_tx_rate_stage1 <= pipe_tx_rate_i;
            pipe_tx_deemph_stage1 <= pipe_tx_deemph_i;
            pipe_tx_margin_stage1 <= pipe_tx_margin_i;
            pipe_tx_swing_stage1 <= pipe_tx_swing_i;

            // Pipeline stage 2
            pipe_tx_rcvr_det_stage2 <= pipe_tx_rcvr_det_stage1;
            pipe_tx_reset_stage2 <= pipe_tx_reset_stage1;
            pipe_tx_rate_stage2 <= pipe_tx_rate_stage1;
            pipe_tx_deemph_stage2 <= pipe_tx_deemph_stage1;
            pipe_tx_margin_stage2 <= pipe_tx_margin_stage1;
            pipe_tx_swing_stage2 <= pipe_tx_swing_stage1;
        end
    end

    always @(*) begin
        case (PIPE_PIPELINE_STAGES)
            0: begin
                pipe_tx_rcvr_det_o = pipe_tx_rcvr_det_i;
                pipe_tx_reset_o = pipe_tx_reset_i;
                pipe_tx_rate_o = pipe_tx_rate_i;
                pipe_tx_deemph_o = pipe_tx_deemph_i;
                pipe_tx_margin_o = pipe_tx_margin_i;
                pipe_tx_swing_o = pipe_tx_swing_i;
            end
            1: begin
                pipe_tx_rcvr_det_o = pipe_tx_rcvr_det_stage1;
                pipe_tx_reset_o = pipe_tx_reset_stage1;
                pipe_tx_rate_o = pipe_tx_rate_stage1;
                pipe_tx_deemph_o = pipe_tx_deemph_stage1;
                pipe_tx_margin_o = pipe_tx_margin_stage1;
                pipe_tx_swing_o = pipe_tx_swing_stage1;
            end
            2: begin
                pipe_tx_rcvr_det_o = pipe_tx_rcvr_det_stage2;
                pipe_tx_reset_o = pipe_tx_reset_stage2;
                pipe_tx_rate_o = pipe_tx_rate_stage2;
                pipe_tx_deemph_o = pipe_tx_deemph_stage2;
                pipe_tx_margin_o = pipe_tx_margin_stage2;
                pipe_tx_swing_o = pipe_tx_swing_stage2;
            end
            default: begin
                pipe_tx_rcvr_det_o = pipe_tx_rcvr_det_i;
                pipe_tx_reset_o = pipe_tx_reset_i;
                pipe_tx_rate_o = pipe_tx_rate_i;
                pipe_tx_deemph_o = pipe_tx_deemph_i;
                pipe_tx_margin_o = pipe_tx_margin_i;
                pipe_tx_swing_o = pipe_tx_swing_i;
            end
        endcase
    end

endmodule
