module rtlpp_06605
(
    input [11:0] din,
    output [11:0] dout,
    input wen,
    input ren,
    input reset,
    output empty,
    output full,
    input clk
);

    reg [11:0] mem [255:0];
    reg [7:0] waddr;
    reg [7:0] raddr;
    reg [7:0] raddrr;
    reg empty;
    reg full;

    always @(posedge clk or posedge reset) 
    begin
        if (reset) 
        begin
            waddr <= 8'b0;
            raddr <= 8'b0;
            empty <= 1'b1;
            full <= 1'b0;
        end
        else 
        begin
            if (wen && (!full)) 
            begin
                mem[waddr] <= din;
                waddr <= waddr + 1;
            end
    
            if (ren && (!empty)) 
            begin
                raddrr <= raddr;
                raddr <= raddr + 1;
            end
    
            full <= (waddr + 1) == raddr;
            empty <= waddr == raddr;
        end
    end
    
    assign dout = mem[raddrr];

endmodule
