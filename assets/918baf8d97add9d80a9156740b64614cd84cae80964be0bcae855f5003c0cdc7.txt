module mg_07367 (
    input pla_ph1,
    input pla_ph2,
    input dsp_reset,
    input [7:0] padd,
    output reg [7:0] out2_r
);

    reg [7:0] stage1_reg;

    always @(posedge pla_ph1 or posedge dsp_reset) begin
        if (dsp_reset) begin
            stage1_reg <= 8'b0;
        end else begin
            stage1_reg <= padd;
        end
    end

    always @(posedge pla_ph2 or posedge dsp_reset) begin
        if (dsp_reset) begin
            out2_r <= 8'b0;
        end else begin
            out2_r <= stage1_reg;
        end
    end

endmodule
