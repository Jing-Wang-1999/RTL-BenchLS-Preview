def mul(a,b):
    out=0
    for _ in range(8):
        if b&1:out^=a
        a=((a<<1)^ (0x11b if a&128 else 0))&255;b>>=1
    return out


def sbox(x):
    inv=0 if x==0 else 1
    if x:
        for _ in range(254):inv=mul(inv,x)
    return inv ^ ((inv<<1)|(inv>>7))&255 ^ ((inv<<2)|(inv>>6))&255 ^ ((inv<<3)|(inv>>5))&255 ^ ((inv<<4)|(inv>>4))&255 ^ 0x63


SBOX=[sbox(i) for i in range(256)]
assert SBOX[0]==0x63 and SBOX[0x53]==0xed


def sub(x,width=16):return int.from_bytes(bytes(SBOX[v] for v in x.to_bytes(width,'big')),'big')
def shift(x):
    a=x.to_bytes(16,'big');return int.from_bytes(bytes(a[4*((c+r)%4)+r] for c in range(4) for r in range(4)),'big')
def mix(x):
    a=x.to_bytes(16,'big');out=[]
    for c in range(4):
        b=a[4*c:4*c+4]
        for r in range(4):out.append(mul(2,b[r])^mul(3,b[(r+1)%4])^b[(r+2)%4]^b[(r+3)%4])
    return int.from_bytes(bytes(out),'big')
def next_key(x,rcon):
    w=[x>>(32*i)&0xffffffff for i in [3,2,1,0]]
    t=sub(((w[3]<<8)|(w[3]>>24))&0xffffffff,4)^rcon
    out=[]
    for v in w:t^=v;out.append(t)
    return sum(v<<(32*(3-i)) for i,v in enumerate(out))
def keys(x):
    result=[];rc=1
    for _ in range(10):x=next_key(x,rc<<24);result.append(x);rc=mul(rc,2)
    return result
def encrypt(x,k):
    e=Cipher(algorithms.AES(k.to_bytes(16,'big')),modes.ECB()).encryptor()
    return int.from_bytes(e.update(x.to_bytes(16,'big'))+e.finalize(),'big')

def pipeline_tb(module,latency,fn,extra='',instance='',width=128,seed=9030620):
    s=HEAD+f'reg valid=0;reg [{width-1}:0] d=0;wire [{width-1}:0] q;wire v;\n'+extra
    s+=instance or f'{module} dut(clk,rst,valid,d,v,q);\n'
    s+='initial begin reset;\n';rng=random.Random(seed);history=[];last=0
    for n in range(192+latency):
        valid=int(n<192 and n%11!=5);d=rng.getrandbits(width)
        history.append((valid,fn(d)))
        ev,eq=history[n-latency+1] if n>=latency-1 else (0,0)
        if ev:last=eq
        s+=f"@(negedge clk);valid={valid};d={width}'h{d:x};tick;if(v!==1'b{ev} || q!=={width}'h{last:x})$fatal(1,\"{module} cycle {n}\");\n"
    return s+END


def sbox_tb():
    s=HEAD+'reg en=0;reg [7:0] d=0;wire [7:0] q;SBox dut(clk,rst,en,d,q);initial begin reset;\n'
    for d in range(256):
        s+=f"@(negedge clk);en=1;d=8'h{d:02x};tick;if(q!==8'h{SBOX[d]:02x})$fatal(1,\"sbox\");"
        s+=f"@(negedge clk);en=0;d=~d;tick;if(q!==8'h{SBOX[d]:02x})$fatal(1,\"hold\");\n"
    return s+END


def key_exp_tb():
    s=HEAD+'reg valid=0;reg [127:0] key=0;wire [1279:0] w;wire [9:0] v;KeyExpantion dut(clk,rst,valid,key,w,v);initial begin reset;\n'
    rng=random.Random(9030621);history=[];held=[0]*10
    for n in range(120):
        v=int(n<72 and n%7!=3);k=rng.getrandbits(128);history.append((v,keys(k)))
        s+=f"@(negedge clk);valid={v};key=128'h{k:032x};tick;"
        for r in range(10):
            idx=n-4*(r+1)+1;ev=history[idx][0] if idx>=0 else 0
            if ev:held[r]=history[idx][1][r]
            s+=f"if(v[{r}]!==1'b{ev} || w[{1279-r*128}-:128]!==128'h{held[r]:032x})$fatal(1,\"keyexp {n} round {r}\");"
        s+='\n'
    return s+END


def top_tb(reset_mid=False):
    s=HEAD+'''reg dv=0,kv=0;reg [127:0] data=0,key=0;wire valid;wire [127:0] q;
Top_PipelinedCipher dut(clk,rst,dv,kv,key,data,valid,q);
initial begin reset;
'''
    rng=random.Random(9030622+int(reset_mid));history=[];held=0
    for n in range(180):
        if reset_mid and n==67:
            s+='@(negedge clk);dv=0;kv=0;reset;\n';history=[];held=0
        dv=int(n<130 and n%9!=2);kv=int(n<130 and n%13!=7)
        data=rng.getrandbits(128);key=rng.getrandbits(128)
        if n==0:data=0x00112233445566778899aabbccddeeff;key=0x000102030405060708090a0b0c0d0e0f
        history.append((dv&kv,encrypt(data,key)))
        ev,eq=history[-41] if len(history)>=41 else (0,0)
        if ev:held=eq
        s+=f"@(negedge clk);dv={dv};kv={kv};data=128'h{data:032x};key=128'h{key:032x};tick;"
        s+=f"if(valid!==1'b{ev} || q!==128'h{held:032x})$fatal(1,\"top cycle {n}: v=%b data=%h\",valid,q);\n"
    return s+END

Read specification/project.txt and the original project PDF. Implement all nine
modules/files in output_contract.json with the hierarchy described in the PDF.
The integration top is Top_PipelinedCipher. Fix DATA_W=KEY_L=128, NO_ROUNDS=10;
SubBytes also supports its 32-bit use inside key generation. Keep parameter
names/defaults and every declared interface. No decryption is required.

Public timing clarifications, reviewed before experiments:
- All state uses rising clk and asynchronous ACTIVE-LOW reset, despite the
  ambiguous reset port name. Reset clears valid and registered data outputs.
  With invalid input, each stage clears its valid flag and holds stored data.
- SBox is the AES byte substitution, registered when valid_in/en is high.
  SubBytes applies SBox independently to each byte and has one register stage.
  ShiftRows and MixColumns each have one register stage. Interpret the most
  significant byte as AES state byte 0; bytes 0..3 are column 0. ShiftRows rotates
  row r left by r columns; MixColumns uses the standard 02/03/01/01 matrix over
  GF(2^8), irreducible polynomial x^8+x^4+x^3+x+1. AES SBox is multiplicative
  inverse (zero maps to zero) followed by the standard affine transform/0x63.
- AddRoundKey is one register stage: valid_out=data_valid_in AND key_valid_in;
  update data_out with data_in XOR round_key only when both inputs are valid.
- Round chains SubBytes, ShiftRows, MixColumns and AddRoundKey (four stages).
  Its round_key/key_valid_in are sampled by the last stage, not at the input
  stage; external standalone tests hold the round key stable and valid.
- RoundKeyGen has four register stages. RotWord rotates the low 32-bit word
  left by one byte; SubWord uses the AES SBox. New word 0 is old word 0 XOR
  SubWord(RotWord(old word 3)) XOR RCON_Word; words 1..3 XOR the respective
  old word with the previous new word. RCON_Word remains constant for the
  transaction. KeyExpantion chains ten RoundKeyGen blocks with RCON high bytes
  01,02,04,08,10,20,40,80,1b,36. W[1279:1152] is round key 1, W[127:0] key 10;
  valid_out[r-1] marks round key r after 4*r register stages.
- Top XORs the initial key, performs nine normal rounds, then SubBytes,
  ShiftRows, one delay register and final AddRoundKey. Latency is 41 register
  stages: a pair accepted at rising edge n appears after edge n+40. Throughput
  is one pair each clock. Each valid pair can have a different key. Both input
  valid flags must be high together to accept a pair; holes propagate without
  stale valid results. Reset discards all in-flight pairs. The final data holds
  its previous value when output valid is low.

Private points test SBox exhaustively, SubBytes at 128/32 bits, ShiftRows,
MixColumns, AddRoundKey, Round, RoundKeyGen, KeyExpantion, streaming encryption,
and reset during streaming. Algorithm vectors are independently generated;
public feedback only compiles. Save the actual RTL files and finish.
'''
