//--------------------------------------------------------------------------------------------------------
// Module  : uart2axi4
// Type    : synthesizable
// Standard: Verilog 2001 (IEEE1364-2001)
// Function: convert UART command to AXI4 read/write action
//--------------------------------------------------------------------------------------------------------
module uart2axi4 #(
    
    parameter  CLK_FREQ   = 50000000,     
    
    parameter  BAUD_RATE  = 115200,       
    parameter  PARITY     = "NONE",       
    
    parameter  BYTE_WIDTH = 2,            
    parameter  A_WIDTH    = 32            
) (
    input  wire                    rstn,
    input  wire                    clk,
    
    input  wire                    awready,  
    output wire                    awvalid,
    output wire      [A_WIDTH-1:0] awaddr,
    output wire             [ 7:0] awlen,
    input  wire                    wready,   
    output wire                    wvalid,
    output wire                    wlast,
    output wire [8*BYTE_WIDTH-1:0] wdata,
    output wire                    bready,   
    input  wire                    bvalid,
    input  wire                    arready,  
    output wire                    arvalid,
    output wire      [A_WIDTH-1:0] araddr,
    output wire             [ 7:0] arlen,
    output wire                    rready,   
    input  wire                    rvalid,
    input  wire                    rlast,
    input  wire [8*BYTE_WIDTH-1:0] rdata,
    
    input  wire                    i_uart_rx,
    output wire                    o_uart_tx
);
endmodule
