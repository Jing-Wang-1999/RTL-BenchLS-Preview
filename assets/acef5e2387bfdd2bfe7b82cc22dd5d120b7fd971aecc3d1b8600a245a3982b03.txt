module sky130_fd_sc_hs__a32o_4 (
    input  wire A1,
    input  wire A2,
    input  wire A3,
    input  wire B1,
    input  wire B2,
    input  wire VPWR,
    input  wire VGND,
    output wire X
);

    wire and1_out;
    wire and2_out;

    // First AND gate with three inputs: A1, A2, A3
    assign and1_out = A1 & A2 & A3;

    // Second AND gate with two inputs: B1, B2
    assign and2_out = B1 & B2;

    // OR gate combining the outputs of the two AND gates
    assign X = and1_out | and2_out;

endmodule
