module uart_msg_handler (  
        reset_n ,
        sys_clk ,


    
        tx_data_avail,
        tx_rd,
        tx_data,
         

    
        rx_ready,
        rx_wr,
        rx_data,

      
        reg_addr,
        reg_wr,  
        reg_wdata,
        reg_req,
	reg_ack,
	reg_rdata

     );
input        reset_n               ;
input        sys_clk               ;
output         tx_data_avail        ;
output [7:0]   tx_data              ;
input          tx_rd                ;
output         rx_ready            ;
input [7:0]    rx_data             ;
input          rx_wr               ;
output  [15:0] reg_addr           ;
output  [31:0]  reg_wdata          ;
output         reg_req            ;
output         reg_wr             ;
input          reg_ack            ;
input   [31:0] reg_rdata          ;
endmodule
